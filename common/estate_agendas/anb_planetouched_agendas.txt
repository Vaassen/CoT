

#Burghers - Agenda To Stop the Wind
estate_burghers_planetouched_wind = {
	can_select = {
		NOT = { naval_forcelimit = 0.9 }
		any_owned_province = { NOT = { has_trade_building_trigger = yes } has_port = yes }
		dip_tech = 4
	}
	selection_weight = {
		factor = 1
	}
	pre_effect = {
		random_owned_province = {
			limit = { NOT = { has_trade_building_trigger = yes } has_port = yes }
			save_event_target_as = agenda_province
		}
		event_target:agenda_province = {
			set_variable = {
				which = province_dip_var
				value = 0
			}
			export_to_variable = {
				which = province_dip_var
				value = trigger_value:base_production
			}
			change_variable = {
				which = province_dip_var
				value = 2
			}
		}
	}
	immediate_effect = {
	}
	on_invalid = {
	}
	task_requirements = {
		naval_forcelimit = 0.9
		event_target:agenda_province = {
			calc_true_if = {
				amount = 5
				num_of_units_in_province = {
					who = owner
					type = heavy_ship
					amount = 5
				}
				num_of_units_in_province = {
					who = owner
					type = light_ship
					amount = 5
				}
				num_of_units_in_province = {
					who = owner
					type = galley
					amount = 5
				}
				num_of_units_in_province = {
					who = owner
					type = transport
					amount = 5
				}
			}
		}
	}
	task_completed_effect = {
		create_admiral = {
			tradition = 40
		}
		add_navy_tradition = 10
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = 10
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = -10
		}
		set_country_flag = planetouched_agenda_bad
		planetouched_bad_agenda_counter_scripted_effect = yes
	}
	invalid_trigger = {
		event_target:agenda_province = {
			NOT = { owned_by = ROOT }
		}
	}
	failing_effect = {
		create_admiral = {
			tradition = 40
		}
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = -15
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = 10
		}
		set_country_flag = planetouched_agenda_good
		planetouched_good_agenda_counter_scripted_effect = yes
	}
	fail_if = {
		event_target:agenda_province = {
			has_trade_building_trigger = yes
			base_production = "province_dip_var"
		}
	}
}
