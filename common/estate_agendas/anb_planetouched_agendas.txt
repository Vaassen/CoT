

#Burghers - Agenda To Stop the Wind
estate_burghers_planetouched_wind = {
	can_select = {
		has_estate = estate_planetouched
		num_of_ports = 1
		NOT = { navy_size_percentage = 0.9 }
		any_owned_province = { NOT = { has_trade_building_trigger = yes } has_port = yes }
		dip_tech = 4
	}
	selection_weight = {
		factor = 1
	}
	pre_effect = {
		random_owned_province = {
			limit = {
				NOT = { has_trade_building_trigger = yes }
				has_port = yes
			}
			save_event_target_as = agenda_province
		}
		event_target:agenda_province = {
			set_variable = {
				which = province_dip_var
				value = 0
			}
			export_to_variable = {
				which = province_dip_var
				value = trigger_value:base_production
			}
			change_variable = {
				which = province_dip_var
				value = 2
			}
		}
	}
	immediate_effect = {
	}
	on_invalid = {
	}
	task_requirements = {
		navy_size_percentage = 0.9
		OR = {
			heavy_ships_in_province = 5
			light_ships_in_province = 5
			galleys_in_province = 5
			transports_in_province = 5
		}
	}
	task_completed_effect = {
		create_admiral = {
			tradition = 40
		}
		add_navy_tradition = 10
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = 10
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = -10
		}
		planetouched_bad_agenda_counter_scripted_effect = yes
	}
	invalid_trigger = {
		event_target:agenda_province = {
			NOT = { owned_by = ROOT }
		}
	}
	failing_effect = {
		create_admiral = {
			tradition = 40
		}
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = -15
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = 10
		}
		planetouched_good_agenda_counter_scripted_effect = yes
	}
	fail_if = {
		event_target:agenda_province = {
			has_trade_building_trigger = yes
			base_production = "province_dip_var"
		}
	}
}

#Burghers - Agenda an Accursed in the Court

	can_select = {
		has_estate = estate_planetouched

		has_dip_advisor = yes

	selection_weight = {
		factor = 1
	}

		if = {
			limit = { has_dip_advisor_4 = yes }

		}
		else_if = {
			limit = { has_dip_advisor_3 = yes }

			limit = { has_dip_advisor_2 = yes }

		}

	}
	on_invalid = {
		clr_country_flag = planetouched_has_dip_advisor_2
		clr_country_flag = planetouched_has_dip_advisor_3
		clr_country_flag = planetouched_has_dip_advisor_4
	}
	task_requirements = {
		NOT = { has_dip_advisor = yes }
	}
	task_completed_effect = {
		random_list = {
			1 = {
				define_advisor = {
					type = colonial_governor
					cost_multiplier = 0.5
				}
			}
			1 = {
				define_advisor = {
					type = diplomat
					cost_multiplier = 0.5
				}
			}
			1 = {
				define_advisor = {
					type = naval_reformer
					cost_multiplier = 0.5
				}
			}
			1 = {
				define_advisor = {
					type = navigator
					cost_multiplier = 0.5
				}
			}
			1 = {
				define_advisor = {
					type = spymaster
					cost_multiplier = 0.5
				}
			}
			1 = {
				define_advisor = {
					type = statesman
					cost_multiplier = 0.5
				}
			}
			1 = {
				define_advisor = {
					type = trader
					cost_multiplier = 0.5
				}
			}
		}
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = 10
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = -10
		}
		planetouched_bad_agenda_counter_scripted_effect = yes
		clr_country_flag = planetouched_has_dip_advisor_2
		clr_country_flag = planetouched_has_dip_advisor_3

	}
	failing_effect = {
		add_country_modifier = {
			name = estate_planetouched_advisor_burghers
			duration = 3650 #10 years
		}
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = -15
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = 10
		}
		planetouched_good_agenda_counter_scripted_effect = yes
		clr_country_flag = planetouched_has_dip_advisor_2
		clr_country_flag = planetouched_has_dip_advisor_3
		clr_country_flag = planetouched_has_dip_advisor_4
	}
	fail_if = {
		if = {
			limit = { has_country_flag = planetouched_has_dip_advisor_4 }
			has_dip_advisor_5 = yes
		}
		if = {
			limit = { has_country_flag = planetouched_has_dip_advisor_3 }
			has_dip_advisor_4 = yes
		}
		if = {

		}
		else = {
			has_dip_advisor_2 = yes
		}
	}
}

#Burghers - Agenda the Accursed of the other realms
estate_burghers_planetouched_accursed = {
	can_select = {
		has_estate = estate_planetouched
		any_neighbor_country = {
			NOT = {
				is_at_war = yes
				is_subject = yes
				has_opinion_modifier = {
					who = ROOT
					modifier = improved_relation
					value = 60
				}
			}
		}
	}
	selection_weight = {
		factor = 1
	}
	pre_effect = {
		random_neighbor_country = {
			limit = {
				NOT = {
					is_at_war = yes
					is_subject = yes
					has_opinion_modifier = {
						who = ROOT
						modifier = improved_relation
						value = 60
					}
				}
			}
			save_event_target_as = agenda_country
		}
	}
	immediate_effect = {
	}
	on_invalid = {
	}
	task_requirements = {
		event_target:agenda_country = {
			has_opinion_modifier = {
				who = ROOT
				modifier = improved_relation
				value = 90
			}
		}
	}
	task_completed_effect = {
		add_country_modifier = {
			name = estate_burghers_planetouched_accursed_mod
			duration = 3650 #10 years
		}
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = 10
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = -10
		}
		planetouched_bad_agenda_counter_scripted_effect = yes
	}
	invalid_trigger = {
		event_target:agenda_country = {
			OR = {
				exists = no
				is_subject = yes
			}
		}
	}
	failing_effect = {
		random_owned_province = {
			limit = { NOT = { development = 10 } }
			add_random_development = 1
		}
		add_estate_loyalty = {
			estate = estate_burghers
			loyalty = -15
		}
		add_estate_loyalty = {
			estate = estate_planetouched
			loyalty = 10
		}
		planetouched_good_agenda_counter_scripted_effect = yes
	}
	fail_if = {
		has_spy_network_in = {
			who = event_target:agenda_country
			value = 50
		}
	}
}
