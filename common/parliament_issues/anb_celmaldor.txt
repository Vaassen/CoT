
z22_expand_acts_of_deland_centaur = {

	category = 4

	allow = {
		has_government_attribute = z22_acts_of_deland_attribute
		any_owned_province = {
			has_any_centaur_pop_trigger = yes
		}
		NOT = {
			z22_acts_centaur = yes
		}
	}

	effect = {
		z22_add_race_to_deland_acts_effect = {
			race = centaur
		}
		change_government_reform_progress = -25

		largest_increase_of_centaur_tolerance_effect = yes
	}

	modifier = {
		diplomatic_reputation = 1
	}

	chance = {
		factor = 100000
	}

	ai_will_do = {
		factor = 100
	}

}

z22_support_our_league = {
	category = 2
	allow = {
		has_reform = z22_silver_diet
	}

	effect = {
		add_dip_power = -50
		add_years_of_income = -5
		every_known_country = {
			limit = {
				OR = {
					is_giesadh_or_trade_league_member = yes
					transfers_trade_power_to = ROOT
				}
			}
			if = {
				limit = {
					NOT = {
						monthly_income = 5
					}
				}
				add_treasury = 150
			}
			else = {
				add_years_of_income = 5
			}
			add_favors = {
				who = ROOT
				amount = 20
			}
			add_country_modifier = {
				name = z22_league_support
				duration = 3650
			}
		}
	}

	modifier = {
		trade_efficiency = 0.1
	}


	ai_will_do = {
		factor = 10
	}
}

z22_unification_of_league = {
	category = 2

	allow = {
		mission_completed = z22_greater_reach
		any_known_country = {
			OR = {
				is_giesadh_or_trade_league_member = yes
				is_subject_of = ROOT
			}
			if = {
				limit = {
					NOT = {
						mission_completed = z22_toward_unification
					}
				}
				capital_scope = {
					province_group = western_alenic_reach
				}
			}
			else = {
				capital_scope = {
					superregion = gerudia_superregion
				}
			}
			trust = {
				who = ROOT
				value = 90
			}
		}
	}

	effect = {
		ROOT = {
			change_government_reform_progress = -15
		}
		random_known_country = {
			limit = {
				OR = {
					is_giesadh_or_trade_league_member = yes
					is_subject_of = ROOT
				}
				if = {
					limit = {
						NOT = {
							mission_completed = z22_toward_unification
						}
					}
					capital_scope = {
						province_group = western_alenic_reach
					}
				}
				else = {
					capital_scope = {
						superregion = gerudia_superregion
					}
				}
				trust = {
					who = ROOT
					value = 90
				}
			}
			custom_tooltip = z22_eligible_integration_tt
			every_owned_province = {
				ROOT = {
					add_dip_power = -10
				}
			}
			ROOT = {
				inherit = PREV
				country_event = {
					id = 610 #annexation event
				}
			}
		}
	}

	modifier = {
		global_autonomy = -0.025
	}

	ai_will_do = {
		factor = 100
	}
}
