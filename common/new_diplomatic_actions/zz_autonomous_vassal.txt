cancel_autonomous_vassal =  {
	category = influence
	require_acceptance = no
	is_visible = {
		overlord_of = FROM
		FROM = {
			OR = {
				is_subject_of_type = autonomous_vassal
				AND = {
					ROOT = {
						has_reform = karassk_laushan_city
						has_country_flag = L68_unlock_tribs_to_vassals
					}
					FROM = {
						is_subject_of_type = karassk_idzarshin_trib_subject
					}
				}
			}
		}
		NOT = {
			AND = {
				FROM = { tag = Z44 } ##Aelnar can't make Silblaidd a normal vassal
				tag = Z43 #aelnar
			}
			AND = {
				FROM = { tag = R81 } ##Jade March can't be made into a regular vassal
				tag = R62 #the command
			}
			AND = {
				FROM = { 
					AND = {
						OR = {
							tag = G55
							tag = U27
						}
						has_country_modifier = G55_degakheioni_funding
					}
				}
				AND = {
					OR = {
						tag = G46
						tag = H22
					}
					has_country_modifier = G55_for_degakheion_funding_eneion
				}
			}
		}
	}
	
	is_allowed = {
		hidden_trigger = {
			overlord_of = FROM
			FROM = {
				OR = {
					is_subject_of_type = autonomous_vassal
					AND = {
						ROOT = {
							has_reform = karassk_laushan_city
							has_country_flag = L68_unlock_tribs_to_vassals
						}
						FROM = {
							is_subject_of_type = karassk_idzarshin_trib_subject
						}
					}
				}
			}
			NOT = {
				AND = {
					FROM = { tag = Z44 } ##Aelnar can't make Silblaidd a normal vassal
					tag = Z43
				}
			}
		}
		FROM = {
			is_subject_of_type = autonomous_vassal
			NOT = { liberty_desire = 50 }
		}
		reverse_has_opinion = {
			who = FROM
			value = 0
		}
		FROM = {
			OR = {
				is_subject_of_type = autonomous_vassal
				is_subject_of_type = karassk_idzarshin_trib_subject
			}
			is_subject_of = ROOT		
		}
		stability = 0
	}
	
	on_accept = {
		if = {
			limit = {
				ROOT = {
					has_reform = karassk_laushan_city
					has_country_flag = L68_unlock_tribs_to_vassals
				}
			}
			ROOT = {
				set_variable = { which = totalkaratribDev value = 0 }
				export_to_variable = { which = tempkaratriDev value = total_development who = FROM }
				change_variable = { which = totalkaratribDev which = tempkaratriDev }
			}
			while = { limit = { check_variable = { which = totalkaratribDev value = 1 } }
				add_dip_power = -2
				change_variable = { which = totalkaratribDev value = -1 }
			}
		}
		add_trust = {
			who = FROM
			value = -20
			mutual = yes
		}
		if = {
			limit = {
				OR = {
					NOT = {
						reverse_has_opinion = {
							who = FROM
							value = 0
						}
					}
					FROM = { liberty_desire = 50 }
				}
			}
			FROM = { grant_independence = yes }
			add_truce_with = FROM
		}
		else = {
			create_subject = {
				subject_type = vassal
				subject = FROM  # dk check
			}
			FROM = {
				add_liberty_desire = 65
			}
		}
		if = {
			limit = {
				NOT = { full_idea_group = diplomatic_ideas has_reform = goblin_shadow_chancellor } #reduced_stab_impacts = yes
			}
			add_stability = -1
		}

	}
	
	ai_will_do = {
		stability = 2 # Need to have the stab for it
		NOT = { vassal = 2 } # Gotta have a small number of other subjects, so you don't make em all disloyal
		FROM = { # They must have low liberty desire
			liberty_desire = 0
		}
	}
}
