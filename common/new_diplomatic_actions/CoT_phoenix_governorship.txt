# Create new diplomatic actions like this. AI will currently never send them.
cot_create_governorship = {
    category = influence          
  
    
	alert_index = 40
	alert_tooltip = demand_dummy_alert_tooltip
	
    require_acceptance = no
 
    is_visible = {
		is_subject = no
		FROM = {
			is_subject_of_type = vassal
			is_subject_of = ROOT		
		}
		if = {
			limit = { tag = H78 } #seghdihr
			FROM = { NOT = { culture_is_dwarven = yes } }
		}
		if = {
			limit = { tag = H73 } #skewered drake
			FROM = { NOT = { tag = M01 } } #grizehud
		}
        OR = {
			primary_culture = sun_elf
			primary_culture = desert_elf
			primary_culture = sunrise_elf
		}
    }
    
    # Determines when the action can be taken
    is_allowed = {
        OR = {
			primary_culture = sun_elf
			primary_culture = desert_elf
			primary_culture = sunrise_elf
		}
		FROM = { is_subject_of = ROOT }
        # variable_arithmetic_trigger can be used here to use export_to_variable to extract and compare values such as ruler_age, etc
    }
    
    # Which effects to execute on acceptance
    on_accept = {
        create_subject = {
			subject_type = cot_phoenix_governorship
			subject = FROM
		}
    }
    
    # Controls the AI's usage of the action
    ai_will_do = {
		always = no
    }
}
