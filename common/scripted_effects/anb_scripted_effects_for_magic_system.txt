
# Country flags and ruler flags are ENTIRELY separate things.
# You can set a country flag named "A" and a ruler flag named "A" and they will be tracked separately.
# A ruler flag named "A" will NOT fulfill the requirements for has_country_flag = A,
# and the clr_country_flag effect will NOT clear ruler flags.

# The same, however, is NOT true of modifiers.
# A ruler modifier "A" WILL fulfill the requirements for has_country_modifier = A.
# It can be removed with the remove_country_modifier effect (there is no remove_ruler_modfier effect).
# A country modifier will NOT count as a ruler modifier for the purposes of has_ruler_modifier trigger,
# but you ALSO cannot use the add_ruler_modifier effect in any meaningful way while you have a country modifier of the same name.

#monthly mana calc, updates for all countries with mages

	update_mana_regen_vars = yes

		which = currentMana

#for if mana exceeds mana capacity
cap_mana = {
	if = {
		limit = {
			NOT = {
				check_variable = { currentMana = 0 }
			}
		}
		set_variable = { currentMana = 0 }
		# log = "[This.GetName] has Mana out of bounds (currenaMana = [This.currentMana.GetValue]). Please contact Devs."
	}
	export_to_variable = {
		variable_name = manaCapacity
		value = modifier:mages_mana_capacity
	}
	if = {
		limit = {
			check_variable = {
				which = currentMana
				which = manaCapacity
			}
		}
		set_variable = {
			which = currentMana
			which = manaCapacity
		}
	}
}

#for spells, MT effects, etc. use this. e.g. change_mana = { amount = 10 } to increase it by 10.
change_mana = {
	change_variable = {
		which = currentMana
		value = $amount$
	}
	hidden_effect = {
		cap_mana = yes
		update_mana_bar = yes
	}
}

update_mana_bar = {
	#This prevents the framevar going negative and then showing 100% mana
	if = { limit = { check_variable = { which = currentMana value = 0 } }
		set_variable = { which = current_mana_percent which = currentMana }
		divide_variable = { which = current_mana_percent which = manaCapacity }
		multiply_variable = { current_mana_percent = 100 }
	}
	else = {
		set_variable = { current_mana_percent = 0 }
	}
}

set_magic_experience_target = {
	[[infrastructure] set_country_flag = infra_temp_flag ]
	[[xptarget] set_country_flag = xptarget_temp_flag ]
	
	# [[!infrastructure]] [[!xptarget]]
	if = {
		limit = {
			NOT = {
				has_country_flag = infra_temp_flag
				has_country_flag = xptarget_temp_flag
			}
		}
		set_variable = { country_experience_target = 100 } #Base experience
		set_variable = { expFoo = 250 } 				#Increase per level
		## Maybe target level will be set elsewhere as a variable
		#For now:
		[[study]
			trigger_switch = {
				on_trigger = has_$type$_flag
				$study$_1 = { multiply_variable = { expFoo = 2 } } # Lvl 1 to 2
				$study$_2 = { multiply_variable = { expFoo = 3 } } # Lvl 2 to 3
			}
		]
		[[project]
			if = {
				limit = { check_variable = { which = $project$_level value = 2 } }
				multiply_variable = { expFoo = 3 }
			}
			else_if = {
				limit = { check_variable = { which = $project$_level value = 1 } }
				multiply_variable = { expFoo = 2 }
			}
		]
		change_variable = { which = magic_experience_target which = expFoo }
	}
	# [[infrastructure]]
	else_if = {
		limit = {
			has_country_flag = infra_temp_flag
		}
		set_variable = { magic_experience_target = 400 } #Base experience
		trigger_switch = {
			on_trigger = has_country_modifier
			magical_infrastructure_1 = { multiply_variable = { magic_experience_target = 1 } } # Lvl 1 to 2
			magical_infrastructure_2 = { multiply_variable = { magic_experience_target = 2 } } # Lvl 2 to 3
			magical_infrastructure_3 = { multiply_variable = { magic_experience_target = 3 } } # Lvl 3 to 4
			magical_infrastructure_4 = { multiply_variable = { magic_experience_target = 4 } } # Lvl 4 to 5
		}
	}
	#[[xptarget]]
	else_if = {
		limit = {
			has_country_flag = xptarget_temp_flag
		}
		set_variable = { which = magic_experience_target value = $xptarget$ }
	}
	clr_country_flag = infra_temp_flag
	clr_country_flag = xptarget_temp_flag
	update_advancement_progress_bar = { type = $type$ }
}

	}
}

passive_magic_advancement_progress = {
	export_to_variable = {
		variable_name = monthly_magic_xp_gain
		value = modifier:mages_experience_gain
	}
	export_to_variable = {
		variable_name = ruler_experience_modifier
		value = modifier:mages_ruler_experience_mod
	}
	export_to_variable = {
		variable_name = country_experience_modifier
		value = modifier:mages_estate_experience_mod
	}
	set_variable = { temp_exp = 0 }

	if = { limit = { has_country_flag = magic_artificery_mixed }
		multiply_variable = { which = monthly_magic_xp_gain value = 0.8 }
	}
	
	if = {
		limit = {
			check_variable = { passive_experience_bank = 1 }
			has_ongoing_magical_advancement = yes
		}
		set_variable = { which = expFoo which = monthly_magic_xp_gain }
		multiply_variable = { expFoo = 0.5 }
		if = { limit = { check_variable = { which = passive_experience_bank which = expFoo } }
			change_variable = { which = temp_exp which = expFoo }
			subtract_variable = { which = passive_experience_bank which = expFoo }
		}
		else = {
			change_variable = { which = temp_exp which = passive_experience_bank }
			subtract_variable = { which = passive_experience_bank which = temp_exp }
		}
	}

	if = { limit = { has_ruler_flag = magic_advancement }
		if = { limit = { is_long_lived_ruler = yes }
			multiply_variable = { which = monthly_magic_xp_gain value = 0.5 }
			multiply_variable = { which = temp_exp value = 0.5 }
		}
		calc_school_experience_modifiers = { type = ruler }
		change_variable = { which = ruler_experience_modifier which = school_experience_modifier }
		if = { limit = {
				NOT = { is_variable_equal = { which = ruler_experience_modifier value = 0 } } # XP modifier not 0
			}
			set_variable = { which = expFoo which = ruler_experience_modifier }
			change_variable = { which = ruler_experience_modifier which = magic_ruler_xp_mod_legacy } ##Legacy
			change_variable = { expFoo = 1 }
			multiply_variable = { which = monthly_magic_xp_gain which = expFoo }
		}
		change_variable = { which = monthly_magic_xp_gain which = temp_exp }
		change_variable = {
			which = ruler_magic_experience
			which = monthly_magic_xp_gain
		}
		
		update_advancement_progress_bar = { type = ruler }
	}
	
	else_if = {
		limit = { has_heir_flag = magic_advancement }
		if = { limit = { is_long_lived_heir = yes }
			multiply_variable = { which = monthly_magic_xp_gain value = 0.5 }
			multiply_variable = { which = temp_exp value = 0.5 }
		}
		calc_school_experience_modifiers = { type = heir }
		change_variable = { which = ruler_experience_modifier which = school_experience_modifier }
		if = { limit = {
				NOT = { is_variable_equal = { which = ruler_experience_modifier value = 0 } } # XP modifier not 0
			}
			set_variable = { which = expFoo which = ruler_experience_modifier }
			change_variable = { which = ruler_experience_modifier which = magic_ruler_xp_mod_legacy } ##Legacy
			change_variable = { expFoo = 1 }
			multiply_variable = { which = monthly_magic_xp_gain which = expFoo }
		}
		change_variable = { which = monthly_magic_xp_gain which = temp_exp }
		change_variable = {
			which = heir_magic_experience
			which = monthly_magic_xp_gain
		}
		update_advancement_progress_bar = { type = heir }
	}

	else_if = { limit = { has_country_flag = magic_advancement }
		divide_variable = { which = monthly_magic_xp_gain value = 3 }
		divide_variable = { which = temp_exp value = 3 }
		calc_school_experience_modifiers = { type = country }
		change_variable = { which = country_experience_modifier which = school_experience_modifier }
		if = {
			limit = {
				NOT = { is_variable_equal = { which = country_experience_modifier value = 0 } } # XP modifier not 0
			}
			set_variable = { which = expFoo which = country_experience_modifier }
			change_variable = { which = country_experience_modifier which = magic_estate_xp_mod_legacy } ##Legacy
			change_variable = { expFoo = 1 }
			multiply_variable = { which = monthly_magic_xp_gain which = expFoo }
		}
		change_variable = { which = monthly_magic_xp_gain which = temp_exp }
		change_variable = {
			which = country_magic_experience
			which = monthly_magic_xp_gain
		}
		update_advancement_progress_bar = { type = country }
	}
	else = {
		change_variable = { which = passive_experience_bank which = monthly_magic_xp_gain }
	}
	
	if = { limit = { magic_advancement_done = yes }
		country_event = { id = magic_study.100 }
	}
}
