volcanic_truth_sacred_site_modifier_checks = {
	if = {
		limit = {
			has_global_flag = vt_incident_organizational_crisis_lesser_sacred_sites
		}
		if = {
			# First, check if the province and owner satisfy the conditions
			limit = {
				religion = volcanic_truth
				owner = { religion = volcanic_truth }
			}
			# Then check the province modifier
			if = {
				limit = {
					has_province_modifier = $site_name$_sacred_site_lesser_inactive
				}
				remove_province_modifier = $site_name$_sacred_site_lesser_inactive
				add_permanent_province_modifier = {
					name = $site_name$_sacred_site_lesser
					duration = -1
				}
			}
		}
		else_if = { # If you don't satisfy the conditions, and the active version is there, swap it
			limit = {
				has_province_modifier = $site_name$_sacred_site_lesser
			}
			remove_province_modifier = $site_name$_sacred_site_lesser
			add_permanent_province_modifier = {
				name = $site_name$_sacred_site_lesser_inactive
				duration = -1
			}
		}
	}
	if = {
		limit = {
			has_global_flag = vt_incident_organizational_crisis_greater_sacred_sites
		}
		if = {
			# First, check if the province and owner satisfy the conditions. The Greater version requires the gov reform
			limit = {
				religion = volcanic_truth
				owner = {
					religion = volcanic_truth
					has_reform = cuahu_teotlehu_reform
				}
			}
			# Then check the province modifier
			if = {
				limit = {
					has_province_modifier = $site_name$_sacred_site_greater_inactive
				}
				remove_province_modifier = $site_name$_sacred_site_greater_inactive
				add_permanent_province_modifier = {
					name = $site_name$_sacred_site_greater
					duration = -1
				}
			}
		}
		else_if = { # If you don't satisfy the conditions, and the active version is there, swap it
			limit = {
				has_province_modifier = $site_name$_sacred_site_greater
			}
			remove_province_modifier = $site_name$_sacred_site_greater
			add_permanent_province_modifier = {
				name = $site_name$_sacred_site_greater_inactive
				duration = -1
			}
		}
	}
}


volcanic_truth_kiamoa_entrance_modifier_checks = {
	if = {
		# If the province and new owner are the right religion...
		limit = {
			religion = volcanic_truth
			owner = { religion = volcanic_truth }
		}
		if = { #...and it had the inactive version of a modifier, replace it with the active version
			limit = {
				has_province_modifier = volcanic_truth_kiamoa_entrance_guard_inactive
			}
			remove_province_modifier = volcanic_truth_kiamoa_entrance_guard_inactive
			add_permanent_province_modifier = {
				name = volcanic_truth_kiamoa_entrance_guard
				duration = -1
			}
		}
		# If the modifiers were present, or if the active version was already present, it is
		# correct to leave the situation as is
	}
	else_if = { #If either the new owner or the province is no longer VT, replace the active version with inactive
		limit = {
			has_province_modifier = volcanic_truth_kiamoa_entrance_guard
		}
		remove_province_modifier = volcanic_truth_kiamoa_entrance_guard
		add_permanent_province_modifier = {
			name = volcanic_truth_kiamoa_entrance_guard_inactive
			duration = -1
		}
		# If the modifiers were present, or if the inactive version was already present, it is
		# correct to leave the situation as is
	}
}

insyaa_charkuchin_kobolds_released_setup = {
	clr_country_flag = charkuchin_kobold_start #I'm pretty sure flags stick around if you are annexed then released

	# Since your government type depends on the type of the country that released you as a vassal or forced you to be
	# released via peace, and we ony have 7 starting Insyaa kobold tags, I'll hardcode their starting gov types
	# If you're not in that list (maybe more kobolds in the future) you get a monarchy if you were native
	# or tribal
	trigger_switch = {
		on_trigger = tag
		P30 = { change_government = monarchy }
		P31 = { change_government = theocracy }
		P32 = { change_government = monarchy }
		P33 = { change_government = monarchy }
		P34 = { change_government = monarchy }
		P35 = { change_government = republic }
		P36 = { change_government = republic }
	}
	trigger_switch = {
		on_trigger = government
		monarchy = { add_government_reform = charkuchin_monarchy_reform }
		republic = { add_government_reform = charkuchin_council_reform }
		theocracy = { add_government_reform = charkuchin_citadel_clerical_administration }
		tribal = {
			change_government = monarchy
			add_government_reform = charkuchin_monarchy_reform
		}
		native = {
			change_government = monarchy
			add_government_reform = charkuchin_monarchy_reform
		}
	}
	country_event = { #Initial variable setup and some modifiers
		id = charkuchin_kobold.0
	}
	#Update the supplies counters for your provinces, since the starting event assumes
	#you only have your capital with a citadal
	every_owned_province = {
		limit = {
			is_capital = no
		}
		insyaa_charkuchin_kobold_citadel_supplies_update_count_add = yes
	}
	#Update your gain from admin tech
	export_to_variable = {
		which = kobold_admin_tech
		value = adm_tech
	}
	#Since the starting event assumes tech 2
	subtract_variable = { kobold_admin_tech = 2 }
	while = {
		limit = {
			check_variable = {
				which = kobold_admin_tech
				value = 1
			}
		}
		subtract_variable = { kobold_admin_tech = 1 }
		insyaa_charkuchin_kobold_citadel_supplies_tech = yes
	}
}

gene_pillar_generic_then_specific_event = {
	hidden_effect = {
		# Rather than having a huge number of "if" branches, I've arbitrarily decided on a priority to
		# determine the timing of the next event, based on a default of 1 specific and 1 generic per year:
		# 1. Double speed flag
		# 2. Temporary delay flags
		# 3. Fast exploration privilege
		# We need to be careful though, since we can't fire an event from an else or else_if
		if = { # Double speed
			limit = {
				has_country_flag = gene_pillars_exploration_events_double_speed
			}
			country_event = { id = gene_pillars.1000 days = 91 } #Generic event handler
			country_event = { id = gene_pillars.$specific_event$ days = 182 } #Next event in chain
		}
		if = {
			limit = {
				NOT = { has_country_flag = gene_pillars_exploration_events_double_speed }
			}
			if = { # 90-180 day delay
				limit = {
					has_country_flag = gene_pillars_exploration_next_event_small_delay
				}
				country_event = { id = gene_pillars.1000 days = 272 random = 90 } #Generic event handler
				country_event = { id = gene_pillars.$specific_event$ days = 455 random = 90 } #Next event in chain
			}
			if = {
				limit = {
					NOT = { has_country_flag = gene_pillars_exploration_next_event_small_delay }
				}
				if = { # 180-270 day delay
					limit = {
						has_country_flag = gene_pillars_exploration_next_event_medium_delay
					}
					country_event = { id = gene_pillars.1000 days = 362 random = 90 } #Generic event handler
					country_event = { id = gene_pillars.$specific_event$ days = 545 random = 90 } #Next event in chain
				}
				if = {
					limit = {
						NOT = { has_country_flag = gene_pillars_exploration_next_event_medium_delay }
					}
					if = { # 270-360 day delay
						limit = {
							has_country_flag = gene_pillars_exploration_next_event_large_delay
						}
						country_event = { id = gene_pillars.1000 days = 452 random = 90 } #Generic event handler
						country_event = { id = gene_pillars.$specific_event$ days = 635 random = 90 } #Next event in chain
					}
					if = {
						limit = {
							NOT = { has_country_flag = gene_pillars_exploration_next_event_large_delay }
						}
						if = { # Increased speed privilege
							limit = { always = no } #Replace with has_estate_privilege = gene_pillar_fast exploration
							country_event = { id = gene_pillars.1000 days = 135 } #Generic event handler
							country_event = { id = gene_pillars.$specific_event$ days = 270 } #Next event in chain
						}
						if = {
							limit = { always = yes } #Replace with NOT = { has_estate_privilege = gene_pillar_fast exploration }
							country_event = { id = gene_pillars.1000 days = 182 } #Generic event handler
							country_event = { id = gene_pillars.$specific_event$ days = 365 } #Next event in chain
						}
					}
					clr_country_flag = gene_pillars_exploration_next_event_large_delay
				}
				clr_country_flag = gene_pillars_exploration_next_event_medium_delay
			}
			clr_country_flag = gene_pillars_exploration_next_event_small_delay
		}
		gene_pillars_advance_progress_bar = yes
	}
}

	tooltip = {
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = $amount$
		}
	}
	hidden_effect = {
		event_target:gene_pillar_exploration_province = {
			multiply_variable = {
				which = aPUI_gene_pillar_health_var
				value = 0.1
			}
			change_variable = {
				which = aPUI_gene_pillar_health_var
				value = $amount$
			}
			if = {
				limit = {
					owner = {
						gene_pillars_has_any_frontliner_class_country = yes
					}
				}
				change_variable = {
					which = aPUI_gene_pillar_health_var
					value = $amount$
				}
			}
			multiply_variable = {
				which = aPUI_gene_pillar_health_var
				value = 10
			}
			if = {
				limit = {
					check_variable = {
						which = aPUI_gene_pillar_health_var
						which = aPUI_gene_pillar_health_capacity
					}
				}
				set_variable = {
					which = aPUI_gene_pillar_health_var
					which = aPUI_gene_pillar_health_capacity
				}
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							which = aPUI_gene_pillar_health_var
							value = 0
						}
					}
				}
				owner = {
					reduce_stability_or_adm_power = yes
					set_country_flag = gene_pillars_got_killed
					gene_pillar_abandon_expedition_effect = yes
				}
			}
		}
	}
}

		$success_factor$ = {
			gene_pillars_add_loot_effect = { amount = $party_loot$ }
			gene_pillars_add_xp_effect = { amount = $party_xp$ }
			change_variable = { # So I can track what to remove if the trap is sprung
				which = gene_pillars_generic_traps_and_treasure_loot_gained
				value = $party_loot$
			}
			change_variable = { # So I can track what to remove if the trap is sprung
				which = gene_pillars_generic_traps_and_treasure_xp_gained
				value = $party_xp$
			}
			change_variable = {
				which = gene_pillars_generic_traps_and_treasure_count
				value = 1
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							which = gene_pillars_generic_traps_and_treasure_count
							value = 5
						}
					}
				}
				country_event = {
					id = gene_pillars.1005 # get this event again
					days = 1 # can't trigger itself without a delay
				}
			}
			
		}
		$fail_factor$ = {
			country_event = {
				id = gene_pillars.1006 # Trap sprung
			}
		}
	}
}
