cot_increase_long_lived_stat_capped_adm = {
    if = { #WOS
		limit = {
			OR = {
				is_long_lived_ruler = yes
				a_vampire_rules_the_country = yes
			}
			OR = {
				adm = 6
				AND {
					adm = 5
					NOT = { has_global_modifier_value = { which = monarch_admin_power value = 4 } }
				}
				AND {
					adm = 4
					NOT = { has_global_modifier_value = { which = monarch_admin_power value = 3 } }
				}
				AND {
					adm = 3
					NOT = { has_global_modifier_value = { which = monarch_admin_power value = 2 } }
				}
				AND {
					adm = 2
					NOT = { has_global_modifier_value = { which = monarch_admin_power value = 1 } }
				}
			}
		}
		add_adm_power = 50
	}
	else = {
		increase_ruler_adm_effect = yes
	}
}

# Custom Carneter effects. Province effects are called from country-rooted chains.

carneter_apply_grove = {
    remove_province_modifier = carneter_grove_base
    remove_province_modifier = carneter_grove_communion
    if = {
        limit = { owner = { has_country_flag = carneter_spirits_done } }
        add_province_modifier = { name = carneter_grove_communion duration = -1 }
    }
    else = { add_province_modifier = { name = carneter_grove_base duration = -1 } }
}

carneter_apply_sanctuary = {
    # The old modifier is defined in the earlier Ilvandet wildlife package.
    remove_province_modifier = ilvandet_wildlife_sanctuary
    remove_province_modifier = carneter_sanctuary_base
    remove_province_modifier = carneter_sanctuary_breeding
    remove_province_modifier = carneter_sanctuary_living
    if = {
        limit = { owner = { has_country_flag = carneter_living_forest_done } }
        add_province_modifier = { name = carneter_sanctuary_living duration = -1 }
    }
    else_if = {
        limit = { owner = { has_country_flag = carneter_breeding_done } }
        add_province_modifier = { name = carneter_sanctuary_breeding duration = -1 }
    }
    else = { add_province_modifier = { name = carneter_sanctuary_base duration = -1 } }
}

carneter_refresh_sites = {
    every_owned_province = {
        limit = {
            OR = {
                has_province_flag = ilvandet_wildlife_sanctuary
                has_province_modifier = ilvandet_wildlife_sanctuary
                has_province_flag = carneter_royal_sanctuary
            }
        }
        set_province_flag = carneter_royal_sanctuary
        carneter_apply_sanctuary = yes
    }
    every_owned_province = {
        limit = { has_province_flag = carneter_royal_grove }
        carneter_apply_grove = yes
    }
}

carneter_update_census = {
    set_variable = { which = carneter_total_production_development value = 0 }
    set_variable = { which = carneter_woodland_production_development value = 0 }
    set_variable = { which = carneter_grove_count value = 0 }
    set_variable = { which = carneter_sanctuary_count value = 0 }
    every_owned_province = {
        limit = { is_city = yes }
        export_to_variable = { which = carneter_total_production_development value = trigger_value:base_production }
        ROOT = { change_variable = { which = carneter_total_production_development which = PREV } }
        if = {
            limit = { OR = { carneter_woodland = yes has_province_flag = carneter_royal_sanctuary } }
            export_to_variable = { which = carneter_woodland_production_development value = trigger_value:base_production }
            ROOT = { change_variable = { which = carneter_woodland_production_development which = PREV } }
        }
        if = {
            limit = { carneter_owned_grove = yes }
            ROOT = { change_variable = { which = carneter_grove_count value = 1 } }
        }
        if = {
            limit = { carneter_owned_sanctuary = yes }
            ROOT = { change_variable = { which = carneter_sanctuary_count value = 1 } }
        }
    }
    clr_country_flag = carneter_last_census
    set_country_flag = carneter_last_census
}

carneter_initialize = {
    if = {
        limit = { tag = B78 NOT = { has_country_flag = carneter_forest_initialized } }
        set_country_flag = carneter_forest_initialized
        carneter_refresh_sites = yes
        carneter_update_census = yes
        swap_non_generic_missions = yes
        country_event = { id = carneter_forest.4 days = 1 }
    }
}

carneter_select_grove_site = {
    every_owned_province = { clr_province_flag = carneter_grove_preview_seen }
    if = {
        limit = { capital_scope = { carneter_grove_candidate = yes } }
        capital_scope = { save_event_target_as = carneter_grove_site }
    }
    else = { random_owned_province = { limit = { carneter_grove_candidate = yes } save_event_target_as = carneter_grove_site } }
    event_target:carneter_grove_site = { set_province_flag = carneter_grove_preview_seen }
}
