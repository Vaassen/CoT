# Injected FIRST into BOTH country battle callbacks, before other handlers
# can change variables or contexts. Never install the same callback twice.
ar_hs_on_battle = {
    if = {
        limit = { ar_hs_qualifying_battle = yes }
        set_variable = { which = ar_hs_battle_souls value = 0 }
        change_variable = { which = ar_hs_battle_souls which = land_units_killed }
        change_variable = { which = ar_hs_total_souls which = ar_hs_battle_souls }
        change_variable = { which = ar_hs_unspent_souls which = ar_hs_battle_souls }
        while = {
            limit = {
                check_variable = { which = ar_hs_unspent_souls which = ar_hs_souls_per_tier }
                NOT = { check_variable = { which = ar_hs_tier value = 5 } }
            }
            subtract_variable = { which = ar_hs_unspent_souls which = ar_hs_souls_per_tier }
            change_variable = { which = ar_hs_tier value = 1 }
        }
        ar_hs_apply_tier = yes
    }
}

        limit = { check_variable = { which = ar_hs_tier value = 2 } }
        add_country_modifier = { name = ar_hs_tier_2 duration = -1 }
    }

    if = {
        limit = {
            NOT = { has_country_modifier = ar_hs_active }
            OR = {
                has_country_flag = ar_hs_live
                has_country_modifier = ar_hs_tier_1
                has_country_modifier = ar_hs_tier_2
                has_country_modifier = ar_hs_tier_3

        ar_hs_end = yes
    }
    ar_hs_expire_if_due = yes
}
