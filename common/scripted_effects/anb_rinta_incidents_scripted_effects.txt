#Adding Reso
add_infernal_resonance = {
	add_permanent_province_modifier = {
		name = rinta_infernal_resonance
		duration = -1
	}
	hidden_effect = {
		REB = {
			change_variable = {
				which = rinta_factor
				value = 1
			}
		}	
	}
}
#Removing Reso

	remove_province_modifier = rinta_infernal_resonance

			}
		}	
	}
}

rinta_get_telescope_multiplier = {
	export_to_variable = {
		which = $variable$
		value = modifier:telescope_investment_efficency
		who = event_target:night_sky_country_target #Using ROOT instead doesn't work if anyone's curious
	}
	change_variable = {
		which = $variable$
		value = 1
	}
	[[value] #For flat amounts
		multiply_variable = {
			which = $variable$
			value = $cost$
		}
	]
	[[which] #For dynamic amounts (everything else)
		multiply_variable = {
			which = $variable$
			which = $cost$
		}
	]
}

rinta_steal_funds_event_effect = {
	if = {
		limit = { has_global_flag = rinta_supercharge_stealing_funds_flag }
		add_spy_network_in = {
			who = event_target:$event_target$
			value = -20
		}
	}
	else = {
		add_spy_network_in = {
			who = event_target:$event_target$
			value = -35
		}
	}
	rinta_invest_in_telescope = { level = 0_5 }
	event_target:$event_target$ = {
		tooltip = {
			add_opinion = {
				who = ROOT
				modifier = rinta_stole_astronomy_funds
			}
			add_casus_belli = {
				type = cb_spy_discovered
				target = ROOT
				months = 60 #5 years
			}
		}
		country_event = {
			id = rinta_seeker_events.520 #Astronomical Center Funds Stolen
		}
	}
	clr_country_flag = stole_telescope_funds_flag
	set_country_flag = stole_telescope_funds_flag
}
