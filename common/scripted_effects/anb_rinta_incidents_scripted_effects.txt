#Adding Reso
add_infernal_resonance = {
	add_permanent_province_modifier = {
		name = rinta_infernal_resonance
		duration = -1
	}
	hidden_effect = {
		REB = {
			change_variable = {
				which = rinta_factor
				value = 1
			}
		}	
	}
}
#Removing Reso
remove_infernal_resonance = {
	remove_province_modifier = rinta_infernal_resonance
	hidden_effect = {
		REB = {
			change_variable = {
				which = rinta_factor
				value = -1
			}
		}	
	}
}

rinta_get_telescope_multiplier = {
	export_to_variable = {
		which = $variable$
		value = modifier:telescope_investment_efficency
		who = event_target:night_sky_country_target #Using ROOT instead doesn't work if anyone's curious
	}
	change_variable = {
		which = $variable$
		value = 1
	}
	[[value] #For flat amounts
		multiply_variable = {
			which = $variable$
			value = $cost$
		}
	]
	[[which] #For dynamic amounts (everything else)
		multiply_variable = {
			which = $variable$
			which = $cost$
		}
	]
}
