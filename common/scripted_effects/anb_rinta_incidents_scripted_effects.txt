#Rinta Dark Magic check for end of Invasion
alefisol_invasion_siege_won_effect = {
	if = {
		limit = { #Alefisol losing
			OR = {
				NOT = { war_score = -98 }
				AND = {
					check_variable = {
						which = alefisol_lost_6
						value = 6
					}
					OR = {
						capital_scope = {
							NOT = { controlled_by = S93 }
						}
						NOT = { war_score = -74 }
					}
				}
			}
		}
		every_known_country = {
			limit = { has_disaster = darkmagicks }
			country_event = { id = rinta_seeker_events.211 days = 1 }
		}
		if = {
			limit = { S93 = { exists = yes } }
			S93 = {
				remove_country_modifier = alefisol_modifier
				every_owned_province = {
					remove_core = ROOT
					cede_province = previous_owner
				}
				if = {
					limit = { S93 = { exists = yes } }
					remove_country = yes
				}
			}
		}
		clr_global_flag = darkmagicks_flag
		clr_global_flag = incident_rinta_dark_magic_ongoing_flag
	}
	if = {
		limit = { #Alefisol win
			war_score = 95
		}
		country_event = { id = rinta_seeker_events.212 days = 1 }
	}
}
