#Rinta Dark Magic check for end of Invasion
alefisol_invasion_siege_won_effect = {
	if = {
		limit = { #Alefisol losing
			OR = {
				NOT = { war_score = -98 }
				AND = {
					check_variable = {
						which = alefisol_lost_6
						value = 6
					}
					OR = {
						capital_scope = {
							NOT = { controlled_by = S93 }
						}
						NOT = { war_score = -74 }
					}
				}
			}
		}
		every_known_country = {
			limit = { has_disaster = darkmagicks }
			country_event = { id = rinta_seeker_events.211 days = 1 }
		}
		if = {
			limit = { S93 = { exists = yes } }
			S93 = {
				remove_country_modifier = alefisol_modifier
				every_owned_province = {
					remove_core = ROOT
					cede_province = previous_owner
				}
				if = {
					limit = { S93 = { exists = yes } }
					remove_country = yes
				}
			}
		}
		clr_global_flag = darkmagicks_flag
	}
	if = {
		limit = { #Alefisol win
			war_score = 95
		}
		country_event = { id = rinta_seeker_events.212 days = 1 }
	}
}

rinta_get_telescope_multiplier = {
	export_to_variable = {
		which = $variable$
		value = modifier:telescope_investment_efficency
		who = event_target:night_sky_country_target #Using ROOT instead doesn't work if anyone's curious
	}
	change_variable = {
		which = $variable$
		value = 1
	}
	[[value] #For flat amounts
		multiply_variable = {
			which = $variable$
			value = $cost$
		}
	]
	[[which] #For dynamic amounts (everything else)
		multiply_variable = {
			which = $variable$
			which = $cost$
		}
	]
}

rinta_reduce_investment_in_telescope = {

			subtract_variable = { #Update your total amount raised

				value = $amount$
			}
			rinta_pay_telescope_cost = { #Remove the amount invested
				pay = $refund$
				effect = "add_treasury = $amount$"
			}
		}
	]
	[[all] #Invest all of your country's money
		custom_tooltip = rinta_invest_in_telescope_all_tt

			export_to_variable = {
				which = sky_incident_treasury_country_temp

			}

				which = rinta_telescope_amount_invested
				which = sky_incident_treasury_country_temp
			}
			rinta_pay_telescope_cost = { #Double your cash
				pay = $refund$
				effect = "
					var_effect = {
						var = sky_incident_treasury_country_temp
						effect = add_treasury
						sign = +
					}
				"
			}
		}
	]
	[[half] #Invest 1/2 your country's money
		custom_tooltip = rinta_invest_in_telescope_half_tt
		hidden_effect = {
