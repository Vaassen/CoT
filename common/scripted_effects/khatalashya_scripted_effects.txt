add_faith_in_the_prophecy = {
    custom_tooltip = add_faith_in_the_prophecy_tt
    if = {
        limit = { has_country_modifier = 333_total_faith_in_the_prophecy }
        add_adm_power = 100
        add_dip_power = 100
        add_mil_power = 100
    }
    else_if = {
        limit = { has_country_modifier = 333_strong_faith_in_the_prophecy }
        remove_country_modifier = 333_strong_faith_in_the_prophecy
        add_country_modifier = {
            name = 333_total_faith_in_the_prophecy

        }
    }
    else_if = {
        limit = { has_country_modifier = 333_questioning_faith_in_the_prophecy }
        remove_country_modifier = 333_questioning_faith_in_the_prophecy
        add_country_modifier = {
            name = 333_strong_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_wavering_faith_in_the_prophecy }
        remove_country_modifier = 333_wavering_faith_in_the_prophecy
        add_country_modifier = {
            name = 333_questioning_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_failing_faith_in_the_prophecy }
        remove_country_modifier = 333_failing_faith_in_the_prophecy
        add_country_modifier = {
            name = 333_wavering_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {

khatalashya_add_debuff_hive_main = {
    tooltip = {
        SS6 = {
            add_country_modifier = {
                name = khatalashya_debuff_$mutation$
                duration = -1
            }
        }
    }
    custom_tooltip = add_debuff_space_colony_tt
    hidden_effect = {
        every_war_enemy_country = {
            limit = {
                has_country_modifier = khatalashya_exterminate_voices
            }
            add_country_modifier = {
                name = khatalashya_debuff_$mutation$
                duration = -1
            }
            add_debuff_space_colony = yes
        }
    }
}
khatalashya_remove_debuff_hive_main = {
    tooltip = {
        SS6 = {
            remove_country_modifier = khatalashya_debuff_$mutation$
        }
    }
    custom_tooltip = remove_debuff_space_colony_tt
    hidden_effect = {
        every_war_enemy_country = {
            limit = {
                has_country_modifier = khatalashya_exterminate_voices
            }
            remove_country_modifier = khatalashya_debuff_$mutation$
            remove_debuff_space_colony = yes
        }
    }
}
