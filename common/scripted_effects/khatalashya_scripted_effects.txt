#333 Empire Khatalashya

#Faith in the Prophecy

add_faith_in_the_prophecy = {
    if = {
        limit = { has_country_modifier = 333_total_faith_in_the_prophecy }
        add_adm_power = 100
        add_dip_power = 100
        add_mil_power = 100
    }
    else_if = {
        limit = { has_country_modifier = 333_strong_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_total_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_questioning_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_strong_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_wavering_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_questioning_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_failing_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_wavering_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_colapsed_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_failing_faith_in_the_prophecy
            duration = -1
        }
    }
}
remove_faith_in_the_prophecy = {
    if = {
        limit = { has_country_modifier = 333_total_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_strong_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_strong_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_questioning_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_questioning_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_wavering_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_wavering_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_failing_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_failing_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_colapsed_faith_in_the_prophecy
            duration = -1
        }
    }
    else_if = {
        limit = { has_country_modifier = 333_colapsed_faith_in_the_prophecy }
        add_country_modifier = {
            name = 333_failing_faith_in_the_prophecy
            duration = -1
        }
        every_owned_province = {
            add_devastation = 10
        }
    }
}
