	if = {
		limit = { has_ruler_modifier = lichdom_mod_2 }
		country_event = { id = magic_project.702 }
	}
	if = {
		limit = { has_ruler_modifier = lichdom_mod_1 }
		country_event = { id = magic_project.701 }
	}
	if = {
		limit = {
			NOT = {
				has_ruler_modifier = lichdom_mod_1
				has_ruler_modifier = lichdom_mod_2
			}
		}
		country_event = { id = magic_project.700 }
	}
}
