#Runs every time the aPUI is opened, and on conquest.
aPUI_gene_pillar_interaction_update_effect = {
	set_aPUI_cap = { name = gene_pillar_health amount = 10 }
	set_variable = { which = aPUI_gene_pillar_health_var value = 3 }
	set_aPUI_cap = { name = gene_pillar_health_max amount = 10 }
	set_variable = { which = aPUI_gene_pillar_health_max_var value = 3 }
	gene_pillars_recount_health_for_fighters = { slot = 1 }
	gene_pillars_recount_health_for_fighters = { slot = 2 }
	gene_pillars_recount_health_for_fighters = { slot = 3 }
	gene_pillars_recount_health_for_fighters = { slot = 4 }
	gene_pillars_recount_health_for_fighters = { slot = 5 }
	owner = {
		if = {
			limit = { NOT = { has_country_modifier = mechanim_administration } }
			set_country_flag = gene_pillars_unlocked_rogue_A
			set_country_flag = gene_pillars_unlocked_ranger_A
			set_country_flag = gene_pillars_unlocked_fighter_A
			set_country_flag = gene_pillars_unlocked_cleric_A
			if = {
				limit = {
					has_not_banned_magic_artificery = yes
				}
				if = {
					limit = { insyaa_has_flamesong = yes }
					set_country_flag = gene_pillars_unlocked_mage_B
				}
				else_if = {
					limit = { insyaa_has_wavesong = yes }
					set_country_flag = gene_pillars_unlocked_mage_C
				}
				else_if = {
					limit = { insyaa_has_stonescrawl = yes }
					set_country_flag = gene_pillars_unlocked_mage_D
				}
				else_if = {
					limit = { insyaa_has_stonestory = yes }
					set_country_flag = gene_pillars_unlocked_mage_E
				}
				else = {
					set_country_flag = gene_pillars_unlocked_mage_A
				}
			}
			if = {
				limit = { has_estate = estate_artificers }
				set_country_flag = gene_pillars_unlocked_artificer_A
			}
		}
		else = {
			set_country_flag = gene_pillars_unlocked_rogue_D
			set_country_flag = gene_pillars_unlocked_rogue_E
			set_country_flag = gene_pillars_unlocked_ranger_E
			set_country_flag = gene_pillars_unlocked_fighter_E
			set_country_flag = gene_pillars_unlocked_cleric_E
			set_country_flag = gene_pillars_unlocked_artificer_E
		}
	}
	if = {
		limit = { NOT = { has_province_flag = aPUI_gene_pillar_interaction_setup_done_province } }
		set_variable = { which = aPUI_gene_pillar_party_slot_framevar_1 value = 1 }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 2 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_2 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_2 value = 2 } }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 3 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_3 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_3 value = 1 } }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 4 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_4 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_4 value = 2 } }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 5 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_5 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_5 value = 2 } }
		owner = {
			PREV = {
				set_variable = {
					which = aPUI_gene_pillar_xp_capacity
					which = PREV
				}
				set_variable = {
					which = aPUI_gene_pillar_xp_var
					which = PREV
				}
				update_aPUI_var_bar = { name = gene_pillar_xp }
				set_aPUI_cap = { name = gene_pillar_loot amount = 100 }
				set_variable = {
					which = aPUI_gene_pillar_loot_var
					which = PREV
				}
				set_aPUI_cap = { name = gene_pillar_progress amount = 0 }
				set_variable = { which = aPUI_gene_pillar_progress_var value = 0 }
				update_aPUI_var_bar = { name = aPUI_gene_pillar_progress_var }
			}
		}
	}
}

#Fires a generic event, and then a scripted event
gene_pillar_generic_then_specific_event = {
	hidden_effect = {
		# Rather than having a huge number of "if" branches, I've arbitrarily decided on a priority to
		# determine the timing of the next event, based on a default of 1 specific and 1 generic per year:
		# 1. Double speed flag
		# 2. Temporary delay flags
		# 3. Fast exploration privilege
		# We need to be careful though, since we can't fire an event from an else or else_if
		if = { # Double speed
			limit = {
				has_country_flag = gene_pillars_exploration_events_double_speed
			}
			if = { # Increased speed privilege
				limit = {
					OR = {
						has_estate_privilege = artifice_invention_pillar_scouts
						has_estate_privilege = artifice_invention_mech_pillar_scouts
					}

				which = aPUI_gene_pillar_health_var
				value = $amount$
			}
			if = {
				limit = {
					owner = {
						gene_pillars_has_any_frontliner_class_country = yes
					}
				}
				change_variable = {
					which = aPUI_gene_pillar_health_var
					value = $amount$
				}
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							which = aPUI_gene_pillar_health_var
							value = 0
						}
					}
				}
				owner = {
					reduce_stability_or_adm_power = yes
					set_country_flag = gene_pillars_got_killed
					gene_pillar_abandon_expedition_effect = yes
				}
			}
		}
	}
}

gene_pillars_recount_health_for_fighters = {
	if = {
		limit = {
			is_variable_equal = {
				which = aPUI_gene_pillar_party_slot_framevar_$slot$
				value = 14
			}
		}
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = -1
		}
		change_variable = {
			which = aPUI_gene_pillar_health_max_var
			value = -1
		}
	}
	if = {
		limit = {
			is_variable_equal = {
				which = aPUI_gene_pillar_party_slot_framevar_$slot$
				value = 15
			}
		}
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = 1
		}
		change_variable = {
			which = aPUI_gene_pillar_health_max_var
			value = 1
		}
	}
}
