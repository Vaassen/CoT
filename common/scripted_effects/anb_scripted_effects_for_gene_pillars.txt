	change_variable = { which = temp which = aPUI_gene_pillar_country_health_var }
	if = {
		limit = { check_variable = { which = temp value = 10 } }
		set_variable = { which = temp value = 10 }
	}

	set_variable = { which = aPUI_gene_pillar_health_max_var which = temp }
	update_aPUI_var_bar = { name = aPUI_gene_pillar_health_var }
	update_aPUI_var_bar = { name = aPUI_gene_pillar_health_max_var }
}

aPUI_gene_pillar_interaction_update_effect = {
	set_aPUI_cap = { name = gene_pillar_health amount = 10 }
	set_variable = { which = aPUI_gene_pillar_health_var value = 3 }
	set_aPUI_cap = { name = gene_pillar_health_max amount = 10 }
	set_variable = { which = aPUI_gene_pillar_health_max_var value = 3 }
	gene_pillars_recount_health_for_fighters = { slot = 1 add = yes }
	gene_pillars_recount_health_for_fighters = { slot = 2 add = yes }
	gene_pillars_recount_health_for_fighters = { slot = 3 add = yes }
	gene_pillars_recount_health_for_fighters = { slot = 4 add = yes }
	gene_pillars_recount_health_for_fighters = { slot = 5 add = yes }
	owner = {
		if = {
			limit = { NOT = { has_country_modifier = mechanim_administration } }
			set_country_flag = gene_pillars_unlocked_rogue_A
			set_country_flag = gene_pillars_unlocked_ranger_A
			set_country_flag = gene_pillars_unlocked_fighter_A
			set_country_flag = gene_pillars_unlocked_cleric_A
			if = {
				limit = {
					has_not_banned_magic_artificery = yes
				}
				if = {
					limit = { insyaa_has_flamesong = yes }
					set_country_flag = gene_pillars_unlocked_mage_B
				}
				else_if = {
					limit = { insyaa_has_wavesong = yes }
					set_country_flag = gene_pillars_unlocked_mage_C
				}
				else_if = {
					limit = { insyaa_has_stonescrawl = yes }
					set_country_flag = gene_pillars_unlocked_mage_D
				}
				else_if = {
					limit = { insyaa_has_stonestory = yes }
					set_country_flag = gene_pillars_unlocked_mage_E
				}
				else = {
					set_country_flag = gene_pillars_unlocked_mage_A
				}
			}
			if = {
				limit = { has_estate = estate_artificers }
				set_country_flag = gene_pillars_unlocked_artificer_A
			}
		}
		else = {
			set_country_flag = gene_pillars_unlocked_rogue_D
			set_country_flag = gene_pillars_unlocked_rogue_E
			set_country_flag = gene_pillars_unlocked_ranger_E
			set_country_flag = gene_pillars_unlocked_fighter_E
			set_country_flag = gene_pillars_unlocked_cleric_E
			set_country_flag = gene_pillars_unlocked_artificer_E
		}
	}
	if = {
		limit = { NOT = { has_province_flag = aPUI_gene_pillar_interaction_setup_done_province } }
		set_variable = { which = aPUI_gene_pillar_party_slot_framevar_1 value = 1 }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 2 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_2 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_2 value = 2 } }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 3 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_3 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_3 value = 1 } }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 4 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_4 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_4 value = 2 } }
		if = {
			limit = { owner = { check_variable = { which = aPUI_gene_pillars_party_level_var value = 5 } } }
			set_variable = { which = aPUI_gene_pillar_party_slot_framevar_5 value = 1 }
		}
		else = { set_variable = { which = aPUI_gene_pillar_party_slot_framevar_5 value = 2 } }
		owner = {
			PREV = {
				set_variable = {
					which = aPUI_gene_pillar_xp_capacity
					which = PREV
				}
				set_variable = {
					which = aPUI_gene_pillar_xp_var
					which = PREV
				}
				update_aPUI_var_bar = { name = gene_pillar_xp }
				set_aPUI_cap = { name = gene_pillar_loot amount = 100 }
				set_variable = {
					which = aPUI_gene_pillar_loot_var
					which = PREV
				}
				set_aPUI_cap = { name = gene_pillar_progress amount = 0 }
				set_variable = { which = aPUI_gene_pillar_progress_var value = 0 }
				update_aPUI_var_bar = { name = aPUI_gene_pillar_progress_var }
			}
		}
	}
}

	hidden_effect = {
		# This does not take into account every single combinatio of speed factors, but it handles
		# each flag in combination with the two increased speed estate privileges
		# We need to be careful though, since we can't fire an event from an else or else_if
		if = { # Double speed
			limit = {
				has_country_flag = gene_pillars_exploration_events_double_speed
			}
			if = { # Increased speed privilege
				limit = {
					OR = {
						has_estate_privilege = artifice_invention_pillar_scouts
						has_estate_privilege = artifice_invention_mech_pillar_scouts
					}
				}
				country_event = { id = gene_pillars.1000 days = 68 } #Generic event handler
				country_event = { id = gene_pillars.$specific_event$ days = 136 } #Next event in chain
			}
			if = {
				limit = {
					NOT = {
						has_estate_privilege = artifice_invention_pillar_scouts
						has_estate_privilege = artifice_invention_mech_pillar_scouts
					}
				}
				country_event = { id = gene_pillars.1000 days = 91 } #Generic event handler
				country_event = { id = gene_pillars.$specific_event$ days = 182 } #Next event in chain
			}
		}
		if = {
			limit = {
				NOT = { has_country_flag = gene_pillars_exploration_events_double_speed }
			}
			if = { # 90-180 day delay
				limit = {
					has_country_flag = gene_pillars_exploration_next_event_small_delay
				}
				if = { # Increased speed privilege
					limit = {
						OR = {
							has_estate_privilege = artifice_invention_pillar_scouts
							has_estate_privilege = artifice_invention_mech_pillar_scouts
						}
					}
					country_event = { id = gene_pillars.1000 days = 204 random = 67 } #Generic event handler
					country_event = { id = gene_pillars.$specific_event$ days = 341 random = 67 } #Next event in chain
				}
				if = {
					limit = {
						NOT = {
							has_estate_privilege = artifice_invention_pillar_scouts
							has_estate_privilege = artifice_invention_mech_pillar_scouts
						}
					}
					country_event = { id = gene_pillars.1000 days = 272 random = 90 } #Generic event handler
					country_event = { id = gene_pillars.$specific_event$ days = 455 random = 90 } #Next event in chain
				}
			}
			if = {
				limit = {
					NOT = { has_country_flag = gene_pillars_exploration_next_event_small_delay }
				}
				if = { # 180-270 day delay
					limit = {
						has_country_flag = gene_pillars_exploration_next_event_medium_delay
					}
					if = { # Increased speed privilege
						limit = {
							OR = {
								has_estate_privilege = artifice_invention_pillar_scouts
								has_estate_privilege = artifice_invention_mech_pillar_scouts
							}
						}
						country_event = { id = gene_pillars.1000 days = 271 random = 67 } #Generic event handler
						country_event = { id = gene_pillars.$specific_event$ days = 409 random = 67 } #Next event in chain
					}
					if = {
						limit = {
							NOT = {
								has_estate_privilege = artifice_invention_pillar_scouts
								has_estate_privilege = artifice_invention_mech_pillar_scouts
							}
						}
						country_event = { id = gene_pillars.1000 days = 362 random = 90 } #Generic event handler
						country_event = { id = gene_pillars.$specific_event$ days = 545 random = 90 } #Next event in chain
					}
				}
				if = {
					limit = {
						NOT = { has_country_flag = gene_pillars_exploration_next_event_medium_delay }
					}
					if = { # 270-360 day delay
						limit = {
							has_country_flag = gene_pillars_exploration_next_event_large_delay
						}
						if = { # Increased speed privilege
							limit = {
								OR = {
									has_estate_privilege = artifice_invention_pillar_scouts
									has_estate_privilege = artifice_invention_mech_pillar_scouts
								}
							}
							country_event = { id = gene_pillars.1000 days = 339 random = 67 } #Generic event handler
							country_event = { id = gene_pillars.$specific_event$ days = 476 random = 67 } #Next event in chain
						}
						if = {
							limit = {
								NOT = {
									has_estate_privilege = artifice_invention_pillar_scouts
									has_estate_privilege = artifice_invention_mech_pillar_scouts
								}
							}
							country_event = { id = gene_pillars.1000 days = 452 random = 90 } #Generic event handler
							country_event = { id = gene_pillars.$specific_event$ days = 635 random = 90 } #Next event in chain

		change_variable = {
			which = aPUI_gene_pillar_loot_var

		}
	}
	# Tooltips to remind the player of the bonus loot effects
	if = {
		limit = { has_country_flag = gene_pillars_has_fighter_C }
		custom_tooltip = gene_pillars_add_loot_effect_fighter_C_bonus_tt
	}
	if = {
		limit = { has_country_flag = gene_pillars_has_rogue_A }
		custom_tooltip = gene_pillars_add_loot_effect_rogue_A_bonus_tt
	}

		limit = { has_country_flag = gene_pillars_has_mage_A }
		custom_tooltip = gene_pillars_add_loot_effect_mage_A_bonus_tt
	}
	if = {
		limit = { has_country_flag = gene_pillars_has_mage_B }
		custom_tooltip = gene_pillars_add_loot_effect_mage_B_bonus_tt
	}
	if = {
		limit = { has_country_flag = gene_pillars_has_artificer_A }
		custom_tooltip = gene_pillars_add_loot_effect_artificer_A_bonus_tt
	}
	hidden_effect = {
		event_target:gene_pillar_exploration_province = {
			change_variable = {
				which = aPUI_gene_pillar_loot_var
				value = $amount$
			}
			multiply_variable = {
				which = aPUI_gene_pillar_loot_var
				value = 4
			}
			if = {
				limit = { owner = { has_country_flag = gene_pillars_has_fighter_C } }

					which = aPUI_gene_pillar_loot_var
					value = $amount$
				}
				change_variable = {
					which = aPUI_gene_pillar_loot_var
					value = $amount$
				}
			}
			if = {
				limit = { owner = { has_country_flag = gene_pillars_has_rogue_A } }
				change_variable = {
					which = aPUI_gene_pillar_loot_var
					value = $amount$
				}
			}

				limit = { owner = { has_country_flag = gene_pillars_has_mage_A } }

					which = aPUI_gene_pillar_loot_var

gene_pillars_level_up_check = {
	hidden_effect = {
		if = {
			limit = {
				check_variable = {
					which = aPUI_gene_pillar_xp_var
					value = 100
				}
			}
			change_variable = {
				which = aPUI_gene_pillar_xp_var
				value = -100
			}
	
			change_variable = {
				which = aPUI_gene_pillars_party_level_var
				value = 1
			}
			if = { # Went from level 1 to level 2
				limit = {
					is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 2 }
				}
				set_variable = { which = aPUI_gene_pillar_party_slot_framevar_2 value = 1 } # Change visual as you unlock slot 2

					limit = { NOT = { owner = { has_country_modifier = mechanim_administration } } }
					owner = {
						set_country_flag = gene_pillars_unlocked_fighter_B
						set_country_flag = gene_pillars_unlocked_fighter_C
					}
				}
			}
			else_if = { # Went from level 2 to level 3
				limit = {
					is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 3 }
				}
				set_variable = { which = aPUI_gene_pillar_party_slot_framevar_3 value = 1 } # Change visual as you unlock slot 2
			}
			else_if = { # Went from level 3 to level 4
				limit = {
					is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 4 }
				}
				set_variable = { which = aPUI_gene_pillar_party_slot_framevar_4 value = 1 } # Change visual as you unlock slot 2
				if = {
					limit = { NOT = { owner = { has_country_modifier = mechanim_administration } } }
					owner = { set_country_flag = gene_pillars_unlocked_cleric_B }
				}
			}
			else_if = { # Went from level 4 to level 5
				limit = {
					is_variable_equal = { which = aPUI_gene_pillars_party_level_var value = 5 }
				}
				set_variable = { which = aPUI_gene_pillar_party_slot_framevar_5 value = 1 } # Change visual as you unlock slot 2
			}
		}
	}
}

	set_variable = {
		which = aPUI_gene_pillar_health_framevar
		value = $amount$
	}
	if = {
		limit = {		
			check_variable = {
				which = aPUI_gene_pillar_health_framevar

			}

	else = { gene_pillars_lose_health_effect = { amount = $amount$ } }
	update_aPUI_var_bar = { name = gene_pillar_health }
	update_aPUI_var_bar = { name = gene_pillar_health_max }
}

# If we are removing an existing fighter from the slot, change health in a certain
# direction. If we are adding a new fighter, change health in the other direction
gene_pillars_recount_health_for_fighters = {
	[[remove]
	if = {
		limit = {
			is_variable_equal = {
				which = aPUI_gene_pillar_party_slot_framevar_$slot$
				value = @fighter_B # Paladin, who increases party health
			}
		}
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = -1
		}
		change_variable = {
			which = aPUI_gene_pillar_health_max_var
			value = -1
		}
	}
	if = {
		limit = {
			is_variable_equal = {
				which = aPUI_gene_pillar_party_slot_framevar_$slot$
				value = @fighter_C # Barbarian, who decreases party health
			}
		}
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = 1
		}
		change_variable = {
			which = aPUI_gene_pillar_health_max_var
			value = 1
		}
	}
	]
	[[add]
	if = {
		limit = {
			is_variable_equal = {
				which = aPUI_gene_pillar_party_slot_framevar_$slot$
				value = @fighter_B # Paladin, who increases party health
			}
		}
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = 1
		}
		change_variable = {
			which = aPUI_gene_pillar_health_max_var
			value = 1
		}
	}
	if = {
		limit = {
			is_variable_equal = {
				which = aPUI_gene_pillar_party_slot_framevar_$slot$
				value = @fighter_C # Barbarian, who decreases party health
			}
		}
		change_variable = {
			which = aPUI_gene_pillar_health_var
			value = -1
		}
		change_variable = {
			which = aPUI_gene_pillar_health_max_var
			value = -1
		}
	}
	]
}

gene_pillars_generic_loot_or_trap_repeat = {
	random_list = {
		$success_factor$ = {
			gene_pillars_add_loot_effect = { amount = $party_loot$ }
			gene_pillars_add_xp_effect = { amount = $party_xp$ }
			hidden_effect = {
				change_variable = { # So I can track what to remove if the trap is sprung
					which = gene_pillars_generic_traps_and_treasure_loot_gained
					value = $party_loot$
				}
				change_variable = { # So I can track what to remove if the trap is sprung
					which = gene_pillars_generic_traps_and_treasure_xp_gained
					value = $party_xp$
				}
				change_variable = {
					which = gene_pillars_generic_traps_and_treasure_count
					value = 1
				}
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							which = gene_pillars_generic_traps_and_treasure_count
							value = 5
						}
					}
				}
				country_event = {
					id = gene_pillars.1005 # get this event again
					days = 1 # can't trigger itself without a delay
				}
			}
			# If you succeed 5 times, reset the variables in case you get this again
			else = {
				hidden_effect = {
					set_variable = {
						which = gene_pillars_generic_traps_and_treasure_loot_gained
						value = 0
					}
					set_variable = {
						which = gene_pillars_generic_traps_and_treasure_xp_gained
						value = 0
					}
					set_variable = {
						which = gene_pillars_generic_traps_and_treasure_count
						value = 0
					}
				}
			}
		}
		$fail_factor$ = {
			country_event = {
				id = gene_pillars.1006 # Trap sprung
			}
			# Reset the counter for next time we encounter this event
			hidden_effect = {
				set_variable = {
					which = gene_pillars_generic_traps_and_treasure_count
					value = 0
				}
			}
		}
	}
}

	if = {

	}
