set_government_rank_scripted_effect = {
    custom_tooltip = set_government_rank_$rank$_tt
    hidden_effect = { set_government_rank = $rank$ }
}
increase_government_rank_scripted_effect = {
    if = {
	    limit = { 
            NOT = { government_rank = $rank$ } 
        }
		custom_tooltip = set_government_rank_$rank$_tt
		hidden_effect = { set_government_rank = $rank$ }
	}
        
}
