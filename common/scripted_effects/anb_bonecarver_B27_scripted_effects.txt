reduce_anguish_effect = {
	if = {
		limit = {
			has_country_flag = B27_enable_anguish
		}
		custom_tooltip = reduce_anguish_tt
		hidden_effect = {
			if = { 
				limit = {
					check_variable = {
						which = anguish_level
						value = 0
					}
					NOT = {
						check_variable = {
							which = anguish_level
							value = 1
						}
					}
				}
				add_church_power = 100
			}
			else = {
				subtract_variable = {
					which = anguish_level
					value = 1
				}
			}
			remove_country_modifier =  B27_anguish_1
			remove_country_modifier =  B27_anguish_2
			remove_country_modifier =  B27_anguish_3
			remove_country_modifier =  B27_anguish_4
			remove_country_modifier =  B27_anguish_12
			remove_country_modifier =  B27_anguish_22
			remove_country_modifier =  B27_anguish_32
			
			if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 2
					}
				}
				if = {
					limit = {
						has_country_flag = B27_enable_anguish_2
					}
					add_country_modifier = {
						name = B27_anguish_32
						duration = -1
					}
				}
				else = {
					add_country_modifier = {
						name = B27_anguish_3
						duration = -1
					}
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 1
					}
				}
				if = {
					limit = {
						has_country_flag = B27_enable_anguish_2
					}
					add_country_modifier = {
						name = B27_anguish_22
						duration = -1
					}
				}
				else = {
					add_country_modifier = {
						name = B27_anguish_2
						duration = -1
					}
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 0
					}
				}
				if = {
					limit = {
						has_country_flag = B27_enable_anguish_2
					}
					add_country_modifier = {
						name = B27_anguish_12
						duration = -1
					}
				}
				else = {
					add_country_modifier = {
						name = B27_anguish_1
						duration = -1
					}
				}
			}
		}
	}
}
increase_anguish_effect = {
		if = {
		limit = {
			has_country_flag = B27_enable_anguish
		}
		custom_tooltip = increase_anguish_tt
		hidden_effect = {
			if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 2
					}
					has_country_flag = B27_enable_anguish_2
				}
				add_legitimacy = -10
				add_devotion = -10
				add_republican_tradition = -3
				add_horde_unity = -10
			}
			else_if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 3
					}
				}
				add_stability = -1
			}
			else = {
				change_variable = {
					which = anguish_level
					value = 1
				}
			}
			remove_country_modifier =  B27_anguish_1
			remove_country_modifier =  B27_anguish_2
			remove_country_modifier =  B27_anguish_3
			remove_country_modifier =  B27_anguish_4
			remove_country_modifier =  B27_anguish_12
			remove_country_modifier =  B27_anguish_22
			remove_country_modifier =  B27_anguish_32
			if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 3
					}
				}
				add_country_modifier = {
					name = B27_anguish_4
					duration = -1
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 2
					}
				}
				if = {
					limit = {
						has_country_flag = B27_enable_anguish_2
					}
					add_country_modifier = {
						name = B27_anguish_32
						duration = -1
					}
				}
				else = {
					add_country_modifier = {
						name = B27_anguish_3
						duration = -1
					}
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = anguish_level
						value = 1
					}
				}
				if = {
					limit = {
						has_country_flag = B27_enable_anguish_2
					}
					add_country_modifier = {
						name = B27_anguish_22
						duration = -1
					}
				}
				else = {
					add_country_modifier = {
						name = B27_anguish_2
						duration = -1
					}
				}
			}
		}
	}
}
