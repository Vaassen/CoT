#Scripted effects for Verne
increase_size_network_of_adventure = {
	288 = {
		change_variable = {
			which = verne_network_of_adventure
			value = 1
		}
	}
	if = {
		limit = { mission_completed = A33_in_search_of_adventure }
		if = {
			limit = { NOT = { check_variable = { which = verne_network_of_adventure value = 4 } } }
			add_country_modifier = {
				name = verne_network_of_adventure_tier_1
				duration = -1
			}
		}
		else_if = {
			limit = { NOT = { check_variable = { which = verne_network_of_adventure value = 8 } } }
			remove_country_modifier = { name = verne_network_of_adventure_tier_1 }
			add_country_modifier = {
				name = verne_network_of_adventure_tier_2
				duration = -1
			}
		}
		else_if = {
			limit = { NOT = { check_variable = { which = verne_network_of_adventure value = 13 } } }
			remove_country_modifier = { name = verne_network_of_adventure_tier_2 }
			add_country_modifier = {
				name = verne_network_of_adventure_tier_3
				duration = -1
			}
		}
		else_if = {
			limit = { NOT = { check_variable = { which = verne_network_of_adventure value = 20 } } }
			remove_country_modifier = { name = verne_network_of_adventure_tier_3 }
			add_country_modifier = {
				name = verne_network_of_adventure_tier_4
				duration = -1
			}
		}
		else = {
			remove_country_modifier = { name = verne_network_of_adventure_tier_4 }
			add_country_modifier = {
				name = verne_network_of_adventure_tier_5
				duration = -1
			}
		}
	}
}

create_port_of_adventure = {
	add_province_modifier = {
		name = verne_port_of_adventure
		duration = -1
	}
	increase_size_network_of_adventure = yes
}
