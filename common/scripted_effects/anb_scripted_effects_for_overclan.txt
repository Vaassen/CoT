overclan_create_voltoj_academy = {
	set_province_flag = F78_voltoj_academy
	if = {
		limit = { province_group = hold_province }
		add_permanent_province_modifier = {
			name = F78_small_voltoj_academy_hold
			duration = -1
		}
	}
	else = {
		add_permanent_province_modifier = {
			name = F78_small_voltoj_academy
			duration = -1
		}
	}

	if = {
		limit = { has_estate_privilege = estate_voltoj_academy_privileges }
		add_province_modifier = {
			name = f78_academy_privs
			duration = -1
		}
	}
	if = {
		limit = { has_enacted_parliament_action = { parliament_action = overclan_improve_city_accessibility_act } }
		add_harpy_minority_size_effect = yes
		add_province_modifier = {
			name = F78_harpy_upper_cities_mod
			duration = -1
		}
	}
}
