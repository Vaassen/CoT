overclan_advisor_election_effect = {
	overclan_clear_discontent_bureaucrats_effect = yes
	set_province_flag = f78_ca_$region$
	add_province_modifier = {
		name = F78_ca_$region$
		duration = -1
	}
	overclan_unique_building_becomes_advised = yes
}

overclan_reset_advisor_option_flags = {
	set_country_flag = $region$_elected
	voltoj_city_province = {
		limit = { has_province_flag = f78_ca_$region$ }
		clr_province_flag = f78_ca_$region$
	}
}

overclan_clear_discontent_bureaucrats_effect = {
	remove_province_modifier = F78_discontent_bureaucrats_1

	remove_province_modifier = F78_discontent_bureaucrats_3

	remove_province_modifier = F78_discontent_bureaucrats_5
}

overclan_increase_discontent_bureaucrats_effect = {
	if = {
		limit = { has_province_modifier = F78_ca_overboss }
		#overboss means they're content
	}
	else = {
		if = {
			limit = { has_province_modifier = F78_discontent_bureaucrats_5 }
			#already max
		}
		else_if = {
			limit = { has_province_modifier = F78_discontent_bureaucrats_1 }
			remove_province_modifier = F78_discontent_bureaucrats_1
			add_province_modifier = {
				name = F78_discontent_bureaucrats_2
				duration = -1
			}
		}
		else_if = {
			limit = { has_province_modifier = F78_discontent_bureaucrats_2 }
			remove_province_modifier = F78_discontent_bureaucrats_2
			add_province_modifier = {
				name = F78_discontent_bureaucrats_3
				duration = -1
			}
		}
		else_if = {
			limit = { has_province_modifier = F78_discontent_bureaucrats_3 }
			remove_province_modifier = F78_discontent_bureaucrats_3
			add_province_modifier = {
				name = F78_discontent_bureaucrats_4
				duration = -1
			}
		}
		else_if = {
			limit = { has_province_modifier = F78_discontent_bureaucrats_4 }
			remove_province_modifier = F78_discontent_bureaucrats_4
			add_province_modifier = {
				name = F78_discontent_bureaucrats_5
				duration = -1
			}
		}
		else = {#no modifier
			add_province_modifier = {
				name = F78_discontent_bureaucrats_1
				duration = -1
			}
		}
	}
}

spend_adm_development = {
	for = {
		amount = $amount$
		effect = "
			random_list = {
				1000 = {
					trigger = { base_tax = 2 }
					add_base_tax = -1
				}
				1 = {
					trigger = { base_production = 2 }
					add_base_production = -1
				}
				1 = {
					trigger = { base_manpower = 2 }
					add_base_manpower = -1
				}
			}
		"
	}
}

overclan_upgrade_academy_to = {
	[[Large]
		if = {
			limit = { province_group = voltoj_hold_city_province }

			spend_adm_development = { amount = 10 }
			remove_province_modifier = F78_medium_voltoj_academy_hold
			add_permanent_province_modifier = {
				name = F78_large_voltoj_academy_hold
				duration = -1
			}
		}
		else = {
			spend_adm_development = { amount = 20 }
			remove_province_modifier = F78_medium_voltoj_academy
			add_permanent_province_modifier = {
				name = F78_large_voltoj_academy
