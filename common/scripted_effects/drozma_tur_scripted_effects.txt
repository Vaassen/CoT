dt_introspection_dev_effect = {
	if = {
		limit = {
			owner = {
				has_church_aspect = spirit_introspection_aspect
			}
			has_tax_building_trigger = yes
			religion = drozma_tur
		}
		owner = {
			change_variable = {
				which = dt_introspection_dev
				value = 1
			}
			if = {
				limit = {
					has_church_aspect = community_advancement_aspect_dip
					check_variable = {
						which = dt_introspection_dev
						value = 10
					}
					NOT = {
						check_variable = {
							which = dt_inno_from_dev
							value = 10
						}
					}
				}
				change_innovativeness = 1
				change_variable = {
					which = dt_inno_from_dev
					value = 1
				}
			}
		}
	}
	if = {
		limit = {
			owner = {
				has_church_aspect = spirit_introspection_aspect
				NOT = {
					has_church_aspect = community_advancement_aspect_dip
				}
				check_variable = {
					which = dt_introspection_dev
					value = 9
				}
			}
			has_tax_building_trigger = yes
			religion = drozma_tur
		}
		owner = {
			change_variable = {
				which = dt_introspection_ten_stack
				value = 1
			}
			subtract_variable = {
				which = dt_introspection_dev
				value = 9
			}
		}
	}
}
