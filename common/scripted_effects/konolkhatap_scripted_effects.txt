
konolkhatep_rule_pack_effect = {
	if = {
		limit = { exists = $tag$ NOT = { $tag$ = { is_subject_of_type = medasi_subject } } }
		$tag$ = { 
			kill_ruler = yes 
			hidden_effect = { grant_independence = yes }
		}
		create_subject = {
			subject_type = personal_union
			who = $tag$
		}	
	}
	else = {
		release = $tag$
		create_subject = {
			subject_type = personal_union
			who = $tag$
		}
	}
	$tag$ = {
		add_liberty_desire = -50
		add_opinion = { who = ROOT modifier = opinion_loyal_pack }
	}
}

	hidden_effect = {
		change_variable = { gnoll_disaster_counter = 1 }

			limit = { is_variable_equal = { which = gnoll_disaster_counter value = 10 } }

		}
	}
}

#Road building

konolkhatep_road_construction = {
	if = { #these provinces are special because they border multiple possible next provinces
		limit = {
			OR = {
				province_id = 625
				province_id = 4412
				province_id = 4399
				province_id = 4408
				province_id = 4496
				province_id = 4500
				province_id = 4611
				province_id = 4613
			}
		}
		trigger_switch = {
			on_trigger = province_id
			625 = { 626 = { save_event_target_as = golden_highway_next_province } }
			4412 = { 4411 = { save_event_target_as = golden_highway_next_province } }
			4399 = { 4420 = { save_event_target_as = golden_highway_next_province } }
			4408 = { 4409 = { save_event_target_as = golden_highway_next_province } }
			4496 = { 4493 = { save_event_target_as = golden_highway_next_province } }
			4500 = { 4620 = { save_event_target_as = golden_highway_next_province } }
			4611 = { 4938 = { save_event_target_as = golden_highway_next_province } }
			4613 = { 4559 = { save_event_target_as = golden_highway_next_province } }
		}
	}
	else = { #so it doesn't fire it with every single province group, prevents the construction from moving to an adjacent province group
		trigger_switch = {
			on_trigger = province_group
			konolkhatep_road_rrammoton = { konolkhatep_road_construction_select_next_province = { section = bulwar_proper } }
			konolkhatep_road_udzalmmoton = { konolkhatep_road_construction_select_next_province = { section = far_bulwar } }
			konolkhatep_road_igraxmmoton = { konolkhatep_road_construction_select_next_province = { section = far_salahad } }
			konolkhatep_road_izermmoton = { konolkhatep_road_construction_select_next_province = { section = rahen_coast } }
			konolkhatep_road_zuraxmmoton = { konolkhatep_road_construction_select_next_province = { section = dhujat } }
			konolkhatep_road_manmmoton = { konolkhatep_road_construction_select_next_province = { section = xianjie } }
			konolkhatep_road_agtmmoton = { konolkhatep_road_construction_select_next_province = { section = upper_yanshen } }
			konolkhatep_road_ozhxummoton = { konolkhatep_road_construction_select_next_province = { section = yanshen_coast } }
			konolkhatep_road_tzazikmmoton = { konolkhatep_road_construction_select_next_province = { section = south_haless } }
			konolkhatep_road_yaikmmoton = { konolkhatep_road_construction_select_next_province = { section = shamakhad } }
			konolkhatep_road_xhurmmoton = { konolkhatep_road_construction_select_next_province = { section = shamakhad } }
		}
	}
}

konolkhatep_road_construction_select_next_province = {
	random_neighbor_province = {
		limit = {
			province_group = konolkhatep_road_$section$
			NOT = {
				has_province_modifier = konolkhatep_rrammoton
				has_province_modifier = konolkhatep_udzalmmoton
				has_province_modifier = konolkhatep_igraxmmoton
				has_province_modifier = konolkhatep_izermmoton
				has_province_modifier = konolkhatep_zuraxhmmoton
				has_province_modifier = konolkhatep_manmmoton
				has_province_modifier = konolkhatep_agtmmoton
				has_province_modifier = konolkhatep_ozhxummoton
				has_province_modifier = konolkhatep_tzazikmmoton
				has_province_modifier = konolkhatep_yaikmmoton
				has_province_modifier = konolkhatep_xhurmmoton
			}
		}
		save_event_target_as = konolkhatep_road_next_province
	}
}
