calculate_lichdom_chance = {
	set_variable ={
		which=$which$ value=0
	}
	
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_2_box
		}
		change_variable = {
			which = $which$
			value = 1
		}
	}
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_5_b
		}
		change_variable = {
			which = $which$
			value = 1
		}
	}
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_5_c
		}
		change_variable = {
			which = $which$
			value = 2
		}
	}
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_6_b
		}
		change_variable = {
			which = $which$
			value = 1
		}
	}
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_6_c
		}
		change_variable = {
			which = $which$
			value = 2
		}
	}
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_7_unicorn
		}
		change_variable = {
			which = $which$
			value = 1
		}
	}
	if = {
		limit = {
			has_ruler_flag = magic_project_lichdom_8_a
		}
		change_variable = {
			which = $which$
			value = 1
		}
	}
}

		}

			has_country_flag = aldresia_renowned_proficiency
		}

		limit = {
			OR = {
				has_reform = magocracy_reform
				has_reform = magisterium_reform
			}
		}

			has_reform = oracular_order_reform # Oracle Order in Haless

# Can handle experience = 50, 100, 200, 250, 300 or 500. If you want to add more then you need to create new tooltips for them.
increase_study_experience = {
	[[experience]
		custom_tooltip = magic_study_experience_$experience$_tt # Loc only exists for 50, 100, 200, 250, 300 and 500 currently
		
		update_magic_study_experience_modifier = { type = ruler }
		hidden_effect = { set_variable = { experienceGain = $experience$ } }
		# If the modifier is positive or negative
		if = {
			limit = { 
				NOT = { is_variable_equal = { which = ruler_experience_modifier value = 0 } } # XP modifier not 0
			}
			custom_tooltip = magic_study_experience_modifier_tt			
			hidden_effect = {
				set_variable = { which = expFoo which = ruler_experience_modifier }
				divide_variable = { expFoo = 100 }
				change_variable = { expFoo = 1 }
				multiply_variable = { which = experienceGain which = expFoo }
			}
		}

		hidden_effect = {
			change_variable = {
				which = ruler_study_experience 
				which = experienceGain
			}
		}
	]
	hidden_effect = {
		country_event = { id = magic_study.100 days = 1 }
	}
}

# Country flags and ruler flags are ENTIRELY separate things.
# You can set a country flag named "A" and a ruler flag named "A" and they will be tracked separately.
# A ruler flag named "A" will NOT fulfill the requirements for has_country_flag = A,
# and the clr_country_flag effect will NOT clear ruler flags.

# The same, however, is NOT true of modifiers.
# A ruler modifier "A" WILL fulfill the requirements for has_country_modifier = A.
# It can be removed with the remove_country_modifier effect (there is no remove_ruler_modfier effect).
# A country modifier will NOT count as a ruler modifier for the purposes of has_ruler_modfier trigger,
# but you ALSO cannot use the add_ruler_modifier effect in any meaningful way while you have a country modifier of the same name.

#monthly mana calc, updates for all countries with mages
update_mana_calc = {
	change_variable = {
		which = currentMana
		which = manaRegenerationRate
	}
	cap_mana = yes #prevent mana overflow. We'll be doing this a lot.

	#Update player mana bar
	if = { limit = { ai = no }
		update_mana_bar = yes
	}
}

update_mana_bar = {
	if = { limit = { check_variable = { which = currentMana value = 0 } }
		set_variable = { which = current_mana_percent which = currentMana }
		divide_variable = { which = current_mana_percent which = manaCapacity }
		multiply_variable = { current_mana_percent = 100 }
	}
	else = {
		set_variable = { current_mana_percent = 0 }
	}
}

#for if mana exceeds mana capacity
cap_mana = {
	if = {
		limit = {
			check_variable = {
				which = currentMana
				which = manaCapacity
			}
		}
		set_variable = {
			which = currentMana
			which = manaCapacity
		}
	}
}

#checking magical infrastructure to increase mana capacity.
update_mana_cap = {
	if = {
		limit = { magical_infrastructure_level = { level = 5 } }
		set_variable = {
			which = manaCapacity
			value = 400
		}
	}
	else_if = {
		limit = { magical_infrastructure_level = { level = 4 } }
		set_variable = {
			which = manaCapacity
			value = 350
		}
	}
	else_if = {
		limit = { magical_infrastructure_level = { level = 3 } }
		set_variable = {
			which = manaCapacity
			value = 300
		}
	}
	else_if = {
		limit = { magical_infrastructure_level = { level = 2 } }
		set_variable = {
			which = manaCapacity
			value = 250
		}
	}
	else_if = {
		limit = { magical_infrastructure_level = { level = 1 } }
		set_variable = {
			which = manaCapacity
			value = 200
		}
	}
	else = { #TODO Jothell remove fallback once infrastructure is in place
		set_variable = {
			which = manaCapacity
			value = 200
		}
	}
}

# Triggered by magic_estate.107
update_mana_regen = {
	hidden_effect = {
		set_variable = {
			which = manaRegenerationRate
			value = 0.8
		}
		if = {
			limit = { magical_infrastructure_level = { level = 5 } }
			change_variable = {
				which = manaRegenerationRate
				value = 1
			}
		}
		else_if = {
			limit = { magical_infrastructure_level = { level = 4 } }
			change_variable = {
				which = manaRegenerationRate
				value = 0.8
			}
		}
		else_if = {
			limit = { magical_infrastructure_level = { level = 3 } }
			change_variable = {
				which = manaRegenerationRate
				value = 0.6
			}
		}
		else_if = {
			limit = { magical_infrastructure_level = { level = 2 } }
			change_variable = {
				which = manaRegenerationRate
				value = 0.4
			}
		}
		else_if = {
			limit = { magical_infrastructure_level = { level = 1 } }
			change_variable = {
				which = manaRegenerationRate
				value = 0.2
			}
		}

		if = {
			limit = { has_country_modifier = necromancy_1_0_mod }
			change_variable = {
				which = manaRegenerationRate
				value = 0.5
			}
		}

		if = {
			limit = { has_country_modifier = undead_military }
			change_variable = {
				which = manaRegenerationRate
				value = -0.5
			}
		}

		if = { #This single thing is the reason why we need to update mana regen each time (instead of changing the variable as appropriate)
			limit = { has_country_flag = magic_artificery_mixed }
			divide_variable = {
				which = manaRegenerationRate
				value = 2
			}
		}
	}
}

			limit = {

			}

### A SCHOOL DEFINES UNLOCK PATTERNS AND OTHER GROUPINGS FOR SPELLS (AKA YOU WILL UNLOCK CONJURATION 2 AFTER CONJURATION 1)
### A LEVEL IS WHAT DEFINES THE COST OF THE SPELL IN MANA: WE RESPECT FOUR LEVELS ONLY (0, 1, 2, 3) THOUGH YOU CAN MAKE MORE
### AN ID DISTINGUISHES BETWEEN DIFFERENT SPELLS OF THE SAME SCHOOL AND LEVEL
### EXAMPLE (abjuration_2_0_effect) and (abjuration_2_1_effect). BOTH ABJURATION, BOTH LEVEL 2, DIFFERENT ID'S

generic_cast_spell = {
	if = {
		limit = {
			OR = {
				has_country_flag = $school$_$level$_$id$_spell_free
				has_country_flag = $school$_spell_free
			}
		}
		custom_tooltip = spell_is_free_tt
	}
	else = {
		pay_mana_for_level_$level$_spell = { school = $school$ id = $id$ }
	}

	if = {
		limit = { has_country_flag = $school$_$level$_$id$_spell_free_temp }
		custom_tooltip = for_next_cast_only_tt
		clr_country_flag = $school$_$level$_$id$_spell_free_temp
		clr_country_flag = $school$_$level$_$id$_spell_free
	}
	else_if = {
		limit = { has_country_flag = $school$_$level$_$id$_spell_half_temp }
		custom_tooltip = for_next_cast_only_tt
		clr_country_flag = $school$_$level$_$id$_spell_half_temp
		clr_country_flag = $school$_$level$_$id$_spell_half
	}
	else_if = {
		limit = { has_country_flag = $school$_spell_free_temp }
		custom_tooltip = for_next_cast_only_tt
		clr_country_flag = $school$_spell_free_temp
		clr_country_flag = $school$_spell_free
	}
	else_if = {
		limit = { has_country_flag = $school$_spell_half_temp }
		custom_tooltip = for_next_cast_only_tt
		clr_country_flag = $school$_spell_half_temp
		clr_country_flag = $school$_spell_half
	}

	[[siege]
	if = {
		limit = { event_target:spell_target_province = { owner = { check_variable = { which = abjuration_project_level value = 3 } } } }
		custom_tooltip = magical_fortress_magic_tax_tt
		change_mana = { amount = -50 }
	}
	]

	[[sabotage]
	if = {
		limit = {
			OR = {
				any_rival_country = { check_variable = { which = abjuration_project_level value = 3 } }
				any_war_enemy_country = { check_variable = { which = abjuration_project_level value = 3 } }
			}
		}
		custom_tooltip = magical_fortress_magic_tax_tt
		change_mana = { amount = -50 }
	}
	]

	[[war]
	custom_tooltip = only_one_war_magic_tt
	hidden_effect = {
		trigger_switch = {
			on_trigger = has_country_modifier
			abjuration_0_0_mod = { remove_country_modifier = abjuration_0_0_mod }
			abjuration_2_1_mod = { remove_country_modifier = abjuration_2_1_mod }
			abjuration_3_0_mod = { remove_country_modifier = abjuration_3_0_mod }
			divination_2_1_mod = { remove_country_modifier = divination_2_1_mod }
			enchantment_1_0_mod = { remove_country_modifier = enchantment_1_0_mod }
			enchantment_2_0_mod = { remove_country_modifier = enchantment_2_0_mod }
			evocation_1_0_mod = { remove_country_modifier = evocation_1_0_mod }
			evocation_1_1_mod = { remove_country_modifier = evocation_1_1_mod }
			evocation_3_0_mod = { remove_country_modifier = evocation_3_0_mod }
		}
	}
	]

	$school$_$level$_$id$_effect = yes
	update_mana_bar = yes
}
