# type = tax | production | manpower
S50_sacred_studies_effect = {
    random_owned_province = {
        limit = {
            has_terrain = shadow_swamp_terrain
            NOT = { has_province_modifier = S50_sacred_studies_$type$ }
        }
        add_base_$type$ = 1
        add_province_modifier = {
            name = S50_sacred_studies_$type$
            duration = 3650
        }
    }
}

S50_remove_advisor_modifiers_effect = {
		if = { limit = { NOT = { OR =  { theologian = 1  inquisitor = 1 } } } 			    remove_country_modifier = S50_shadow_pact_fortune_tellers } 
        if = { limit = { NOT = { OR =  { army_organiser = 1  master_recruiter = 1 } } } 	remove_country_modifier = S50_pack_leaders } 
        if = { limit = { NOT = { OR =  { artist = 1  philosopher = 1 } } } 			        remove_country_modifier = S50_sharp_tongues } 
        if = { limit = { NOT = { OR =  { diplomat = 1  statesman = 1 } } } 			        remove_country_modifier = S50_diplomatic_liasons } 
}

S50_add_advisor_modifiers_effect = {
		if = { limit = { OR =  { theologian = 1  inquisitor = 1 } } 		    add_country_modifier = { name = S50_shadow_pact_fortune_tellers duration = -1 } }
        if = { limit = { OR =  { army_organiser = 1  master_recruiter = 1 } } 	add_country_modifier = { name = S50_pack_leaders duration = -1 } } 
        if = { limit = { OR =  { artist = 1  philosopher = 1 } } 			    add_country_modifier = { name = S50_sharp_tongues duration = -1 } } 
        if = { limit = { OR =  { diplomat = 1  statesman = 1 } } 			    add_country_modifier = { name = S50_diplomatic_liasons duration = -1 } }     
}
