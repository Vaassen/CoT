    if = { limit = { has_global_flag = cot_eternal_winter_2 } clr_global_flag = cot_eternal_winter_2 set_global_flag = cot_eternal_winter_1 }
    if = { limit = { has_global_flag = cot_eternal_winter_3 } clr_global_flag = cot_eternal_winter_3 set_global_flag = cot_eternal_winter_2 }

    if = { limit = { has_global_flag = cot_eternal_winter_5 } clr_global_flag = cot_eternal_winter_5 set_global_flag = cot_eternal_winter_4 }
    
    cot_winter_set_modifiers = yes
}
