    every_province = {
        limit = { cot_winter_affected_region = yes }
        remove_province_modifier = cot_eternal_winter_debuff_5
        remove_province_modifier = cot_eternal_winter_debuff_4
        remove_province_modifier = cot_eternal_winter_debuff_3
        remove_province_modifier = cot_eternal_winter_debuff_2

    if = { limit = { has_global_flag = cot_eternal_winter_1 } clr_global_flag = cot_eternal_winter_1 set_global_flag = cot_eternal_winter_0 }
    if = { limit = { has_global_flag = cot_eternal_winter_2 } clr_global_flag = cot_eternal_winter_2 set_global_flag = cot_eternal_winter_1 }

    if = { limit = { has_global_flag = cot_eternal_winter_5 } clr_global_flag = cot_eternal_winter_5 set_global_flag = cot_eternal_winter_4 }
    
    every_province = {
        limit = { cot_winter_affected_region = yes }
        if = { limit = { has_global_flag = cot_eternal_winter_4 } add_province_modifier = { name = cot_eternal_winter_debuff_4 duration = -1 } }
        if = { limit = { has_global_flag = cot_eternal_winter_3 } add_province_modifier = { name = cot_eternal_winter_debuff_3 duration = -1 } }
        if = { limit = { has_global_flag = cot_eternal_winter_2 } add_province_modifier = { name = cot_eternal_winter_debuff_2 duration = -1 } }

        if = { limit = { has_global_flag = cot_eternal_winter_0 } add_province_modifier = { name = cot_eternal_winter_debuff_0 duration = -1 } }
    }
}
