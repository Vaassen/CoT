S45_upgrade_mercenary_excellence = {
    if = {
        limit = {
            has_country_modifier = S45_mercenary_excellence_4
        }
        if = {
            limit = {
                check_variable = {
                    which = S45_mercenary_momentum
                    value = 100
                }
            }
            add_country_modifier = {
                name = S45_mercenary_excellence_5
                duration = 3650
            }
            set_variable = {
                which = S45_mercenary_momentum_base_value
                value = 25
            }
        }
        else = {
            add_country_modifier = {
                name = S45_mercenary_excellence_4
                duration = 3650

S45_change_momentum = {
    # TODO: tooltip that shows the value before the change

    change_variable = {
        name = S45_mercenary_momentum
        value = $value$
    }
    hidden_effect = {
        if = {
            limit = {
                check_variable = {
                    name = S45_mercenary_momentum
                    value = 101
                }
            }
            set_variable = {
                name = S45_mercenary_momentum
                value = 100
            }
        }
        else_if = {
            limit = {
                NOT = {
                    check_variable = {
                        name = S45_mercenary_momentum
                        value = 0
                    }
                }
            }
            set_variable = {
                name = S45_mercenary_momentum
                value = 0
            }
        }
    }
}
