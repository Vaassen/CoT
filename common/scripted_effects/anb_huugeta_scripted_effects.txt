S45_change_momentum = {
    custom_tooltip = S45_current_momentum_tt
    change_variable = {
        which = mercenaryMomentum
        value = $value$
    }
    hidden_effect = {
        # Increase score tracking for this action type
        change_variable = {
            which = mercActionValue_$action$
            value = $value$
        }

        if = {
            limit = {
                check_variable = {
                    which = mercenaryMomentum
                    value = 101
                }
            }
            set_variable = {
                which = mercenaryMomentum
                value = 100
            }
        }
        else_if = {
            limit = {
                NOT = {
                    check_variable = {
                        which = mercenaryMomentum
                        value = 0
                    }
                }
            }
            set_variable = {
                which = mercenaryMomentum
                value = 0
            }
        }
    }
}
