S45_upgrade_mercenary_excellence = {
    if = {
        limit = {
            has_country_modifier = S45_mercenary_excellence_4
        }
        if = {
            limit = {
                check_variable = {
                    which = mercenaryMomentum
                    value = 100
                }
            }
            add_country_modifier = {
                name = S45_mercenary_excellence_5
                duration = 3650
            }
            set_variable = {
                which = baseMercenaryMomentum
                value = 25
            }
        }

                name = S45_mercenary_excellence_4

S45_change_momentum = {
    custom_tooltip = S45_current_momentum_tt
    change_variable = {
        which = mercenaryMomentum
        value = $value$
    }
    hidden_effect = {
        # Increase score tracking for this action type
        change_variable = {
            which = mercActionValue_$action$
            value = $value$
        }

        if = {
            limit = {
                check_variable = {
                    which = mercenaryMomentum
                    value = 101
                }
            }
            set_variable = {
                which = mercenaryMomentum
                value = 100
            }
        }
        else_if = {
            limit = {
                NOT = {
                    check_variable = {
                        which = mercenaryMomentum
                        value = 0
                    }
                }
            }
            set_variable = {
                which = mercenaryMomentum
                value = 0
            }
        }
    }
}
