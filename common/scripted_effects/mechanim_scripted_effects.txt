on_province_owner_change_migration = {
	if = {
		limit = { owner = { religion = manago_poku } }
		owner = {
			if = {
				limit = {
					NOT = {
						has_country_flag = manago_poku_setup
					}
				}
				set_country_flag = manago_poku_setup
				country_event = {
					id = manago_poku.56
				}
			}
		}
		if = {
			limit = { has_province_modifier = dormant_mechanim NOT = { has_province_modifier = scoured_dormant_mechanim } }
			if = {
				limit = { ai = no }
				add_permanent_province_modifier = {
					name = scoured_dormant_mechanim
					duration = -1
				}
			}
			owner = {
				add_church_power = 75
			}
		}
		if = {
			limit = {
				owner = { NOT = { has_country_flag = manago_poku_visited_holohana } }
				has_province_modifier = holohana_rumbling
			}
			owner = {
				set_country_flag = manago_poku_visited_holohana
				country_event = {
					id = manago_poku.57
				}
			}
		}
		else_if = {
			limit = {
				owner = { NOT = { has_country_flag = manago_poku_visited_rumbling } }
				any_neighbor_province = { has_province_modifier = holohana_rumbling }
			}
			owner = {
				set_country_flag = manago_poku_visited_rumbling
				add_church_power = 50
			}
		}
		if = {
			limit = {
				owner = { NOT = { has_country_flag = manago_poku_visited_ensuary } }
				any_neighbor_province = { has_province_modifier = river_estuary_modifier }
			}
			owner = {
				set_country_flag = manago_poku_visited_ensuary
				add_church_power = 50
			}
		}
		if = {
			limit = {
				owner = { ai = no }
				any_neighbor_province = {
					owner = {
						has_country_modifier = mechanim_administration
						NOT = { has_country_flag = manago_poku_mechanim_met_player }
					}
				}
			}
			random_neighbor_province = {
				limit = {
					owner = {
						has_country_modifier = mechanim_administration
