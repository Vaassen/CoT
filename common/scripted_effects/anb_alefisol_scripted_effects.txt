#Rinta Dark Magic check for end of Invasion
alefisol_invasion_siege_won_effect = {
	if = {
		limit = { #Alefisol losing
			S93 = {
				has_global_flag = darkmagicks_flag
				OR = {
					war_score = -99
					AND = {
						check_variable = {
							which = alefisol_lost_6
							value = 6
						}
						OR = {
							capital_scope = {
								NOT = { controlled_by = S93 }
							}
							war_score = -75
						}
					}
				}
			}
		}
		every_known_country = {
			limit = {
				has_disaster = darkmagicks
			}
			country_event = { id = rinta_seeker_events.211 }
		}
	}
	if = {
		limit = { #Alefisol win
			S93 = {
				has_global_flag = darkmagicks_flag
				is_in_war = {
					casus_belli = cb_alefisol_war
					war_score = 99
				}
			}
		}
		S93 = { country_event = { id = rinta_seeker_events.212 } }
	}
}
