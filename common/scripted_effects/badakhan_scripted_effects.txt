
# binary tree (min:0 max:15)
export_number_of_estate_privs_to_var = {
	if = { # x >= 8
		limit = { num_of_estate_privileges = { estate = $estate$ value = 8 } }
		if = { # x >= 12
			limit = { num_of_estate_privileges = { estate = $estate$ value = 12 } }
			if = { # x >= 14
				limit = { num_of_estate_privileges = { estate = $estate$ value = 14 } }
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 15 } }
					set_variable = { $var$ = 15 }
				}
				else = {
					set_variable = { $var$ = 14 }
				}
			}
			else = { # 12 <= x < 14
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 13 } }
					set_variable = { $var$ = 13 }
				}
				else = {
					set_variable = { $var$ = 12 }
				}
			}
		}
		else = { # 8 <= x < 12
			if = { # 10 <= x < 12
				limit = { num_of_estate_privileges = { estate = $estate$ value = 10 } }
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 11 } }
					set_variable = { $var$ = 11 }
				}
				else = {
					set_variable = { $var$ = 10 }
				}
			}
			else = { # 8 <= x < 10
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 9 } }
					set_variable = { $var$ = 9 }
				}
				else = {
					set_variable = { $var$ = 8 }
				}
			}
		}
	}
	else = { # x < 8
		if = { # 4 <= x < 8
			limit = { num_of_estate_privileges = { estate = $estate$ value = 4 } }
			if = { # 6 <= x < 8
				limit = { num_of_estate_privileges = { estate = $estate$ value = 6 } }
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 7 } }
					set_variable = { $var$ = 7 }
				}
				else = {
					set_variable = { $var$ = 6 }
				}
			}
			else = { # 4 <= x < 6
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 5 } }
					set_variable = { $var$ = 5 }
				}
				else = {
					set_variable = { $var$ = 4 }
				}
			}
		}
		else = { # x < 4
			if = { # 2 <= x < 4
				limit = { num_of_estate_privileges = { estate = $estate$ value = 2 } }
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 3 } }
					set_variable = { $var$ = 3 }
				}
				else = {
					set_variable = { $var$ = 2 }
				}
			}
			else = { # x < 2
				if = {
					limit = { num_of_estate_privileges = { estate = $estate$ value = 1 } }
					set_variable = { $var$ = 1 }
				}
				else = {
					set_variable = { $var$ = 0 }
				}
			}
		}
	}
}

badakhan_monthly_power_gain = {
	if = {
		limit = { NOT = { has_country_flag = badakhan_amriel_confronted_0 } }
		set_variable = { badakhan_num_privs = 0 }
		badakhan_monthly_power_gain_helper = { estate = estate_church }
		badakhan_monthly_power_gain_helper = { estate = estate_nobles }
		badakhan_monthly_power_gain_helper = { estate = estate_burghers }
		badakhan_monthly_power_gain_helper = { estate = estate_mages }
		badakhan_monthly_power_gain_helper = { estate = estate_adventurers }
		badakhan_monthly_power_gain_helper = { estate = estate_artificers }
		trigger_switch = {
			on_trigger = has_country_flag
			badakhan_amriel_confronted_1 = { set_variable = { badakhan_power_gain = 0.01 } }
			badakhan_amriel_confronted_2 = { set_variable = { badakhan_power_gain = 0.02 } }
			badakhan_amriel_confronted_3 = { set_variable = { badakhan_power_gain = 0.03 } }
		}
		multiply_variable = {
			which = badakhan_power_gain
			which = badakhan_num_privs
		}
		set_variable = {
			which = badakhan_power_gain_temp
			which = badakhan_power_gain
		}
		gov_power_var_effect_granular = {
			mechanic = badakhan_orcish_soul_mechanic
			power = badakhan_orcish_soul_struggle
			var = badakhan_power_gain_temp
			sign = +
		}
	}
}

badakhan_monthly_power_gain_helper = {
	export_number_of_estate_privs_to_var = { estate = $estate$ var = temp_num_privs }
	change_variable = {
		which = badakhan_num_privs
		which = temp_num_privs
	}
}

			clr_country_flag = badakhan_amriel_confronted_2

		}
	}
}
