harpylen_new_heir = {
	if = {
		limit = { mission_completed = F98_godking_dragon }
		random_list = {
			1 = { force_add_heir_personality = { personality = iron_crowned_personality } }
			1 = { force_add_heir_personality = { personality = peasant_revolt_leader_personality } }
			1 = { force_add_heir_personality = { personality = great_engineer_personality } }
			1 = { force_add_heir_personality = { personality = legendary_conqueror_personality } }
			1 = { force_add_heir_personality = { personality = merchant_basileus_personality } }
		}
	}
	if = {
		limit = { mission_completed = F98_true_queendom }
		if = {
			limit = { religion = the_hunt }
			add_country_modifier = {
				name = harpy_crane_doctrine_heir
				duration = -1
			}
			add_country_modifier = {
				name = harpy_myna_doctrine_heir
				duration = -1
			}
			add_country_modifier = {
				name = harpy_eagle_doctrine_heir
				duration = -1
			}
		}
		else = {
			hidden_effect = {
				while = {
					limit = { NOT = { heir_adm = 5 } }
					change_heir_adm = 1
				}
				while = {
					limit = { NOT = { heir_dip = 5 } }
					change_heir_dip = 1
				}
				while = {
					limit = { NOT = { heir_mil = 5 } }
					change_heir_mil = 1
				}
			}
		}
	}
}
