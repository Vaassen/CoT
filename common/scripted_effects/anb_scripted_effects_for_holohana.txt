holohana_province_save_effect = {
	random_province = {
		limit = {
			has_province_modifier = holohana_modifier_$holohana$
		}
		save_global_event_target_as = holohana_target_$holohana$
		if = {
			limit = {
				NOT = { is_empty = yes }
			}
			owner = {
				save_global_event_target_as = holohana_province_owner_$holohana$
			}
		}
	}
}

holohana_rebel_spawn_effect = {
	event_target:holohana_target_$holohana$ = {
		spawn_rebels = {
			size = 200
			[[leader_the_watcher] leader = "The Watcher"]
			[[leader_forest_wide_and_dim] leader = "Forest Wide and Dim"]
			[[leader_maliu_fanua] leader = "Maliu Fanua"]
			type = holohana_$holohana$_rebels
		}
		decolonize = THIS
		remove_core = P99
		cede_province = event_target:holohana_province_owner_$holohana$
	}
}

holohana_migrate_effect = {
	event_target:holohana_target_$holohana$ = {
		remove_province_modifier = holohana_modifier_$holohana$
		clear_rebels = yes
		random_empty_neighbor_province = {
			limit = {
				is_valid_holohana_move_target = yes
			}
			add_permanent_province_modifier = {
				name = holohana_modifier_$holohana$
				duration = -1
			}
			set_province_flag = holohana_$holohana$_flag
		}
		random_neighbor_province = {
			limit = {
				is_valid_holohana_move_target = yes
			}
			add_permanent_province_modifier = {
				name = holohana_modifier_$holohana$
				duration = -1
			}
			set_province_flag = holohana_$holohana$_flag
		}
		every_neighbor_province = {
			limit = {
				has_province_modifier = holohana_rumbling
				has_holohana_rumbling = yes
			}
			remove_province_modifier = holohana_rumbling
		}
		every_empty_neighbor_province = {
			limit = {
				has_province_modifier = holohana_rumbling
				has_holohana_rumbling = yes
			}
			remove_province_modifier = holohana_rumbling
		}
	}
	clear_global_event_target = holohana_target_$holohana$
}
