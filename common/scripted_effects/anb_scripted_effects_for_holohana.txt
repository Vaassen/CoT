#Applies all modifiers needed for the Holohana to work
holohana_spawn_effect = {
	P99 = {
		event_target:holohana_target_$holohana$ = {
			set_province_flag = holohana_$holohana$_flag
			if = {
				limit = {
					is_empty = yes
				}
				create_colony = 1000
				add_core = P99
			}
			else = {
				owner = {
					add_country_modifier = {
						name = holohana_forcelimit_modifier
						duration = -1
					}
				}
			}
			holohana_add_rumbling = { holohana = $holohana$ }
		}
	}
}

#Migrates the target Holohana to a new Province.
holohana_migrate_effect = {
	event_target:holohana_target_$holohana$ = {
		#Target selection for later
		random_empty_neighbor_province = {
			limit = {
				is_valid_holohana_move_target = yes
			}
			save_event_target_as = holohana_empty_target_1
		}
		random_neighbor_province = {
			limit = {
				is_valid_holohana_move_target = yes
			}
			save_event_target_as = holohana_city_target_1
		}
		random_neighbor_province = {
			limit = {
				is_valid_holohana_move_target = yes
				NOT = { province_id = event_target:holohana_city_target_1 }
			}
			save_event_target_as = holohana_city_target_2
		}
		#####
		remove_province_modifier = holohana_modifier_$holohana$
		if = { 
			limit = { NOT = { has_global_flag = holohana_$holohana$_dead } } 
			random_list = {
				50 = {
					trigger = {
						has_saved_event_target = holohana_empty_target_1
					}
					event_target:holohana_empty_target_1 = {
						if = { 
							limit = { NOT = { has_global_flag = holohana_$holohana$_dead } }
							add_permanent_province_modifier = {
								name = holohana_modifier_$holohana$
								duration = -1
							}
							save_global_event_target_as = holohana_target_$holohana$
						}
						remove_province_modifier = holohana_rumbling
					}
				}
				30 = { #Splited to 2
					trigger = {
						has_saved_event_target = holohana_empty_target_1
					}
					event_target:holohana_empty_target_1 = {
						add_permanent_province_modifier = {
							name = holohana_modifier_$holohana$
							duration = -1
						}
						save_global_event_target_as = holohana_target_$holohana$
						remove_province_modifier = holohana_rumbling
					}
				}
				20 = { #Otherwise Holohana will have 50/50 to kill migratory tribe as they will only border them
					trigger = {
						has_saved_event_target = holohana_empty_target_2
					}
					event_target:holohana_empty_target_2 = {
						add_permanent_province_modifier = {
							name = holohana_modifier_$holohana$
							duration = -1
						}
						save_global_event_target_as = holohana_target_$holohana$
						remove_province_modifier = holohana_rumbling
					}
				}
			}
		}
		#Remove Rumbling Cities
		every_neighbor_province = {
			limit = {
				has_province_modifier = holohana_rumbling
				NOT = {
					has_2_holohana_rumbling = yes
				}
			}
			remove_province_modifier = holohana_rumbling
		}
		every_neighbor_province = {
			clr_province_flag = holohana_rumbling_flag_$holohana$
		}
		#Remove Rumbling Empty
		every_empty_neighbor_province = {
			limit = {
				has_province_modifier = holohana_rumbling
				NOT = {
					has_2_holohana_rumbling = yes
				}
			}
			remove_province_modifier = holohana_rumbling
		}
		every_empty_neighbor_province = {
			clr_province_flag = holohana_rumbling_flag_$holohana$
		}
	}
	if = {
		limit = { 6234 = { NOT = { has_province_flag = holohana_fleeing_$holohana$ } } }
		6234 = {
			clr_province_flag = $holohana$_recently_migrated
			holohana_migrate_timer_effect = { holohana = $holohana$ }
		}
	}
	else = {
		6234 = {
			change_variable = {
				which = $holohana$FleeingTimer
				value = -1
			}
		}
		if = {
			limit = {
				6234 = {
					check_variable = {
						which = $holohana$FleeingTimer
						value = 1
					}
				}
			}
			set_global_flag = holohana_flight_continues
		}
		else = {
			6234 = {
				clr_province_flag = holohana_fleeing_$holohana$
			}
		}
	}
	6234 = { #Heal after 20 years
		if = {
			limit = {
				had_global_flag = {
					flag = holohana_$holohana$_injured
					days = 7300
				} 
			}
			clr_global_flag = holohana_$holohana$_injured
		}
	}
	if = { #Leave Corpse
		limit = {
			has_global_flag = holohana_$holohana$_dead
		}
		add_permanent_province_modifier = {
			name = holohana_dead_body_$holohana$
			duration = -1
		}
	}
	if = { #Clear Rebels
		limit = {
			NOT = {
				has_holohana_province_modifier = yes
			}
		}
		clear_rebels = yes
	}
}

#Counts the Time until a Migration happens for each $holohana$. Can be used in other effects to speed up the Migration of specific Holohana if wished
holohana_migrate_timer_effect = {
	6234 = {
		if = {
			limit = {
				NOT = {
					has_province_flag = $holohana$_recently_migrated
				}
			}
			set_variable = {
				which = $holohana$MigrateTimer
				value = 0
			}
			set_province_flag = $holohana$_recently_migrated
		}
		change_variable = {
			which = $holohana$MigrateTimer
			value = 1
		}
	}
}

#Sets the Migration Timer i.e how many years need to pass before the Holohana migrates $holohan$ > The Holohana $amount$ > The amount of Years
#If a Global Rampage is going on it will always default to 1 Year Migrations
holohana_set_migrate_timer = {
	6234 = {
		if = {
			limit = {
				has_global_flag = holohana_rampage_global
			}
			set_variable = {
				which = $holohana$MigrateStart
				value = 1
			}
			set_variable = {
				which = $holohana$MigrateStartLimit
				which = $holohana$MigrateStart
			}
			change_variable = {
				which = $holohana$MigrateStartLimit
				value = 1
			}
		}
		else = {
			set_variable = {
				which = $holohana$MigrateStart
				value = $amount$
			}
			set_variable = {
				which = $holohana$MigrateStartLimit
				which = $holohana$MigrateStart
			}
			change_variable = {
				which = $holohana$MigrateStartLimit
				value = 1
			}
		}
	}
}

#Injure the losing Holohana when two battle
holohana_injury_effect = {
	if = { 
		limit = { has_global_flag = holohana_$holohana$_injured }
		clr_global_flag = holohana_$holohana$_injured
		set_global_flag = holohana_$holohana$_dead
		holohana_migrate_effect = { holohana = $holohana$ }
	}
	if = {
		limit = {
			NOT = { has_global_flag = holohana_$holohana$_injured }
		}
		set_global_flag = holohana_$holohana$_injured
		6234 = {
			set_variable = {
				which = $holohana$FleeingTimer
				value = 6
			}
			set_province_flag = holohana_fleeing_$holohana$
		}
		if = {
			limit = { NOT = { has_global_flag = holohana_injury_month } }
			set_global_flag = holohana_injury_month
			REB = {
				country_event = {
					id = holohana_events.100
					days = 30
				}
			}
		}
	}
}
holohana_add_rumbling = {
	every_neighbor_province = {
		limit = {
			NOT = { has_province_modifier = holohana_rumbling }
			NOT = { has_holohana_province_modifier = yes }
		}
		add_permanent_province_modifier = {
			name = holohana_rumbling
			duration = -1
		}
	}
	every_neighbor_province = {
		limit = {
			has_province_modifier = holohana_rumbling 
		}
		set_province_flag = holohana_rumbling_flag_$holohana$
	}
	#Uncolonized
	every_empty_neighbor_province = {
		limit = {
			NOT = { has_province_modifier = holohana_rumbling }
			NOT = { has_holohana_province_modifier = yes }
		}
		add_permanent_province_modifier = {
			name = holohana_rumbling
			duration = -1
		}
	}
	every_empty_neighbor_province = {
		limit = {
			has_province_modifier = holohana_rumbling 
		}
		set_province_flag = holohana_rumbling_flag_$holohana$
	}
}
