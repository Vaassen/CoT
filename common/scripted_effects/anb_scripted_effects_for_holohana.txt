holohana_province_save_effect = {
	random_province = {
		limit = {
			has_province_modifier = holohana_modifier_$holohana$
		}
		save_event_target_as = holohana_target_$holohana$
		if = {
			limit = {
				NOT = { is_empty = yes }
			}
			owner = {
				save_event_target_as = holohana_province_owner_$holohana$
			}
		}
	}
}
holohana_spawn_effect = {
	P99 = {
		event_target:holohana_target_$holohana$ = {
			every_neighbor_province = {
				limit = {
					NOT = { has_province_modifier = holohana_rumbling }
				}
				add_permanent_province_modifier = {
					name = holohana_rumbling
					duration = -1
				}
			}
			every_empty_neighbor_province = {
				limit = {
					NOT = { has_province_modifier = holohana_rumbling }
				}
				add_permanent_province_modifier = {
					name = holohana_rumbling
					duration = -1
				}
			}
			if = {
				limit = {
					is_empty = yes
				}
				create_colony = 1000
				add_core = P99
				spawn_rebels = {
					size = 200
					[[leader_the_watcher] leader = "The Watcher"]
					[[leader_forest_wide_and_dim] leader = "Forest Wide and Dim"]
					type = holohana_rebels
				}
				decolonize = THIS
				remove_core = P99
			}
			else = {
				cede_province = P99
				spawn_rebels = {
					size = 200
					[[leader_the_watcher] leader = "The Watcher"]
					[[leader_forest_wide_and_dim] leader = "Forest Wide and Dim"]
					type = holohana_rebels
				}
				cede_province = event_target:holohana_province_owner_$holohana$
			}
		}
	}
}
