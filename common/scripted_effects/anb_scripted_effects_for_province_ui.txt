clear_all_aPUIE_flags = {
	clr_province_flag = aPUIE_parent_container_flag
	clr_province_flag = aPUIE_template_flag
}

#Use these scripted effects to properly manage the variable for a progress bar
change_aPUIE_var = {
	# To change the 'currency' use  'change_aPUIE_var = { name = <name> amount = 10 }' to increase it by 10.
	# You can also specify the name of a variable if you wish to calculate the quantity instead of using a static value
	[[amount] change_variable = { aPUIE_$name$_var = $amount$ } ]
	[[var_amount] change_variable = { which = aPUIE_$name$_var which = $var_amount$ } ]
	hidden_effect = {
		cap_aPUIE_var = { name = $name$ }
		if = {
			limit = { ai = no }
			update_aPUIE_var_bar = { name = $name$ }
		}
	}
	# Check for provinces in which the aPUIE is opened
	# Copy the framevariable to the province for the progress bar
	every_owned_province = {
		limit = {
			#This should be the flag specific to each aPUIE system
			has_province_flag = aPUIE_$name$_flag
		}
		set_variable = {
			which = aPUIE_$name$_framevar
			which = FROM
		}
	}
}
set_aPUIE_cap = {
	set_variable = {
		which = aPUIE_$name$_capacity
		value = $amount$
	}
}
change_aPUIE_cap = {
	set_variable = {
		which = aPUIE_$name$_capacity
		value = $amount$
	}
}
update_aPUIE_var_bar = { # This prevents the framevar going negative and then showing 100%
	if = { limit = { check_variable = { which = aPUIE_$name$_var value = 0 } }
		set_variable = { which = aPUIE_$name$_framevar which = aPUIE_$name$_var }
		divide_variable = { which = aPUIE_$name$_framevar which = aPUIE_$name$_capacity }
		multiply_variable = { which = aPUIE_$name$_framevar value = 100 }
	}
	else = { set_variable = { which = aPUIE_$name$_framevar value = 0 } }
}
cap_aPUIE_var = { # When the 'currency' goes above your capacity, or drops into negatives
	if = {
		limit = { NOT = { check_variable = { which = aPUIE_$name$_var value = 0 } } }
		set_variable = { which = aPUIE_$name$_var value = 0 }
	}
	else_if = {
		limit = { check_variable = { which = aPUIE_$name$_var which = aPUIE_$name$_capacity } }
		set_variable = { which = aPUIE_$name$_var which = aPUIE_$name$_capacity }
	}
}
