S98_hero_worship_conquest_effect = {
	if = {
		limit = {
			check_variable = {
				which = S98heroWorshipConquest
				which = S98heroWorshipTotal
			}
		}
		add_ruler_modifier = {
			name = S98_hero_worship_ruler_modifier_conquest
		}

		change_variable = {
			which = S98heroWorshipTotal
			value = 5
		}

		set_variable = {
			which = S98heroWorshipConquest
			value = 0
		}

		set_variable = {
			which = S98heroWorshipBuilding
			value = 0
		}

		set_variable = {
			which = S98heroWorshipDev
			value = 0
		}
	}
}
