khetarchs_curse_effect = {
	remove_country_modifier = U10_khetarch_curse_1
	remove_country_modifier = U10_khetarch_curse_2
	remove_country_modifier = U10_khetarch_curse_3
	remove_country_modifier = U10_khetarch_curse_4
	if = {
        limit = {check_variable = {
            which = khetarchs_curse_lvl

            check_variable = {
                which = khetarchs_curse_lvl
                value = 2
            }
        }}

        limit = {check_variable = {
            which = khetarchs_curse_lvl

        }

check_clergy_on_the_move = {
    if  = {
        limit = {
        NOT = {has_country_flag = U10_curtailed_church_flg}
        OR = {
            AND = {
            calc_true_if = {
                   has_estate_privilege = estate_church_purification_posts_along_the_sorrow
                   has_estate_privilege =  estate_church_potent_exorcists
                    has_estate_privilege = estate_church_ignorance_campaigns
                   has_estate_privilege = estate_church_charms_of_protection
                   amount = 2
            }
            estate_influence = {
                estate = estate_church
                influence = 75
                }
            }
            calc_true_if = {
                   has_estate_privilege = estate_church_purification_posts_along_the_sorrow
                   has_estate_privilege =  estate_church_potent_exorcists
                    has_estate_privilege = estate_church_ignorance_campaigns
                   has_estate_privilege = estate_church_charms_of_protection
                   amount = 4
                }
            }
        }
        country_event = { id = flavour_brrtekuh.23}
    }
}

        }
    }

            which = church_influence_lvl
            value = 4
        }
    }
    else_if  = {
        limit = {
        estate_influence = {
            estate = estate_church
            influence = 33
        }
    }
        set_variable = {
            which = church_influence_lvl
            value = 2
        }
    }
    else = {
        set_variable = {
            which = church_influence_lvl
            value = 1
        }
    }
    if  = {
        limit = {
        estate_influence = {
            estate = estate_nobles
            influence = 66
        }
    }
        set_variable = {
            which = nobles_influence_lvl
            value = 4
        }
    }
    else_if  = {
        limit = {
        estate_influence = {
            estate = estate_nobles

        }
    }

            value = 2
        }
    }

            which = nobles_influence_lvl
            value = 1
        }
    }
    if  = {
        limit = {
        estate_influence = {
            estate = estate_burghers
            influence = 66
        }

            which = burgher_influence_lvl

            value = 2
        }
    }

            which = burgher_influence_lvl
            value = 1
        }
    }
    set_variable = {
        which = combined_influence_lvl
        value = 1
    }
    change_variable = {
        which = combined_influence_lvl
        which = church_influence_lvl
    }

        limit = {
            OR = {
            estate_influence = {
                estate = estate_church

            check_variable = {
                which = church_influence_lvl

    }

                estate = estate_nobles
