custom_button = { #convert ruler to war wizard
    name = convert_ruler_to_war_wizard
    potential = {
		ruler_is_powerful_mage = yes
		is_lesser_in_union = no
		has_regency = no
		has_government_attribute = rulers_can_be_generals
	}
	trigger = {
		hidden_trigger = {
			OR = {
				is_monarch_leader = no
				AND = {
					has_ruler_flag = allow_war_wizard_after_exile #is_monarch_leader will always return true if a monarch leader is exiled and returned
					NOT = { has_ruler_modifier = allow_war_wizard_after_exile_delay } #define_ruler_to_general doesn't work after exile until 1 day has passed or the game has reloaded
				}
			}
			NOT = {
				has_country_flag = in_ruler_mage_general_confirmation_event
				has_ruler_modifier = bulwar_tluukt_injured
			}
		}
		if = { 
			limit = { has_government_attribute = rulers_cant_be_war_wizards } 
			custom_trigger_tooltip = {
				tooltip = cant_be_war_wizard_tt
				always = no
			}
		}
		if = {
			limit = {
				has_country_flag = magic_artificery_artificery_only
				NOT = { has_estate_privilege = artifice_invention_korashi_tuners }
			}
			is_not_in_artificery_mode = yes
		}
	}
	effect = {
		custom_tooltip = make_ruler_war_wizard_tt
		clr_ruler_flag = allow_war_wizard_after_exile
		set_country_flag = in_ruler_mage_general_confirmation_event
		hidden_effect = { country_event = { id = magic_system.14 } }
	}
	tooltip = make_ruler_war_wizard_desc
}

custom_button = {
	name = button_makeconsortgeneral
	potential = {
		has_consort = yes
		has_government_attribute = rulers_can_be_generals
		is_lesser_in_union = no
		OR = {
			has_female_consort = no
			has_global_modifier_value = {
				which = may_recruit_female_generals
				value = 1
			}
		}
		NOT = { has_consort_flag = is_general }
		OR = {
			consort_is_not_powerful_mage = yes
			NOT = {
				has_government_attribute = rulers_cant_be_war_wizards
				has_country_flag = magic_artificery_artificery_only
			}
		}
	}
	trigger = {
	}
	effect = {
		set_consort_flag = is_general
		if = { limit = { consort_is_powerful_mage = yes }
			hidden_effect = {
				export_to_variable = {
					which = target_num_generals
					value = trigger_value:num_of_generals
				}
				change_variable = { target_num_generals = 1 }
				trigger_switch = {
					on_trigger = consort_mil
					6 = { add_country_modifier = { name = war_wizard_mod_6 duration = 1 hidden = yes } }
					5 = { add_country_modifier = { name = war_wizard_mod_5 duration = 1 hidden = yes } }
					4 = { add_country_modifier = { name = war_wizard_mod_4 duration = 1 hidden = yes } }
					3 = { add_country_modifier = { name = war_wizard_mod_3 duration = 1 hidden = yes } }
					2 = { add_country_modifier = { name = war_wizard_mod_2 duration = 1 hidden = yes } }
					1 = { add_country_modifier = { name = war_wizard_mod_1 duration = 1 hidden = yes } }
					0 = { add_country_modifier = { name = war_wizard_mod_0 duration = 1 hidden = yes } }
				}
				country_event = { id = magic_system.15 } #removes the modifiers
			}
		}
	}
}
