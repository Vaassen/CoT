	name = make_autonomous_vassal

			}
		}

			is_subject_of_type = vassal

			limit = { FROM = { tag = H78 } } #seghdihr

		FROM = {
			add_trust = {
				who = ROOT
				value = 10

			}

				subject_type = autonomous_vassal

			}
		}
	}
}
