	potential = { always = yes }

			}

		else_if = { #Actually, We Are the Old World
			limit = { has_country_flag = achievements_page_2_opened }
			OR = {
				custom_trigger_tooltip = {
					tooltip = started_as_caamas_tag_tt

						is_or_was_tag = { tag = H02 } #Murdkather

						is_or_was_tag = { tag = H07 } #Bagcatir
					}
				}
				custom_trigger_tooltip = {
					tooltip = started_as_kheion_tag_tt
					OR = {
						is_or_was_tag = { tag = G46 } #Degakheion
						is_or_was_tag = { tag = G47 } #Oktikheion
						is_or_was_tag = { tag = G48 } #Kherka
						is_or_was_tag = { tag = G49 } #Lokemeion
						is_or_was_tag = { tag = G50 } #Ormam
						is_or_was_tag = { tag = G51 } #Arpedifer
						is_or_was_tag = { tag = G52 } #Ameion
						is_or_was_tag = { tag = G53 } #Empkeios
						is_or_was_tag = { tag = G54 } #Besolaki
						is_or_was_tag = { tag = G55 } #Eneion
						is_or_was_tag = { tag = G56 } #Deyeion
						is_or_was_tag = { tag = G57 } #Apikhoxi
						is_or_was_tag = { tag = G58 } #Keyolion
						is_or_was_tag = { tag = G59 } #Anisikheion
						is_or_was_tag = { tag = G62 } #Amgremos
					}
				}
			}
			europe = { has_discovered = ROOT }
			current_age = age_of_discovery
		}
	}
	effect = { update_achievements = yes }
	tooltip = achievement_r1_c1_req_button_tt
