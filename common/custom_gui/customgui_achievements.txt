

custom_button = {
	name = achievements_window_button
	potential = { }
	trigger = { }
	effect = {
		if = { 
			limit = {
				NOT = { has_country_flag = achievements_root_menu_opened } 
			}
			set_country_flag = achievements_root_menu_opened 
		}
		else = { clr_country_flag = achievements_root_menu_opened }
	}
}

custom_window = {
	name = achievements_root_window
	potential = {
		hidden_trigger = {
			has_country_flag = achievements_root_menu_opened
		}
	}
}

custom_button = {
	name = achievement_window_button_close
	effect = {
		clr_country_flag = achievements_root_menu_opened
	}
}
