custom_button = {
    name = siege_magic_button
    potential = {
		fort_level = 1
	}
	trigger = {
		hidden_trigger = {
			OR = {
				AND = {
					sieged_by = FROM
					unit_has_leader = yes
					FROM = {
						has_leader_with = {
							shock = 7
						}
						NOT = {
							has_country_flag = banned_magic
							has_ruler_flag = magic_siege_menu
							has_ruler_flag = magic_just_casted_spell
							has_ruler_flag = magic_menu_opened
							has_ruler_modifier = magic_siege_cooldown
						}
					}
				}
				AND = {
					owned_by = FROM
					has_siege = yes
				}
			}
		}
	}
	effect = {
		hidden_effect = {
			set_province_flag = spell_target_province
			FROM = { country_event = { id = magic_siege.1 } open_single_menu = yes }
		}
	}
	tooltip = siege_magic_button_tt
}

custom_button = {
	name = siege_magic_button_1
	potential = { always = yes }
	trigger = {
		has_siege = yes
		NOT = { sieged_by = FROM }
		FROM = {
			is_overall_abjuration_level_1_minimum = yes
			has_mana_to_cast_spell = { school = abjuration level = 1 id = 0 }
		}
	}
	effect = {
		custom_tooltip = abjuration_1_0_spell
		change_siege = -1
		if = {
			limit = {
				# owned_by = FROM
				FROM = { has_country_flag = magical_fortress_2 }
			}
			add_province_modifier = { name = abjuration_1_0_mod_buffed duration = 3650 }
		}
		else = { add_province_modifier = { name = abjuration_1_0_mod duration = 3650 } }
	}
	frame = {
		number = 1
		trigger = { always = yes }
	}
	tooltip = abjuration_1_0_spell
}
