custom_diplo_ui_feiten_ptm = {
	potential = { always = yes }
	trigger = { always = yes }
	on_activation = {
		event_target:feiten_or_was_feiten_target = {
			# custom_tooltip = feitenConstructSkypostInfoShortTT
			add_treasury = -1000
			add_dip_power = -50
		}
		owner = {
			home_trade_node_effect_scope = {
				random_trade_node_member_province = {
					limit = {
						owned_by = ROOT #owner of ROOT, will use event target if this needs to be more conventional
						province_has_center_of_trade_of_level = 1
					}
					add_province_triggered_modifier = feiten_skyport_handler_mod
				}
			}
		}

		remove_province_triggered_modifier = custom_diplo_ui_feiten_ptm
	}
}

custom_diplo_ui_vampire_spread_action_ptm = {
	potential = { always = yes }
	trigger = { always = yes }
	# hidden = yes
	on_activation = {
		owner = {
			save_event_target_as = vampire_origin
			event_target:vampire_origin = { #scope into it so that PREV works
				random_known_country = {
					limit = { has_country_flag = spread_vamps_@PREV }
					save_event_target_as = vampire_spread_action_target
				}
				if = {
					limit = { has_saved_event_target = vampire_spread_action_target }
					event_target:vampire_spread_action_target = {
						clr_country_flag = spread_vamps_@PREV
						country_event = { id = vampires_estate_events.11 days = 7 }
					}
				}
			}
		}
		remove_province_triggered_modifier = custom_diplo_ui_vampire_spread_action_ptm
	}
}
