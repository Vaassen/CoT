#Xhazobkult Krah Rights
estate_nomadic_krah_rights = {
	icon = privilege_dev_cost_desert
	max_absolutism = 0
	influence = 0.1
	loyalty = 0.1
	is_valid = {
		religion = xhazobkult
	}
	can_select = {
		any_owned_province = {
			AND = {
				NOT = { has_owner_accepted_culture = yes }
				NOT = { culture_group = gnollish }
			}
		}
	}
	on_granted = {
		custom_tooltip = estate_nomadic_krah_rights_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					AND = {
						NOT = { has_owner_accepted_culture = yes }
						NOT = { culture_group = gnollish }
					}
				}
				add_province_modifier = {
					name = estate_nomadic_krah_rights_mod # +1 local unrest, +33% local manpower, +100 manpower, +20% local sailors
					duration = -1
				}
			}
		}
	}
	on_revoked = {
		custom_tooltip = estate_burghers_enslaving_rights_rt
		hidden_effect = {
			every_owned_province = {
				limit = {
					AND = {
						NOT = { has_owner_accepted_culture = yes }
						NOT = { culture_group = gnollish }
					}
				}
				remove_province_modifier = estate_nomadic_krah_rights_mod # +1 local unrest, +33% local manpower, +100 manpower, +20% local sailors
			}
		}
	}
	penalties = {
		global_unrest = 2
		global_trade_goods_size_modifier = -0.1
	}
	benefits = {
		no_religion_penalty = yes
	}
	ai_will_do = {
		factor = 1
	}
}
