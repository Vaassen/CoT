
darkmagicks = {
	potential = {
		religion = rinta_seekers
	}

	
	
	can_start = {
		custom_trigger_tooltip = {
			tooltip = darkmagicks_start_tt
			has_global_flag = darkmagicks
		}
		
		religion = rinta_seekers
		has_any_disaster = no
	}
	
	can_stop = {
		always = no
	}

	progress = {
		modifier = {
			factor = variable:rinta_factor
			custom_trigger_tooltip = {
				tooltip = darkmagicks_progres_tt
				always = yes
			}
		}
	}
	
	can_end = {
		custom_trigger_tooltip = {
			tooltip = darkmagicks_end_tt
			NOT = { has_global_flag = darkmagicks }
		}
	}
	
	on_start = hoardcurse.5
	on_end = hoardcurse.25
}
