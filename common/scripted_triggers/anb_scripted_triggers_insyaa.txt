
tag_can_use_mechanim_ward_conduit = {
	# Mechanim formables
	OR = {
		tag = M22 # Kiretua
		tag = M23 # Rangitema
		tag = M24 # Anahira
		tag = M25 # Pa-Aroha
		tag = M26 # Rakauma
	}
}

province_can_get_mechanim_ward_conduit = {
	OR = {
		AND = {
			owner = { tag = M22 } # Kiretua
			OR = {
				area = titanoflora_lakelands_5_area
				area = titanoflora_lakelands_6_area
			}
		}
		AND = {
			owner = { tag = M23 } # Rangitema
			OR = {
				area = titanoflora_lakelands_10_area
				area = titanoflora_lakelands_11_area
				area = titanoflora_lakelands_12_area
				area = titanoflora_lakelands_13_area
			}
		}
		AND = {
			owner = { tag = M24 } # Anahira
			OR = {
				area = titanoflora_riverlands_11_area
				area = titanoflora_riverlands_13_area
				province_id = 3496
				province_id = 3331
				province_id = 3495
			}
		}
		AND = {
			owner = { tag = M25 } # Pa-Aroha
			OR = {
				area = titanoflora_riverlands_16_area
				area = titanoflora_riverlands_17_area
				province_id = 3183
				province_id = 8181
				province_id = 3322
				province_id = 3323
			}
		}
		AND = {
			owner = { tag = M26 } # Rakauma
			OR = {
				area = turuputai_area
				area = manuata_area
				province_id = 3159
				province_id = 3560
			}
		}
	}
}

utubodi_unification_can_collect_resource = {
	region = great_plateau_region
	if = {
		limit = { trade_goods = grain }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorGrainVar
					value = 8
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = livestock }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorLivestockVar
					value = 1
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = wool }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorWoolVar
					value = 3
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = tropical_wood }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorExWoodVar
					value = 2
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = cloth }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorClothVar
					value = 1
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = iron }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorIronVar
					value = 2
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = incense }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorIncenseVar
					value = 2
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = fur }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorFurVar
					value = 1
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = copper }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorCopperVar
					value = 1
				}
			}
		}
	}
	else_if = {
		limit = { trade_goods = damestear }
		FROM = {
			NOT = {
				check_variable = {
					which = utubodiKakunorDamestearVar
					value = 1
				}
			}
		}
	}
}

utubodi_unification_has_collectable_resource = {
	OR = {
		trade_goods = grain
		trade_goods = livestock
		trade_goods = wool
		trade_goods = tropical_wood
		trade_goods = cloth
		trade_goods = iron
		trade_goods = incense
		trade_goods = fur
		trade_goods = copper
		trade_goods = damestear
	}
}
