khatalashya_can_colonize_this_planet = {
	if = {
		limit = { #Romutha
			province_id = 5921
		}
		if = {
			limit = { has_global_flag = khatalashya_romutha_2 }
			owner = {
				treasury = 500
			}
			has_colonist = yes
			num_of_units_in_province = {
				who = ROOT
				amount = 10
			}
		}
		else = {
			owner = {
				adm_power = 50
				dip_power = 50
				mil_power = 50
				treasury = 250
			}
		}
	}
	else_if = {
		limit = { #Agradent
			province_id = 5763
		}
		if = {
			limit = { has_global_flag = khatalashya_agradent_2 }
			owner = {
				num_of_owned_provinces_with = {
					value = 5
					base_production = 8
					base_manpower = 8
					trade_goods = grain
				}
			}
			num_of_units_in_province = {
				who = ROOT
				amount = 10
			}
		}
		else = {
			owner = {
				any_owned_province = {
					base_production = 15
					trade_goods = precursor_relics
				}
			}
		}
	}
	else_if = {
		limit = { #Zulkanez
			province_id = 5863
		}
		if = {
			limit = { has_global_flag = khatalashya_zulkanez_2 }
			owner = {
				num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
					value = 3
					OR = {
						trade_goods = wool
						trade_goods = fur
					}
					base_production = 10
					has_production_building_trigger = yes
				}
			}
			owner = {
				has_estate_privilege = estate_nobility_shanyza_mutation
			}
		}
		else = {
			has_great_project = {
				type = kuiika_gamyi
				tier = 3
			}
			base_manpower = 20
		}
	}
	else_if = {
		limit = { #Shagatan
			province_id = 5585
		}
		if = {
			limit = { has_global_flag = khatalashya_shagatan_2 }
			owner = {
				num_of_marines = 30
				num_of_owned_provinces_with = {
					value = 5
					has_building = wharf
				}
			}
			5586 = {
				transports_in_province = 30
			}
		}
		else = {
			base_manpower = 12
			owner = {
				have_cast_aid_construction = yes
			}
		}
	}
	else_if = {
		limit = { #Teinglios
			province_id = 5971
		}
		if = {
			limit = { has_global_flag = khatalashya_teinglios_2 }
			OR = {
				has_building = furnace
				treasury = 10000
			}
			5971 = {
				has_great_project = {
					type = khaga_yatsa
					tier = 3
				}
			}
			owner = {
				has_estate_privilege = estate_burghers_gasabi_mutation
			}
			has_supply_depot = ROOT
			num_of_units_in_province = {
				who = ROOT
				amount = 10
			}
		}
		else = {
			province_has_current_tech_fort_trigger = yes
			has_manufactory_trigger = yes
			num_of_times_expanded_infrastructure = 2
			owner = {
				has_estate_privilege = estate_burghers_gasabi_mutation
			}
		}
	}
	else_if = {
		limit = { #Zangea
			province_id = 6032
		}
		if = {
			limit = { has_global_flag = khatalashya_zangea_2 }
			transports_in_province = 100
			infantry_in_province = 100
		}
		else = {
			OR = {
				has_great_project = {
					type = arskitse
					tier = 3
				}
				AND = {
					transports_in_province = 50
					infantry_in_province = 50
					owner = {
						num_of_owned_provinces_with = {
							value = 5
							trade_goods = naval_supplies
							base_production = 8
						}
					}
				}
			}
		}
	}
	else_if = {
		limit = { #Atamumba
			province_id = 6685
		}
		if = {
			limit = { has_global_flag = khatalashya_atamumba_2 }
			custom_tooltip = khatalashya_passed_disaster_hive
			NOT = { has_country_flag = the_hivemind_can_start }
		}
		else = {
			OR = {
				has_building = fort_18th
				AND = {
					province_has_current_tech_fort_trigger = yes
					has_building = ramparts
				}
			}
		}
	}
	else_if = {
		limit = { #Rakh
			province_id = 5650
		}
		if = {
			limit = { has_global_flag = khatalashya_rakh_2 }
			development = 50
			num_of_units_in_province = {
				who = ROOT
				amount = 10
			}
			custom_tooltip = khatalashya_passed_disaster_hive
			NOT = { has_country_flag = the_hivemind_can_start }
		}
		else = {
			adm_power = 50
			dip_power = 50
			mil_power = 50
			treasury = 1000
		}
	}
	else_if = {
		limit = { #SPACE
			province_id = 5894
		}
		has_building = manufactory
		local_autonomy = 0.1
		OR = {
			has_building = training_fields
			base_manpower = 15
		}
	}
	else = {
		always = no
	}
}
