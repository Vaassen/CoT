cot_winter_affected_region = {
    OR = { 
        superregion = western_cannor_superregion
        superregion = gerudia_superregion
        superregion = escann_superregion 
        superregion = deepwoods_superregion 
    }
}

cot_winter_has_food = {
    OR = { 
        trade_goods = fish 
        trade_goods = grain 
        trade_goods = livestock 
    }
}

cot_winter_has_damestear = {
	OR = {
		any_owned_province = { trade_goods = damestear }
		overlord = { any_owned_province = { trade_goods = damestear } }
		any_subject_country = { any_owned_province = { trade_goods = damestear } }
	}
}
