transmutative_path_change_trade_good_trigger = {
	NOT = {
		trade_goods = damestear
		trade_goods = gold
		trade_goods = precursor_relics
	}
	OR = {
		AND = { #Naamjyut
			has_terrain = jungle
			NOT = { trade_goods = tea }
			owner = { culture = naamjyut }
		}
		AND = { #Khom
			trade_goods = cloth
			owner = { culture = khom }
		}
		AND = { #Phonan
			OR = {
				has_terrain = hills
				has_terrain = highlands
				has_terrain = mountain
			}
			NOT = { trade_goods = copper }
			owner = { culture = phonan }
		}
		AND = { #Hinphat
			trade_goods = livestock
			owner = { culture = hinphat }
		}
		AND = { #Binhrung
			trade_goods = fish
			owner = { culture = binhrung }
		}
		AND = { #Bokai
			trade_goods = wool
			owner = { culture = bokai }
		}
		AND = { #Azkare
			trade_goods = cotton
			owner = {
				OR = {
					tag = Y86
					was_tag = Y86
				}
			}
		}
		AND = {  #Thirabnir
			OR = {
				has_terrain = drylands
				has_terrain = desert
			}
			NOT = { trade_goods = salt }
			owner = {
				OR = {
					tag = Y59
					was_tag = Y59
				}
			}
		}
		AND = { #Khao Elnak
			trade_goods = grain
			owner = {
				OR = {
					tag = Y62
					was_tag = Y62
				}
			}
		}
		AND = { #Antmvura
			has_terrain = jungle
			NOT = { trade_goods = coffee }
			owner = {
				culture = antmvura
			}
		}
		#AND = { #Lo Ngeon - Currently unused due to being OP? but thought i just leave it here
		#	owner = {
		#		OR = {
		#			tag = Y60
		#			was_tag = Y60
		#		}
		#	}
		#	trade_goods = salt
		#}
	}
}

transmutative_path_infuse_province_trigger = {
	AND = {
		OR = {
			AND = { #Harimari
				owner = {
					culture_group = harimari
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				province_has_center_of_trade_of_level = 1
			}
			AND = { #Naamjyut
				owner = {
					culture = naamjyut
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = tea
			}
			AND = { #Khom
				owner = {
					culture = khom
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = silk
			}
			AND = { #Phonan
				owner = {
					culture = phonan
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = copper
			}
			AND = { #Hinphat/Bokai
				owner = {
					OR = {
						culture = hinphat
						culture = bokai
					}
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = grain
			}
			AND = { #Binhrung
				owner = {
					culture = binhrung
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = gems
			}
			AND = { #Xiaken
				owner = {
					OR = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				has_terrain = hills
			}
			AND = { #Azkare
				owner = {
					tag = Y86
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = cloth
			}
			AND = { #Thirabnir
				owner = {
					OR = {
						tag = Y59
						was_tag = Y59
					}
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = salt
			}
			AND = { #Lo Ngoen
				owner = {
					OR = {
						tag = Y60
						was_tag = Y60
					}
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = gold
			}
			AND = { #Khao Elnak
				owner = {
					OR = {
						tag = Y62
						was_tag = Y62
					}
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = livestock
			}
			AND = { #Antmvura
				owner = {
					culture = antmvura
					NOT = {
						has_reform = wulin
						has_reform = xiaken
						has_reform = one_xia
						has_reform = indep_xiaken
					}
				}
				trade_goods = coffee
			}
			AND = { #Generic
				is_capital = yes
				owner = {
					NOT = { culture = antmvura }
					NOT = { culture_group = kai }
					NOT = { culture_group = harimari }
					NOT = { culture = goldscale_kobold }
					NOT = { culture = naamjyut }
					NOT = { has_reform = wulin }
					NOT = { has_reform = xiaken }
					NOT = { has_reform = one_xia }
					NOT = { has_reform = indep_xiaken }
				}
			}
		}
	}
}
