voltoj_can_afford_upgrade_to_medium = {
	has_voltoj_academy = yes
	has_major_voltoj_academy = no
	owner = { has_enacted_parliament_action = { parliament_action = overclan_expand_voltoj_recruitment_act } }
	if = {
		limit = { province_group = voltoj_hold_city_province }
		development = 50
	}
	else = {
		development = 30
	}
}
voltoj_can_afford_upgrade_to_large = {

	if = {

	trigger_switch = {

		521 = {
			521 = {
				NOT = { has_province_modifier = F78_grand_officer_university }
			}
		}
		526 = {
			526 = {
				owner = { has_country_flag = F78_inscription_buildings_unlocked }
				NOT = { has_province_modifier = F78_hammer_head_co }
			}
		}
		510 = {
			510 = {
				owner = { has_issue_unlocked = { issue = overclan_sponsor_dwarven_settlement } }
				NOT = { has_province_modifier = F78_bal_ouord_modernization }
			}
		}
		536 = {

				NOT = { has_province_modifier = F78_greysheep_company }

	}

}

overclan_voltoj_academy_growth_tt = { #province event scope #region
	if = {
		limit = { has_large_voltoj_academy = yes }
		custom_tooltip = voltoj_academy_largest_tt
	}
	else_if = {
		limit = { has_major_voltoj_academy = no } #small
		if = {
			limit = {
				owner = {
					calc_true_if = {
						all_owned_province = {
							has_medium_voltoj_academy = yes
							province_group = voltoj_city_$region$_province
						}
						amount = 2
					}
				}
			}
			custom_tooltip = voltoj_academy_medium_limit_tt
		}
		else_if = {
			limit = { voltoj_can_afford_upgrade_to_medium = no }
			if = {
				limit = { province_group = voltoj_hold_city_province }
				custom_tooltip = voltoj_academy_medium_hold_dev_tt
			}
			else = {
				custom_tooltip = voltoj_academy_medium_dev_tt
			}
		}
	}
	else_if = {
		limit = { has_medium_voltoj_academy = yes }
		if = {
			limit = {
				owner = {
					any_owned_province = {
						province_group = voltoj_city_$region$_province
						has_large_voltoj_academy = yes
					}
				}
			}
			custom_tooltip = voltoj_academy_large_limit_tt
		}
		else_if = {
			limit = { voltoj_can_afford_upgrade_to_large = no }
			if = {
				limit = { province_group = voltoj_hold_city_province }
				custom_tooltip = voltoj_academy_large_hold_dev_tt
			}
			else = {
				custom_tooltip = voltoj_academy_large_dev_tt
			}
		}
	}
}
