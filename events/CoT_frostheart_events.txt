namespace = cot_frostheart

#Gerudia debuff
country_event = {
	id = cot_frostheart.0
	title = cot_frostheart.0.t
	desc = cot_frostheart.0.d
	picture = BIG_BOOK_eventPicture

	fire_only_once = yes
	is_triggered_only = yes
	hidden = yes

	trigger = { owns = 701 }

	immediate = {
		every_province = {
			if = {
				limit = { superregion = gerudia_superregion }
				add_province_modifier = {
					name = cot_warth_of_elkaesal
					duration = -1
				}
			}
		}
	}

	option = {
		name = flavor_castanor_go_back
		ai_chance = { factor = 1 }
	}
}

#Refresh malus reductions
country_event = {
	id = cot_frostheart.1
	title = cot_frostheart.1.t
	desc = cot_frostheart.1.d
	picture = BIG_BOOK_eventPicture

	hidden = yes

	trigger = {
		OR = {
			AND = {
				has_country_modifier = cot_test_artifact
				any_owned_province = {
					superregion = gerudia_superregion
					NOT = { has_province_modifier = cot_test_artifact_malus_reduction }
				}
			}
			AND = {
				NOT = { has_country_modifier = cot_test_artifact }
				any_owned_province = {
					superregion = gerudia_superregion
					has_province_modifier = cot_test_artifact_malus_reduction
				}
			}
		}
	}

	option = {
		name = cot_frostheart.1.a
		every_owned_province = {
			if = {
				limit = { superregion = gerudia_superregion }
				remove_province_modifier = cot_test_artifact_malus_reduction
			}
		}
		every_owned_province = {
			if = {
				limit = { 
					superregion = gerudia_superregion
					owner = { has_country_modifier = cot_test_artifact }
				}
				add_province_modifier = {
					name = cot_test_artifact_malus_reduction
					duration = -1
				}
			}
		}
	}
}
