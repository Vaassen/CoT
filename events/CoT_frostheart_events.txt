# Upgrade blessing when new age
country_event = {
	id = cot_frostheart.1
	title = cot_frostheart.1.t
	desc = cot_frostheart.1.d
	picture = BIG_BOOK_eventPicture

	hidden = yes

	trigger = {
		OR = {
			AND = {
				has_country_modifier = cot_blessing_of_elkaesal_1
				current_age = age_of_reformation
			}
			AND = {
				has_country_modifier = cot_blessing_of_elkaesal_2
				current_age = age_of_absolutism
			}
			AND = {
				has_country_modifier = cot_blessing_of_elkaesal_3
				current_age = age_of_revolutions
			}
		}
	}

	immediate = {
		remove_country_modifier = cot_blessing_of_elkaesal_1
		remove_country_modifier = cot_blessing_of_elkaesal_2
		remove_country_modifier = cot_blessing_of_elkaesal_3

		if = {
			limit = { current_age = age_of_reformation }
			remove_country_modifier = cot_blessing_of_elkaesal_1
			add_country_modifier = { name = cot_blessing_of_elkaesal_2 duration = -1 }
		}
		if = {
			limit = { current_age = age_of_absolutism }
			remove_country_modifier = cot_blessing_of_elkaesal_2
			add_country_modifier = { name = cot_blessing_of_elkaesal_3 duration = -1 }
		}
		if = {
			limit = { current_age = age_of_revolutions }
			remove_country_modifier = cot_blessing_of_elkaesal_3
			add_country_modifier = { name = cot_blessing_of_elkaesal_4 duration = -1 }
		}
	}

	option = {
		name = flavor_castanor_go_back
		ai_chance = { factor = 1 }
	}
}


# Remove blessing when losing an artifact
country_event = {
	id = cot_frostheart.2
	title = cot_frostheart.2.t
	desc = cot_frostheart.2.d
	picture = BIG_BOOK_eventPicture

	hidden = yes

	trigger = {
		OR = {
			has_country_modifier = cot_blessing_of_elkaesal_1
			has_country_modifier = cot_blessing_of_elkaesal_2
			has_country_modifier = cot_blessing_of_elkaesal_3
			has_country_modifier = cot_blessing_of_elkaesal_4
		}
		NOT = {
			AND = {
				has_country_modifier = cot_artifact_1
				has_country_modifier = cot_artifact_2
				has_country_modifier = cot_artifact_3
				has_country_modifier = cot_artifact_4
				has_country_modifier = cot_artifact_5
				has_country_modifier = cot_artifact_6
				has_country_modifier = cot_artifact_7
			}
		}
	}

	immediate = {
		remove_country_modifier = cot_blessing_of_elkaesal_1
		remove_country_modifier = cot_blessing_of_elkaesal_2
		remove_country_modifier = cot_blessing_of_elkaesal_3
		remove_country_modifier = cot_blessing_of_elkaesal_4
	}

	option = {
		name = flavor_castanor_go_back
		ai_chance = { factor = 1 }
	}
}
