# Phase One event handler because scripted effects are jank and hate "else"s
province_event = {
	id = sedentarization.1
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	option = {
		# First Contact events come first
		if = {
			limit = {
				NOT = { has_province_flag = sed_contact_complete }
			}
			random_list = {
				1 = {
					province_event = {
						id = sedentarization.100
						days = 1
						random = 359
					}
				}
				4 = {
					trigger = { sedentarization_province_valid_for_centaur_friendly_options = yes }
					province_event = {
						id = sedentarization.101
						days = 1
						random = 359
					}
				}
			}
		}
		# Raid Events forever-after (until the player establishes guards)
		else = {
			random_list = {
				# 73.7% chance of happening each roll
				737 = {
					province_event = {
						id = sedentarization.120
						days = 1
						random = 359
					}
				}
				# ~60% chance of happening within 3 rolls
				263 = {
					province_event = {
						id = sedentarization.121
						days = 1
						random = 359
					}
				}
			}
		}
	}
}

# Event queue handler. Part of the system that takes care of invalidated choices and edge-case simultaneous outpost completion
province_event = {
	id = sedentarization.20
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	option = {
		# Always go through the queued events for the current province before scoping to random other provinces
		if = {
			limit = { has_province_flag = sed_settler_policy_event_queued }
			province_event = { id = sedentarization.300 }
		}
		else_if = {
			limit = { has_province_flag = sed_outpost_completion_event_queued }
			province_event = { id = sedentarization.310 }
		}
		# Finally, look for other provinces with either type of queued event
		else = {
			owner = {
				random_owned_province = {
					limit = {
						OR = {
							has_province_flag = sed_outpost_type_event_queued
							has_province_flag = sed_settler_policy_event_queued
							has_province_flag = sed_outpost_completion_event_queued
						}
					}
					# Always fire the earlier-phase events before later ones (outpost type before settler policy)
					if = {
						limit = { has_province_flag = sed_outpost_type_event_queued }
						province_event = { id = sedentarization.200 }
					}
					else_if = {
						limit = { has_province_flag = sed_settler_policy_event_queued }
						province_event = { id = sedentarization.300 }
					}
					else = {
						province_event = { id = sedentarization.310 }
					}
				}
			}
		}
	}
}

province_event = {
	id = sedentarization.200
	title = sedentarization.200.t
	desc = sedentarization.200.d
	picture = none

	is_triggered_only = yes

	option = {
		name = sedentarization.200.gozabhar
		trigger = { sedentarization_province_valid_for_centaur_friendly_options = yes }
		if = {
			limit = { has_province_flag = sed_friendly_warning }
			custom_tooltip = "With a \"friendly\" warning from the centuars, this is the only option to prevent raids."
		}
		custom_tooltip = "Outpost gains 1 base tax on completion, but our Phase Three settler policy will be locked to Centaur Tribute, giving the following effects:"
		tooltip = {
			add_province_modifier = {
				name = sed_centaur_tributary_system
				duration = -1
			}
		}
		custom_tooltip = "Will invalidate if purging or expelling centaurs."
		set_province_flag = sed_outpost_type_gozabhar
	}
	option = {
		name = sedentarization.200.settlement
		sedentarization_neglecting_friendly_warning_effect = yes
		custom_tooltip = "Outpost gains 1 base tax on completion."
		set_province_flag = sed_outpost_type_settlement
		owner = {
			add_dip_power = -25
			sedentarization_phase_two_settlement_culture_effect = yes
		}
	}
	option = {
		name = sedentarization.200.nomadic_base
		highlight = yes
		# The triggers MUST match the ones in "sedentarization_phase_two_nomadic_base_culture_effect"
		trigger = { sedentarization_phase_two_nomadic_control_is_valid = yes }
		sedentarization_neglecting_friendly_warning_effect = yes
		custom_tooltip = "Outpost gains 1 base manpower on completion."
		set_province_flag = sed_outpost_type_nomadic_base
		owner = {
			add_mil_power = -25
			sedentarization_phase_two_nomadic_base_culture_effect = yes
		}
	}
	option = {
		name = sedentarization.200.kelaktar
		highlight = yes
		trigger = {
			owner = { primary_culture = fathide_ogre }
		}
		sedentarization_neglecting_friendly_warning_effect = yes
		custom_tooltip = "Outpost gains 1 base manpower and 1 production on completion."
		set_province_flag = sed_outpost_type_kelaktar
		owner = { add_mil_power = -25 }
		change_culture = fathide_ogre
		change_religion = ROOT
	}
	
	after = {
		set_province_flag = sed_outpost_type_selected
		# Check that the AI didn't pick the first option when it's invalid
		if = {
			limit = {
				has_province_flag = sed_outpost_type_gozabhar
				sedentarization_province_valid_for_centaur_friendly_options = no
			}
			clr_province_flag = sed_outpost_type_gozabhar
			random_list = {
				1 = {
					set_province_flag = sed_outpost_type_settlement
					owner = {
						add_dip_power = -25
						sedentarization_phase_two_settlement_culture_effect = yes
					}
				}
				1 = {
					trigger = { sedentarization_phase_two_nomadic_control_is_valid = yes }
					set_province_flag = sed_outpost_type_nomadic_base
					owner = {
						add_mil_power = -25
						sedentarization_phase_two_nomadic_base_culture_effect = yes
					}
				}
				1 = {
					trigger = {
						owner = { primary_culture = fathide_ogre }
					}
					set_province_flag = sed_outpost_type_kelaktar
					owner = { add_mil_power = -25 }
					change_culture = fathide_ogre
					change_religion = ROOT
				}
			}
		}
		# Fire other queued events, part of the system that takes care of invalidated choices
		clr_province_flag = sed_outpost_type_event_queued
		province_event = { id = sedentarization.20 }
	}
}

# Outpost Completion
province_event = {
	id = sedentarization.310
	title = sedentarization.310.t
	desc = sedentarization.310.d
	picture = none

	is_triggered_only = yes

	option = {
		name = sedentarization.310.a
		# Bonus from outpost type
		if = {
			limit = { has_province_flag = sed_outpost_type_settlement }
			add_base_tax = 1
		}
		else_if = {
			limit = { has_province_flag = sed_outpost_type_nomadic_base }
			add_base_manpower = 1
		}
		else_if = {
			limit = { has_province_flag = sed_outpost_type_kelaktar }
			add_base_production = 1
			add_base_manpower = 1
		}
		# Completion effect from settler policy
		hidden_effect = { save_event_target_as = sed_outpost }
		if = {
			limit = { has_province_flag = sed_settler_policy_centaur_tribute }
			sedentarization_outpost_completion_centaur_tribute_effect = yes
			custom_tooltip = "Within the next three months, the rest of the area will gain:"
			tooltip = {
				sedentarization_outpost_completion_centaur_tribute_effect = { culture_source = event_target:sed_outpost }
				set_local_autonomy = 50
			}
			sedentarization_propagate_completion_effect_to_area = { event_number = 311 }
		}
		else_if = {
			limit = { has_province_flag = sed_settler_policy_nomadic_control }
			sedentarization_outpost_completion_nomadic_control_effect = yes
			custom_tooltip = "Within the next three months, the rest of the area will gain:"
			tooltip = {
				sedentarization_outpost_completion_centaur_tribute_effect = { culture_source = event_target:sed_outpost }
				set_local_autonomy = 50
			}
			sedentarization_propagate_completion_effect_to_area = { event_number = 312 }
		}

			limit = { has_province_flag = sed_settler_policy_heavy_settlement }

				set_local_autonomy = 50
			}
			sedentarization_propagate_completion_effect_to_area = { event_number = 313 }
		}
	}
	
	after = {
