# Phase One event handler because scripted effects are jank and hate "else"s
province_event = {
	id = sedentarization.1
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	option = {
		# First Contact events come first
		if = {
			limit = {
				NOT = { has_province_flag = sed_contact_complete }
			}
			random_list = {
				1 = {
					province_event = { id = sedentarization.100 }
				}
				4 = {
					trigger = { sedentarization_province_valid_for_centaur_friendly_options = yes }
					province_event = { id = sedentarization.101 }
				}
			}
		}
		# Raid Events forever-after (until the player establishes guards)
		else_if = {
			limit = { sedentarization_raid_events_can_fire = yes }
			random_list = {
				# 73.7% chance of happening each roll
				737 = {
					province_event = { id = sedentarization.120 }
				}
				# ~60% chance of happening within 3 rolls
				263 = {
					province_event = { id = sedentarization.121 }
				}
			}
		}
	}
}

# Phase Two event handler
province_event = {
	id = sedentarization.2
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	option = {
		# There's currently only 1 random Phase Two/Three event, so no need for complex effects and triggers yet
		if = {
			limit = {
				OR = {
					trade_goods = wool
					trade_goods = livestock
				}
			}
			save_event_target_as = sed_trade_goods_upgrade_target
		}
		else = {
			random_area_province = {
				limit = {
					owned_by = event_target:sed_outpost_owner
					OR = {
						trade_goods = wool
						trade_goods = livestock
					}
				}
				save_event_target_as = sed_trade_goods_upgrade_target
			}
		}
		if = {
			limit = { has_saved_event_target = sed_trade_goods_upgrade_target }
			province_event = {
				id = sedentarization.210
				days = 1
				random = 359
			}
		}
	}
}

# Event queue handler. Part of the system that takes care of invalidated choices and edge-case simultaneous outpost completion.
# Can't be a scripted effect because of the "province_event" effects in elses.
# TODO: hmm, this queuing system was originally coded to be resilient to a single province having both its Phase Two and Three flags invalidated then letting it continue from Phase Three, but now the system resets to the earliest invalidated Phase. This is possibly overbuilt now; maybe in "sedentarization_reset_province_outpost".
province_event = {
	id = sedentarization.15
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	option = {
		# First Contact can only get queued if moving capital onto an outpost that hasn't had First Contact yet,
		# i.e. it can ONLY queue onto a Phase One outpost, meaning it won't need to check itself for Phase Two nor Three queued events,
		# Always go through the queued events for the current province before scoping to random other provinces
		if = {
			limit = { has_province_flag = sed_settler_policy_event_queued }
			province_event = { id = sedentarization.300 }
		}
		else_if = {
			limit = { has_province_flag = sed_outpost_completion_event_queued }
			province_event = { id = sedentarization.310 }
		}
		# Finally, look for other provinces with either type of queued event
		else = {
			owner = {
				random_owned_province = {
					limit = {
						OR = {
							has_province_flag = sed_sudden_raid_event_queued
							has_province_flag = sed_outpost_type_event_queued
							has_province_flag = sed_settler_policy_event_queued
							has_province_flag = sed_outpost_completion_event_queued
						}
					}
					# Always fire the earlier-phase events before later ones (outpost type before settler policy)
					if = {	# Look for the capital on an outpost with a queued First Contact event
						limit = { has_province_flag = sed_sudden_raid_event_queued }
						province_event = { id = sedentarization.100 }
					}
					else_if = {
						limit = { has_province_flag = sed_outpost_type_event_queued }
						province_event = { id = sedentarization.200 }
					}
					else_if = {
						limit = { has_province_flag = sed_settler_policy_event_queued }
						province_event = { id = sedentarization.300 }
					}
					else = {
						province_event = { id = sedentarization.310 }
					}
				}
			}
		}
	}
}

	hidden = yes

	option = {

			}

		}
		else_if = {
			limit = {
				has_province_modifier = sed_outpost_phase_ii
				NOT = { local_autonomy = 65.01 }
			}
			sedentarization_outpost_try_phase_change = { current_phase_is_two = yes }
		}
		else_if = {
			limit = {
				has_province_modifier = sed_outpost_phase_iii
				NOT = { local_autonomy = 50.01 }
			}
			sedentarization_outpost_try_phase_change = { current_phase_is_three = yes }
		}
	}
}

# Scripted effect "sedentarization_reset_province_outpost" turned into an event
province_event = {
	id = sedentarization.20	# regex "sedentarization\.20\b" to avoid "sedentarization.200"
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	option = {
		# These are non-exclusive; and clearing should happen before any outpost placement
		if = {
			limit = {
				NOT = { has_province_flag = sed_reset_to_phase_one }
			}
			sedentarization_INTERNAL_remove_outpost_modifiers = { phase = i }
		}
		if = {
			limit = {
				NOT = { has_province_flag = sed_reset_to_phase_two }
			}
			sedentarization_INTERNAL_remove_outpost_modifiers = { phase = ii }
		}
		if = {
			limit = {
				NOT = { has_province_flag = sed_reset_to_phase_three }
			}
			sedentarization_INTERNAL_remove_outpost_modifiers = { phase = iii }
		}
		clr_province_flag = sed_province_has_outpost
		clr_province_flag = sed_outpost_type_event_queued
		clr_province_flag = sed_settler_policy_event_queued
		clr_province_flag = sed_outpost_completion_event_queued
		if = {
			limit = { has_province_flag = sed_reset_to_phase_one }
			sedentarization_INTERNAL_place_outpost = {
				phase_numeral = phase_i
				old_min_autonomy = 100
			}
		}
		else_if = {
			limit = { has_province_flag = sed_reset_to_phase_two }
			sedentarization_INTERNAL_place_outpost = {
				phase_numeral = phase_ii
				old_min_autonomy = 80
			}
			set_province_flag = sed_trade_goods_upgrade_events_unlocked
			# Set a queueing flag. Normally used for taking care of invalidated choices,
			# it's also here for the edge-case that two provinces simultaneously progress.
			set_province_flag = sed_outpost_type_event_queued
			province_event = { id = sedentarization.200 }
		}
		else_if = {
			limit = { has_province_flag = sed_reset_to_phase_three }
			sedentarization_INTERNAL_place_outpost = {
				phase_numeral = phase_iii
				old_min_autonomy = 65
			}
			# Another queueing flag. Same purpose as the one above.
			set_province_flag = sed_settler_policy_event_queued
			# Don't fire simultaneous events on the same province; refire settler policy event after outpost type event finishes.
			if = {
				limit = {
					NOT = { has_province_flag = sed_outpost_type_event_queued }
				}
				province_event = { id = sedentarization.300 }
			}
		}
		else_if = {
			limit = { has_province_flag = sed_reset_completion }
			area = {
				# This immediately blocks the ability to place new outposts in the area;
				# necessary since other ways to detect a completed province won't get placed for a few days to a few months.
				add_province_modifier = {
					name = sed_system_completed
					duration = -1
					hidden = yes
				}
			}
			owner = {
				subtract_variable = {
					which = sed_outpost_cap_usage
					value = 1
				}
			}
			# Another queueing flag. Same purpose as the ones above.
			set_province_flag = sed_outpost_completion_event_queued
			if = {
				limit = {
					NOT = {
						has_province_flag = sed_outpost_type_event_queued
						has_province_flag = sed_settler_policy_event_queued
					}
				}
				province_event = { id = sedentarization.310 }
			}
		}
		clr_province_flag = sed_reset_to_phase_one
		clr_province_flag = sed_reset_to_phase_two
		clr_province_flag = sed_reset_to_phase_three
		clr_province_flag = sed_reset_completion
	}
}
