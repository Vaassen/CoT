country_event = { #Drunken Dmarvuks
	id = fp_flavour.2
	title = fp_flavour.2.t
	desc = fp_flavour.2.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = {
		forbidden_lands_superregion = {
			owned_by = ROOT
			NOT = { has_province_modifier = sed_forbidden }
			OR = {
				trade_goods = wine
				trade_goods = grain
				trade_goods = sugar
				trade_goods = tea
				trade_goods = glass
			}
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = forbidden_lands_superregion
					NOT = { has_province_modifier = sed_forbidden }
					OR = {
						trade_goods = wine
						trade_goods = grain
						trade_goods = sugar
						trade_goods = tea
						trade_goods = glass
					}
				}
				save_event_target_as = event_province
			}
		}
	}

	option = {
		name = fp_flavour.2.a
		event_target:event_province = {
			add_province_modifier = {
				name = centaur_fauna_bratty_dmarvuks
				duration = 1825 #5 years
			}
		}
	}
}

country_event = { #Rock Crabs Slide Through, Rimworld Alpha Animals style
	id = fp_flavour.3
	title = fp_flavour.3.t
	desc = fp_flavour.3.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = {
		forbidden_lands_superregion = {
			owned_by = ROOT
			has_port = yes
			province_group = fp_rivers
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = forbidden_lands_superregion
					has_port = yes
					province_group = fp_rivers
				}
				save_event_target_as = event_province
			}
		}
	}

	option = {
		name = fp_flavour.3.a
		add_mil_power = 15
		add_army_tradition = 5
		add_manpower = -1
	}

	option = {
		name = fp_flavour.3.b
		add_adm_power = 25
		event_target:event_province = {
			random_list = {
				45 = { remove_random_development = yes }
				45 = { add_random_development = 1 }
				10 = {
					add_province_modifier = {
						name = centaur_fauna_rock_crab_vex
						duration = 1825 #5 years
					}
				}
			}
		}
	}

	option = {
		name = fp_flavour.3.c
		trigger = { has_estate = estate_adventurers }
		highlight = yes
		add_mil_power = 15
		add_army_tradition = 5
		add_estate_influence_modifier = {
			estate = estate_adventurers
			desc = rock_crab_novice_tt
			influence = 10
			duration = 1825 #5 years
		}
		add_country_modifier = {
			name = centaur_fauna_rock_crab_novice
			duration = 1825 #5 years
		}
	}
}

country_event = { #Heeding the Headless
	id = fp_flavour.5
	title = fp_flavour.5.t
	desc = fp_flavour.5.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = { any_owned_province = { region = nuzurbokh_region } }

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = { region = nuzurbokh_region }
				save_event_target_as = event_province
			}
		}
	}

	option = {
		name = fp_flavour.5.a
		event_target:event_province = {
			add_province_modifier = {
				name = fp_fauna_headless_hyena
				duration = 3650 #10 years
			}
		}
	}

	option = {
		name = fp_flavour.5.b
		add_mil_power = -20
		event_target:event_province = {
			add_province_modifier = {
				name = fp_fauna_dire_hyena
				duration = 3650 #10 years
			}
		}
	}

	option = {
		name = fp_flavour.5.c
		if = {
			limit = { monthly_income = 83.333 }
			add_treasury = -100
		}
		else = { add_years_of_income = -0.1 }
		event_target:event_province = {
			if = {
				limit = { development = 25 }
				owner = { add_manpower = -2 }
			}
			else_if = {
				limit = { development = 15 }
				owner = { add_manpower = -1 }
			}
			else_if = {
				limit = { development = 7 }
				owner = { add_manpower = -0.75 }
			}
			else = { owner = { add_manpower = -0.5 } }
			add_province_modifier = {
				name = fp_faina_headed_off_hyena
				duration = 3650 #10 years
			}
		}
	}
}
