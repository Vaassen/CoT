country_event = {
	id = fp_flavour.1
	title = fp_flavour.1.t
	desc = fp_flavour.1.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = { forbidden_lands_superregion = { owned_by = ROOT } }

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = { superregion = forbidden_lands_superregion }
				save_event_target_as = event_province
			}
		}
	}

	option = {
		name = fp_flavour.1.a
		add_treasury = 25
	}
}

country_event = {
	id = fp_flavour.2
	title = fp_flavour.2.t
	desc = fp_flavour.2.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = {
		forbidden_lands_superregion = {
			owned_by = ROOT
			NOT = { has_province_modifier = sed_forbidden }
			OR = {
				trade_goods = wine
				trade_goods = grain
				trade_goods = sugar
				trade_goods = tea
				trade_goods = glass
			}
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = forbidden_lands_superregion
					NOT = { has_province_modifier = sed_forbidden }
					OR = {
						trade_goods = wine
						trade_goods = grain
						trade_goods = sugar
						trade_goods = tea
						trade_goods = glass
					}
				}
				save_event_target_as = event_province
			}
		}
	}

	option = {
		name = fp_flavour.2.a
		event_target:event_province = {
			add_province_modifier = {
				name = centaur_fauna_bratty_dmarvuks
				duration = 1825 #5 years
			}
		}
	}
}

country_event = {
	id = fp_flavour.3
	title = fp_flavour.3.t
	desc = fp_flavour.3.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = {
		forbidden_lands_superregion = {
			owned_by = ROOT
			has_port = yes
			province_group = fp_rivers
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = forbidden_lands_superregion
					has_port = yes
					province_group = fp_rivers
				}
				save_event_target_as = event_province
			}
		}
	}

	option = {
		name = fp_flavour.3.a
		add_mil_power = 15
		add_army_tradition = 5
		add_manpower = -1
	}

	option = {
		name = fp_flavour.3.b
		add_adm_power = 25
		event_target:event_province = {
			random_list = {
				45 = { remove_random_development = yes }
				45 = { add_random_development = 1 }
				10 = {
					add_province_modifier = {
						name = centaur_fauna_rock_crab_vex
						duration = 1825 #5 years
					}
				}
			}
		}
	}

	option = {
		name = fp_flavour.3.c
		trigger = { has_estate = estate_adventurers }
		highlight = yes
		add_mil_power = 15
		add_army_tradition = 5
		add_estate_influence_modifier = {
			estate = estate_adventurers
			desc = rock_crab_novice_tt
			influence = 10
			duration = 1825 #5 years
		}
		add_country_modifier = {
			name = centaur_fauna_rock_crab_novice
			duration = 1825 #5 years
		}
	}
}

country_event = {
	id = fp_flavour.4
	title = fp_flavour.4.t
	desc = fp_flavour.4.d
	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = {
		forbidden_lands_superregion = {
			owned_by = ROOT
			OR = {
				trade_goods = wool
				trade_goods = tropical_wood
			}
		}
	}

	option = {
		name = fp_flavour.4.a
		add_country_modifier = {
			name = centaur_fauna_bloody_wool
			duration = 365 #1 year
		}
	}

	option = {
		name = fp_flavour.4.b
		add_years_of_income = -0.5
		add_country_modifier = {
			name = centaur_fauna_cozy_wool
			duration = 365 #1 year
		}
	}

	option = {
		name = fp_flavour.4.c
		trigger = {
			OR = {
				ruler_has_personality = careful_personality
				ruler_has_personality = calm_personality
				ruler_has_personality = tactical_genius_personality
			}
		}
		highlight = yes
		add_years_of_income = -0.5
		add_country_modifier = {
			name = centaur_fauna_cozy_wool
			duration = 1825 #5 year
		}
	}

	option = {
		name = fp_flavour.4.e
		trigger = {
			ruler_is_powerful_mage = yes
			OR = {
				is_ruler_illusion_level_1 = yes
				is_ruler_enchantment_level_1 = yes
			}
		}
		highlight = yes
		add_country_modifier = {
			name = centaur_fauna_cozy_wool
			duration = 365 #1 year
		}
	}
}
