namespace = flavour_carmilla

province_event = {
	id = flavour_carmilla.1
	title = flavour_carmilla.1.t
	desc = flavour_carmilla.1.d
	picture = DIPLOMACY_eventPicture
	goto = ROOT

	is_triggered_only = yes

	trigger = {
		NOT = {
			AND = {
				advisor_exists = 5001 # Erehan
				has_leader = "Erehan Gleisrein"
			}
		}
	}

	option = {
		name = flavour_carmilla.1.a
		
		if = {
			limit = {
				tag = V30
			}
			create_advisor = 5001
		}
		else = {
			create_advisor = 5002
		}
	}
	
	
	option = {
		name = flavour_carmilla.1.b
		
		if = {
			limit = {
				tag = V30
			}
			define_general = {
				name = "Erehan Gleisrein"
				fire = 2
				shock = 4
				manuever = 2
				siege = 2
				age = 30
			}
		}
		else = {
			define_general = {
				name = "Erehan Gleisrein"
				fire = 4
				shock = 6
				manuever = 4
				siege = 4
				age = 30
			}
		}
	}
}
