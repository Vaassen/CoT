#kobold rebellion
country_event = {
    id = vaengheim.5
    title = vaengheim.5.t
    desc = vaengheim.5.d
    picture = COMBINED_HAGS_eventPicture
    is_triggered_only = yes
    trigger = {
        has_country_flag = Z88_kobold_rebellion
        NOT = {
            check_variable = {
                which = Z88_lots_of_kobolds
                value = 30
            }
        }
    }

    option = {
		name = vaengheim.5.a
        custom_tooltip = Z88_kobold_rebellion_tt
        random_owned_province = {
            limit = {
                region = dragon_coast_region
            }
            spawn_rebels = {
                size = 3
            }
        }
        random_owned_province = {
            limit = {
                region = dragon_coast_region
            }
            spawn_rebels = {
                size = 3
            }
        }
        change_variable = {
            which = Z88_lots_of_kobolds
            value = 1
        }
    }
    after = {
        country_event = {
            id = vaengheim.5
            days = 90
        }
    }
}
