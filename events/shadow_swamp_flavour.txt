country_event = {
	#Handler event, determines which event the country gets
	id = shadowswamp.000
	title = shadowswamp.000.t
	desc = shadowswamp.000.d
	picture = MIRACLE_MONK_eventPicture
	is_triggered_only = yes
	hidden = yes
	mean_time_to_happen = {
		days = 1
	}
	trigger = {
		NOT = {
			has_country_modifier = shadowswamp_event_cooldown
		}
		any_owned_province = {
			region = yezel_mora_region
			is_city = yes
		}
	}
	immediate = {
		random_owned_province = {
			limit = {
				region = yezel_mora_region
				is_city = yes
			}
			save_event_target_as = shadowswamp_spawn_target
		}
	}
	after = {
		if = {
			limit = {
				has_global_flag = shadowtree_purified
			}
			add_country_modifier = {
				name = shadowswamp_event_cooldown
				duration = 1825
				hidden = yes
			}
		}
		else = {
			add_country_modifier = {
				name = shadowswamp_event_cooldown
				duration = 3650
				hidden = yes
			}
		}
	}
	option = {
		# not swamptroll or echenka
		name = shadowswamp.000.a
		trigger = {
			NOT = {
				primary_culture = swamp_troll
				primary_culture = echenka
			}
		}
		event_target:shadowswamp_spawn_target = {
			random_list = {
				10 = {
					province_event = {
						id = shadowswamp.102
					}					#Fire's Caterwaul
				}
				10 = {
					province_event = {
						id = shadowswamp.104
					}					#the last wish
				}
				10 = {
					trigger = {
						has_influencing_fort = no
						NOT = {
							has_global_flag = shadowtree_purified
						}
					}
					province_event = {
						id = shadowswamp.103
					}					#Lurkers in foul water
				}
				10 = {
					trigger = {
						NOT = {
							has_global_flag = shadowtree_purified
						}
					}
					province_event = {
						id = shadowswamp.108
					}					#A Break in the Grey
				}
			}
		}
		ai_chance = {
			factor = 100
		}
	}
	option = {
		# can happen to anyone
		name = shadowswamp.000.b
		event_target:shadowswamp_spawn_target = {
			random_list = {
				10 = {
					province_event = {
						id = shadowswamp.106
					}					#shambling mounds
				}
				10 = {
					trigger = {
						NOT = {
							has_global_flag = shadowtree_purified
						}
						owner = {
							primary_culture = swamp_troll
						}
					}
					province_event = {
						id = shadowswamp.109
					}					#A Break in the Grey (trolls)
				}
				10 = {
					province_event = {
						id = shadowswamp.105
					}					#undead swarm
				}
				10 = {
					province_event = {
						id = shadowswamp.107
					}					#Nightmares
				}
				10 = {
					province_event = {
						id = shadowswamp.110
					}					#Hattiatia Trove Discovered
				}
				20 = {
					trigger = {
						trade_goods = coffee
					}
					province_event = {
						id = shadowswamp.101
					}					#Coffee Jitters
				}
			}
		}
		ai_chance = {
			factor = 100
		}
	}
}
