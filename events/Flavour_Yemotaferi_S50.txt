# Migrate to the swamp
country_event = {
	id = yemotaferi.4
	title = yemotaferi.4.t
	desc = yemotaferi.4.d
	picture = BORDER_TENSION_eventPicture
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = yemotaferi.4.a

		custom_tooltip = S50_migration_effect_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					NOT = { region = shadow_swamp_region }
					NOT = {
						# Exclude the two starting provinces.
						OR = {
							province_id = 5616
							province_id = 5617
						}
					}
				}
				export_to_variable = {
					which = migrateDevTax
					value = development
				}
				change_variable = { which = migrateDevTax value = -1 }
				ROOT = { change_variable = { which = migrateDevTax which = PREV } }

				export_to_variable = {
					which = migrateDevProduction
					value = development
				}
				change_variable = { which = migrateDevProduction value = -1 }
				ROOT = { change_variable = { which = migrateDevProduction which = PREV } }

				export_to_variable = {
					which = migrateDevManpower
					value = development
				}
				change_variable = { which = migrateDevManpower value = -1 }
				ROOT = { change_variable = { which = migrateDevManpower which = PREV } }

				add_base_tax = -100
				add_base_production = -100
				add_base_manpower = -100

				random_core_country = {
					limit = {
						NOT = { tag = ROOT }
					}
					save_event_target_as = tag_to_cede_to
				}

				random_core_country = {
					limit = {
						NOT = { tag = ROOT }
						NOT = { tag = L66 } 
						NOT = { tag = S70 }
						NOT = { tag = S65 }
						# Avoid giving land to Yezel Mora, Simegasa, and Mukis if possible
					}
					save_event_target_as = tag_to_cede_to
				}

				cede_province = tag_to_cede_to
			}

			while = {
				limit = { check_variable = { migrateDevTax = 1 } }
				subtract_variable = { migrateDevTax = 1 }
				random_owned_province = {
					add_base_tax = 1
				}
			}

			while = {
				limit = { check_variable = { migrateDevProduction = 1 } }
				subtract_variable = { migrateDevProduction = 1 }
				random_owned_province = {
					add_base_production = 1
				}
			}

			while = {
				limit = { check_variable = { migrateDevManpower = 1 } }
				subtract_variable = { migrateDevManpower = 1 }
				random_owned_province = {
					add_base_manpower = 1
				}
			}
		}


		6718 = {
			if = {
				NOT = {
					has_great_project = {
						type = hawatli_wizti_tali
						tier = 2
					}
				}
			}
			add_great_project_tier = {
				type = hawatli_wizti_tali
				tier = 1
			}
			add_devastation = -20
		}
	}
}
