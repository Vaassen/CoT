# From Chains to Court
country_event = {
	id = gnoll_flavor.3
	title = gnoll_flavor.3.t
	desc = gnoll_flavor.3.desc
	picture = BORDER_TENSION_eventPicture

	is_triggered_only = yes

	trigger = {
		OR = {
			has_estate_privilege = gnoll_the_weak_serve_the_strong
			has_estate_privilege = gnoll_the_weak_serve_the_strong_burghers
		}
		any_owned_province = {
			trade_goods = slaves
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = { trade_goods = slaves }
				save_event_target_as = event_province

				if = {
					limit = {
						NOT = { has_gnollish_majority_trigger = yes }
					}
					save_event_target_as = slave_culture_source_province
				}
				else_if = {
					limit = {
						any_owned_province = { NOT = { has_gnollish_majority_trigger = yes } }
					}
					random_owned_province = {
						limit = {
							NOT = { has_gnollish_majority_trigger = yes }
						}
						save_event_target_as = slave_culture_source_province
					}				
				}
				else = {
					random_known_province = {
						limit = {
							NOT = { has_gnollish_majority_trigger = yes }
						}
						save_event_target_as = slave_culture_source_province
					}
				}
			}
		}
	}

	option = {
		name = gnoll_flavor.3.a

		add_legitimacy_equivalent = {
			amount = -10
			republican_tradition = -5
		}

		trigger_switch = {
			on_trigger = government_rank
			1 = {
				random_list = {
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_duchy_adm
						define_random_adm_advisor_effect = {
							skill = 1
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_duchy_dip
						define_random_dip_advisor_effect = {
							skill = 1
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_duchy_mil
						define_random_mil_advisor_effect = {
							skill = 1
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
				}
			}
			2 = {
				random_list = {
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_kingdom_adm
						define_random_adm_advisor_effect = {
							skill = 2
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_kingdom_dip
						define_random_dip_advisor_effect = {
							skill = 2
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_kingdom_mil
						define_random_mil_advisor_effect = {
							skill = 2
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
				}
			}
			3 = {
				random_list = {
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_empire_adm
						define_random_adm_advisor_effect = {
							skill = 3
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_empire_dip
						define_random_dip_advisor_effect = {
							skill = 3
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
					1 = {
						custom_tooltip = gnoll_flavor.3.a.tooltip_empire_mil
						define_random_mil_advisor_effect = {
							skill = 3
							culture = event_target:slave_culture_source_province
							religion = ROOT
							cost_multiplier = 0.5
						}
					}
				}
			}
		}
	}

	option = {
		name = gnoll_flavor.3.b

		add_legitimacy_equivalent = {
			amount = 10
			republican_tradition = 5
		}

		event_target:event_province = {
			area = {
				add_province_modifier = {
					modifier = gnoll_local_slave_leader
					duration = 3650 # 10 years
				}
			}
		}

		custom_tooltip = gnoll_flavor.3.b.tooltip
	}
}
