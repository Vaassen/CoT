# Ebonsteel price increase
country_event = {
	id = cot_price.1
	title = cot_price.1.t
	desc = cot_price.1.d
	picture = FIREARMS_INCIDENT_eventPicture
	
	fire_only_once = yes
	major = yes
	
	mean_time_to_happen = {
		years = 1
	}
	
	trigger = {
		mil_tech = 8 #1505 Pike and shot
		NOT = { has_global_flag = ebonsteel_armour_happened }
		any_owned_province = {
			trade_goods = ebonsteel
		}
	}
	
	option = {
		name = cot_price.1.a
		ai_chance = { factor = 100 }
		
		change_price = {
			trade_goods = ebonsteel
			key = EBONSTEEL_ARMOUR_HAPPENED
			
			value = 0.25
			duration = -1
		}
		hidden_effect = { set_global_flag = ebonsteel_armour_happened }
	}
}
