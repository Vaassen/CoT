namespace = khetarch_curse_event

#Large Famine
country_event = {
    id = flavour_brrtekuh.91
	title = flavour_brrtekuh.91.t
	desc = flavour_brrtekuh.91.d
	picture = PRIEST_PHANTOM_eventPicture
    is_triggered_only = yes
	trigger = {
		has_country_modifier = U10_khetarch_curse_2
	}
	immediate = {
		random_owned_province = {
			limit = { NOT = { has_province_modifier = curse_severe_famine }}
			save_event_target_as = curse_famine_province
		}
	}
    option = {
		name = flavour_brrtekuh.91.a
		event_target:curse_famine_province = {
			area = {
				add_province_modifier = {
					name = curse_severe_famine
					duration = 7300
				}
				add_devastation = 10
			}
		}
	}
}
