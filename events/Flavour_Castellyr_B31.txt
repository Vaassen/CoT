# Glory or Honor

	id = flavor_castellyr.1
	title = flavor_castellyr.1.t
	desc = flavor_castellyr.1.d
	picture = MILITARY_CAMP_eventPicture
	
	is_triggered_only = yes

		tag = B31
	}

	option = {
		name = flavor_castellyr.1.a #We shall quest for glory!
		set_country_flag = cot_yes_castanor_flag
		hidden_effect = {
			swap_non_generic_missions = yes
		}
	}

	option = {
		name = flavor_castellyr.1.b #We shall safeguard honor! (Disables Castanor)
		set_country_flag = cot_no_castanor_flag
		set_government_rank = 3
		hidden_effect = {
			remove_government_reform = autocracy_reform
			remove_government_reform = centralize_reform
			remove_government_reform = centralize_empire_reform
			change_government_reform_progress = 150
		}
		add_government_reform = honorbound_kingdom_reform
		country_event = {
			id = flavor_castellyr.2
			days = 7
		}
	}
}

# Fate of the Orcs
country_event = {
	id = flavor_castellyr.4
	title = flavor_castellyr.4.t
	desc = flavor_castellyr.4.d
	picture = MILITARY_CAMP_eventPicture
	
	is_triggered_only = yes

	option = {
		name = flavor_castellyr.4.a #Half Orc
		trigger = {
			accepted_culture = green_orc
			# OR = {
			# 	has_coexisting_orcish_pop_trigger = yes
			# 	has_integrated_orcish_pop_trigger = yes
			# }
		} 
		
		custom_tooltip = cot_b31_horc_tt 
		hidden_effect = {
			change_primary_culture = ungulan_half_orc
			clear_racial_military = yes
			clear_racial_administration = yes
			
			add_country_modifier = { 
				name = half_orcish_administration
				duration = -1
				hidden = yes
			}
			
			add_country_modifier = { 
				name = half_orcish_military
				duration = -1
				hidden = yes
			}
			every_owned_province = {
				limit = {
					OR = {
						culture_group = orcish
						culture_group = escanni
						culture_group = rimmeric
						culture_group = damerian
						culture_group = lencori
						culture_group = alenic
						culture_group = esmari_g
					}
				}
				random = {
					chance = 55
					change_culture = ungulan_half_orc
					change_religion = ROOT
				}
			}
		}
	}

	option = {
		name = flavor_castellyr.4.b # Seperate but Orkqual
		
		if = {
			limit = { accepted_culture = green_orc }
			add_country_modifier = {
				name = "cot_B31_acceptance" # +1 accepted cultures +5% shock damage
				duration = -1
			}
		}
		else = {
			add_country_modifier = {
				name = "cot_B31_servitude" # +5% goods produced
				duration = -1
			}
			set_country_flag = cot_B31_slavery
		}
	}

	option = {
		name = flavor_castellyr.4.c # HUMAN First
		trigger = {
			OR = {
				has_country_modifier = racial_pop_orcish_expulsion
				has_country_modifier = racial_pop_orcish_purge
			}
		}
		
		add_country_modifier = {
			name = "cot_B31_humanity" # -10% culture conversion cost -25% culture conversion time +0.25 prestige
			duration = -1
		}
		set_country_flag = cot_B31_slavery
	}
}
