country_event = {
    id = flavour_star_rider.10 #Hidden Event for Temple effects
    title = flavour_star_rider.10.t
    desc = flavour_star_rider.10.d
    picture = SIEGE_eventPicture
    
    is_triggered_only = yes
	hidden = yes
    
    option = {
        name = flavour_star_rider.10.a
		overlord = { change_mana = { amount = 5 } }
		if = {
			limit = { NOT = { has_country_modifier = I69_temple_cooldown } }
			add_liberty_desire = -20
			add_country_modifier = {
				name = I69_temple_cooldown
				duration = 1095
			}
		}
		if = {
			limit = { NOT = { has_country_flag = I69_extorted_manpower } }
			overlord = { 
				add_yearly_manpower = 0.5
			} 
			set_country_flag = I69_extorted_manpower
		}
		if = { #Building a temple in a subject will release all countries, those becoming Tributaries of Star Rider
			limit = { overlord = { has_country_flag = I69_temple_2 } }
			add_liberty_desire = 5
			every_owned_province = {
				set_province_flag = I69_release_then_take
			}
			release_all_possible_countries = yes
			forbidden_lands_superregion = {
				limit = {
					has_province_flag = I69_release_then_take
				}
				owner = {
					if = {
						limit = { 
							NOT = { 
								is_subject_of_type_with_overlord = {
									who = I69
									type = tributary_state_anb
								}
							}
						}
						I69 = {
							create_subject = {
								subject_type = tributary_state_anb
								who = PREV
							}
							I69_magical_feudalism_effect = yes
						}
					}
				}
				clr_province_flag = I69_release_then_take
			}
		}
    }
