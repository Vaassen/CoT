country_event = {
    id = AnzalkatsaSchoolFightEvents.1
	title = AnzalkatsaSchoolFightEvents.1.t
	desc = AnzalkatsaSchoolFightEvents.1.d
	picture = WESTERNISATION_eventPicture
	
	is_triggered_only = yes
    hidden = yes

    option = {
        name = AnzalkatsaSchoolFightEvents.1.a
        6150 = {
            set_variable = {
                which = lizSchoolFightVar
                value = 1
            }
        }
        imperial_332_cities = {
            limit = {
                owner = {
                    religious_school = { group = lizardfolk_religion school = being_sculptors_school }
                }
            }
            add_permanent_province_modifier = { name = lizard_being_controlled_city duration = -1 } 
        }
        imperial_332_cities = {
            limit = {
                owner = {
                    religious_school = { group = lizardfolk_religion school = fate_sculptors_school }
                }
            }
            add_permanent_province_modifier = { name = lizard_fate_controlled_city duration = -1 } 
        }
        imperial_332_cities = {
            limit = {
                owner = {
                    NOT = {religion = anzalkatsa}
                }
            }
            add_permanent_province_modifier = { name = heretic_controlled_city duration = -1 } 
        }
        every_country = {
            limit = {
                religion = anzalkatsa
            }
            set_country_flag = anzalkatsaFollowerFlag
        }

        L68 = { 
            set_country_flag = KarraskStartWarKarshyr
            set_country_flag = KarraskStartWarAsarta
        }
        set_global_flag = noAshhanaDemonsterizeYet 
    }
}

#Subtract + Update
country_event = {
    id = AnzalkatsaSchoolFightEvents.2
	title = AnzalkatsaSchoolFightEvents.2.t
	desc = AnzalkatsaSchoolFightEvents.2.d
	picture = WESTERNISATION_eventPicture
	
	is_triggered_only = yes
    hidden = yes

    option = {
        name = AnzalkatsaSchoolFightEvents.2.a
        if = {
            limit = {
                6150 = {
                    NOT = { is_variable_equal = { which = lizSchoolFightVar value = -3 } }
                } 
            }
            6150 = {
                change_variable = {
                    which = lizSchoolFightVar
                    value = -1
                }
            }
            if = {
                limit = {
                    6150 = { is_variable_equal = { which = lizSchoolFightVar value = -2 } }
                }
                every_country = {
                    limit = {
                        religion = anzalkatsa
                    }
                    if = {
                        limit = {
                            OR = {
                                religious_school = {
                                    group = lizardfolk_religion
                                    school = fate_sculptors_school
                                }
                                religious_school = {
                                    group = lizardfolk_religion
                                    school = fate_sculptors_school_ascended
                                }
                            }
                        }
                        set_religious_school = {
                            group = lizardfolk_religion
                            school = fate_sculptors_school_humiliated
                        }
                    }
                    else = {
                        set_religious_school = {
                            group = lizardfolk_religion
                            school = being_sculptors_school_ascended
                        }
                    }
                    
                }
            }
            else_if = {
                limit = {
                    6150 = { is_variable_equal = { which = lizSchoolFightVar value = 1 } }
                }
                every_country = {
                    limit = {
                        religion = anzalkatsa

                        set_religious_school = {
                            group = lizardfolk_religion
                            school = being_sculptors_school
                        }
                    }
                    
                }
            }
        }
    }
}
