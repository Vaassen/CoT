# The Barons' Revolt
country_event = {
	id = flavor_parusahan_saya.8
	title = flavor_parusahan_saya.8.t
	desc = flavor_parusahan_saya.8.d
	picture = TIGER_THRONE_eventPicture

	hidden = no
	is_triggered_only = yes

	trigger = {
		
	}
	immediate = {
		if = {
			limit = {
				calc_true_if = {
					amount = 3
					has_country_modifier = R54_bypassed_tollkeepers
					has_country_modifier = R54_artisanal_obstruction
					has_country_modifier = R54_landless_loomlords
				}
			}
			set_country_flag = R54_baron_revolt_severity_high
		} 
		else_if = {
			limit = {
				calc_true_if = {
					amount = 2
					has_country_modifier = R54_bypassed_tollkeepers
					has_country_modifier = R54_artisanal_obstruction
					has_country_modifier = R54_landless_loomlords
				}
			}
			set_country_flag = R54_baron_revolt_severity_med
		}
		else = {
			set_country_flag = R54_baron_revolt_severity_low
		}

		if = {
			limit = { has_country_flag = R54_baron_revolt_severity_high }
			set_variable = {
				which = R54_baron_revolt_rebel_count
				value = 15
			}
		}
		else_if = {
			limit = { has_country_flag = R54_baron_revolt_severity_med }
			set_variable = {
				which = R54_baron_revolt_rebel_count
				value = 10
			}
		}
		else_if = {
			limit = { has_country_flag = R54_baron_revolt_severity_low }
			set_variable = {
				which = R54_baron_revolt_rebel_count
				value = 5
			}
		}
	}

	after = {
		set_country_flag = R54_barons_revolted
		remove_country_modifier = R54_landless_loomlords
		remove_country_modifier = R54_artisanal_obstruction
		remove_country_modifier = R54_bypassed_tollkeepers
	}

	option = {
		name = flavor_parusahan_saya.8.a

		if = {
			limit = {
				has_country_flag = R54_baron_revolt_severity_high
			}
			add_stability = -3
			every_owned_province = {
				add_local_autonomy = 30
			}
			add_estate_loyalty = {
				estate = estate_uppercastes
				loyalty = -40
			}
			add_estate_loyalty = {
				estate = estate_middlecastes
				loyalty = -40
			}
			add_estate_loyalty = {
				estate = estate_lowercastes
				loyalty = -40
			}
		}
		else_if = {
			limit = {
				has_country_flag = R54_baron_revolt_severity_med
			}
			add_stability = -2
			every_owned_province = {
				add_local_autonomy = 20
			}
			add_estate_loyalty = {
				estate = estate_uppercastes
				loyalty = -30
			}
			add_estate_loyalty = {
				estate = estate_middlecastes
				loyalty = -30
			}
			add_estate_loyalty = {
				estate = estate_lowercastes
				loyalty = -30
			}
		}
		else = {
			add_stability = -1
			every_owned_province = {
				add_local_autonomy = 10
			}
			add_estate_loyalty = {
				estate = estate_uppercastes
				loyalty = -20
			}
			add_estate_loyalty = {
				estate = estate_middlecastes
				loyalty = -20
			}
			add_estate_loyalty = {
				estate = estate_lowercastes
				loyalty = -20
			}
		}

		while = {
			limit = { check_variable = { which = R54_baron_revolt_rebel_count value = 1 } }
			random_owned_province = {
				spawn_rebels = {
					type = anti_tax_rebels
					size = 1
				}
			}
			change_variable = {
				which = R54_baron_revolt_rebel_count
				value = -1
			}
		}
	}
}
