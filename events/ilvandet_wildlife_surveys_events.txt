namespace = ilvandet_surveys

country_event = {
    id = ilvandet_surveys.1
    title = ilvandet_surveys.1.t
    desc = ilvandet_surveys.1.d
    picture = EXPLORATION_eventPicture
    is_triggered_only = yes

    trigger = {
        has_country_flag = ilvandet_trails_survey_started
        NOT = { has_country_flag = ilvandet_trails_survey_completed }
    }

    immediate = {
        if = {
            limit = {
                has_country_flag = ilvandet_trails_survey_started
                NOT = { has_country_flag = ilvandet_trails_survey_completed }
            }
            set_country_flag = ilvandet_trails_survey_completed
            change_variable = {
                which = ilvandet_surveys_completed
                value = 1
            }
        }
    }

    option = {
        name = ilvandet_surveys.1.a
        custom_tooltip = ilvandet_survey_completed_tt
    }
}
