country_event = {
    id = ar_magisterium.20
    title = ar_mx_college_site.t
    desc = ar_mx_college_site.d
    picture = ARTWORK_eventPicture
    is_triggered_only = yes
    trigger = {
        has_country_flag = ar_mx_menu_open
        has_country_flag = ar_mx_menu_college
    }
    option = {
        name = ar_mx_site.confirm
        trigger = {
            ar_mx_can_buy_college = yes
            event_target:ar_mx_site = {
                ar_mx_college_candidate = yes
            }
        }
        ai_chance = {
            factor = 100
        }
        custom_tooltip = ar_mx_college_cost_tt
        hidden_effect = {
            if = {
                limit = {
                    has_country_flag = ar_mx_menu_college
                }
                ar_mx_buy_college = yes
            }
            ar_mx_close_menu = yes
        }
    }
    option = {
        name = ar_mx_site.next
        trigger = {
            ar_mx_can_buy_college = yes
            any_province = {
                ar_mx_college_candidate = yes
                NOT = {
                    province_id = event_target:ar_mx_site
                }
            }
        }
        ai_chance = {
            factor = 0
        }
        hidden_effect = {
            ar_mx_next_site = {
                mode = college
            }
            country_event = {
                id = ar_magisterium.21
            }
        }
    }
    option = {
        name = ar_mx_site.cancel
        ai_chance = {
            factor = 1
        }
        hidden_effect = {
            ar_mx_close_menu = yes
        }
    }
}
