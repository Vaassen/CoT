country_event = {
	id = flavour_karshyr.51
	title = flavour_karshyr.51.t
	desc = flavour_karshyr.51.d
	picture = DIPLOMACY_eventPicture

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		any_country = {
			has_opinion = {
				who = S15
				value = 0
			}
			capital_scope = { superregion = western_cannor_superregion }
		}
		any_owned_province = {
			development = 30
			has_seat_in_parliament = yes
			NOT = { has_province_modifier = S15_early_cannorian_investment }
		}
	}

	immediate = {
		hidden_effect = {
			random_country = {
				limit = {
					has_opinion = {
						who = S15
						value = 0
					}
					capital_scope = { superregion = western_cannor_superregion }
				}
				save_event_target_as = event_country
			}
			random_owned_province = {
				limit = {
					development = 30
					has_seat_in_parliament = yes
					NOT = { has_province_modifier = S15_early_cannorian_investment }
				}
				save_event_target_as = event_province
			}
		}
	}

	mean_time_to_happen = {
		days = 1
		modifier = {
			factor = 3
			has_country_flag = S15_declined_51
		}
	}

	option = {
		name = flavour_karshyr.51.a
		trade_company_adzalaz_gulf = {
			add_trade_modifier = {
				who = event_target:event_country
				duration = 3560 #10 years
				power = 50
				key = "Generous Cannorian Investment"
			}
		}
		event_target:event_province = {
			random_list = {
				1 = { add_base_tax = 1 }
				1 = { add_base_production = 1 }
				1 = { add_base_manpower = 1 }
			}
			add_local_autonomy = 0.1
			add_province_modifier = {
				name = S15_early_cannorian_investment
				duration = 3650 #10 years
			}
		}
		decrease_monstrous_1 = yes
	}

	option = {
		name = flavour_karshyr.51.b
		set_country_flag = S15_declined_51
		add_opinion = {
			who = event_target:event_country
			modifier = S15_declined_generous_investment
		}
	}
}
