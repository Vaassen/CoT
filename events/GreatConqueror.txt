
namespace = great_conqueror

country_event = {
	id = great_conqueror.0
	title = great_conqueror.0.t
	desc = great_conqueror.0.d
	picture = ORTHODOX_eventPicture

	trigger = {
		has_dlc = "Rights of Man"
		OR = {
			has_global_flag = great_conqueror
			has_global_flag = mythical_conqueror 
		}
		is_subject = no
		#If delayed conquerors are on, then it can't be the age of discovery.
		OR = {
			NOT = { has_global_flag = delayed_conquerors }
			NOT = { current_age = age_of_discovery }
		}
		ai = yes
		NOT = { has_country_flag = vile_one_vc }
	}
	
	hidden = yes
	is_triggered_only = yes
	
	immediate = {
		if = {
			limit = { has_country_flag = previous_ruler_was_great_conqueror_flag }
			clr_country_flag = previous_ruler_was_great_conqueror_flag
		}
		if = {
			limit = { has_country_flag = great_conqueror_flag }
			REB = { subtract_variable = { nbGC = 1 } }
			clr_country_flag = great_conqueror_flag
			set_country_flag = previous_ruler_was_great_conqueror_flag
			export_to_variable = {
				which = ownDev
				value = total_development
				who = ROOT
			}
			add_ruler_modifier = {
				name = legacy_of_great_conqueror
				duration = -1
			}
			every_owned_province = {
				remove_province_modifier = seat_of_centaur_gc
			}
			log = "GREAT CONQUEROR ---------------- [Root.GetName] |[Root.ownDev.GetValue]| stop being a Great Conqueror in [GetYear] [GetMonth]. It's a [Root.GovernmentName] | Number of GC: [REB.nbGC.GetValue]"
		}
	}
	
	option = {
		name = great_conqueror.0.a
		ai_chance = { 
			factor = 1
			
			modifier = {
				factor = 1.5
				overextension_percentage = 0.5
			}
			
			modifier = {
				factor = 1.5
				average_unrest = 1
			}
			
			modifier = {
				factor = 1.5
				num_of_loans = 5
			}
			
			modifier = {
				factor = 1.5
				corruption = 5
			}
			
			modifier = {
				factor = 1.5
				NOT = { stability = 0 }
			}
			
			modifier = {
				factor = 50
				government = republic
			}
			
			modifier = {
				factor = 25
				government = theocracy
			}
			
			modifier = {
				factor = 10
				is_subject = yes
			}
		}
	}

	option = {
		name = great_conqueror.0.b
		trigger = {
			NOT = { has_country_flag = previous_ruler_was_great_conqueror_flag } 					#Can't have it 2time in a row
		}
		ai_chance = { 
			factor = 5
			
			modifier = {
				factor = 0
				OR = {
					if = {
						limit = { has_global_flag = mythical_conqueror }
						REB = { check_variable = { nbGC = 13 } }									#No more than 13 Great Conqueror in the world
					}
					else = {
						REB = { check_variable = { nbGC = 10 } }									#No more than 10 Great Conqueror in the world
					}
					is_bankrupt = yes																#Can't be bankrupt bitch
					AND = {
						any_neighbor_country = { ai = no }											#Doesn't target AI next to player before 1550
						NOT = { is_year = 1550 }
					}
					coalition_target = ROOT															#Must not be the coalition target
					NOT = { num_of_cities = 2 }														#Must have more than one province
					if = {
						limit = { government = tribal }
						tribal_elligible_for_gc = no
					}
				}
			}
			
			modifier = {
				factor = 1.5
				is_great_power = yes
			}
			
			modifier = {
				factor = 1.5
				MIL = 3 ADM = 3 DIP = 3
			}
			modifier = {
				factor = 2
				MIL = 4 ADM = 4 DIP = 4
			}
			modifier = {
				factor = 2.5
				MIL = 5 ADM = 5 DIP = 5
			}
			modifier = {
				factor = 3
				MIL = 6 ADM = 6 DIP = 6
			}
			
			modifier = {
				factor = 3
				NOT = { ruler_age = 30 }
			}
			
			modifier = {
				factor = 1.5
				ruler_is_powerful_mage = yes
			}
			
			modifier = {
				factor = 8
				REB = { NOT = { check_variable = { nbGC = 2 } } }									#More chance if less than 2 Great Conqueror
			}
			
			modifier = {
				factor = 4
				REB = { NOT = { check_variable = { nbGC = 4 } } }									#More chance if less than 4 Great Conqueror
			}
			
			modifier = {
				factor = 2
				REB = { NOT = { check_variable = { nbGC = 5 } } }									#More chance if less than 5 Great Conqueror
			}
			
			modifier = {
				factor = 2
				capital_scope = { superregion = escann_superregion }
				total_development = 500
			}
			
			modifier = {																			#Weak neighbours give more chance to become one
				factor = 2
				calc_true_if = {
					all_neighbor_country = {
						NOT = { is_part_of_hre = yes }
						NOT = { alliance_with = ROOT }
						NOT = { is_subject_of = ROOT }
						NOT = { army_strength = { who = ROOT value = 0.4 } }
					}
					amount = 2
				}
			}
			
			modifier = {
				factor = 3
				calc_true_if = {
					all_neighbor_country = {
						NOT = { is_part_of_hre = yes }
						NOT = { alliance_with = ROOT }
						NOT = { is_subject_of = ROOT }
						NOT = { army_strength = { who = ROOT value = 0.4 } }
					}
					amount = 3
				}
			}
			
			modifier = {
				factor = 4
				calc_true_if = {
					all_neighbor_country = {
						NOT = { is_part_of_hre = yes }
						NOT = { alliance_with = ROOT }
						NOT = { is_subject_of = ROOT }
						NOT = { army_strength = { who = ROOT value = 0.4 } }
					}
					amount = 4
				}
			}
			
			modifier = {
				factor = 5
				calc_true_if = {
					all_neighbor_country = {
						NOT = { is_part_of_hre = yes }
						NOT = { alliance_with = ROOT }
						NOT = { is_subject_of = ROOT }
						NOT = { army_strength = { who = ROOT value = 0.4 } }
					}
					amount = 5
				}
			}
			
			modifier = {
				factor = 6
				calc_true_if = {
					all_neighbor_country = {
						NOT = { is_part_of_hre = yes }
						NOT = { alliance_with = ROOT }
						NOT = { is_subject_of = ROOT }
						NOT = { army_strength = { who = ROOT value = 0.4 } }
					}
					amount = 6
				}
			}
			
			modifier = {
				factor = 2
				all_neighbor_country = {
					NOT = { tech_difference = 1 }													#More chance to become one if technologically advanced compared to all neighbor
				} 
			}
			
			modifier = {
				factor = 5
				is_revolution_target = yes
			}
			
			modifier = {
				factor = 3
				power_projection = 50
			}
			
			modifier = {
				factor = 5
				has_global_flag = mythical_conqueror
				is_great_power = yes
			}

			modifier = {
				factor = 3
				centaur_can_become_vc = yes															#More common if chance to become Khüraen Caehn
			}
		}
		
		if = {
			limit = {
				has_global_flag = mythical_conqueror
				OR = {
					is_great_power = yes
					total_development = 1000
				}
			}
			create_mythical_conqueror = yes
		}
		else = {
			create_great_conqueror = yes
		}

		#Randomly spawns a variant conqueror
		random_list = {
			#Variant Conqueror
			1 = {
				modifier = {
					factor = 5
					ruler_has_personality_or_ancestor = { personality = mythical_conqueror_personality }
				}
				modifier = {
					# No variant conquerors without turning them on in settings
					factor = 0
					NOT = { has_global_flag = variant_conqueror }
				}
				modifier = {
					# Requires that a player can see the tag
					factor = 0
					NOT = {
						any_country = {
							ai = no
							knows_country = ROOT
						}
					}
				}

				#Vampire Conqueror
				if = {
					limit = {
						has_estate = estate_vampires
					}
					country_event = { id = great_conqueror.3 }
				}
				#Umbral Covenant
				else_if = {
					limit = {
						sarhali_can_become_vc = yes
					}
					set_country_flag = vile_one_vc
					set_global_flag = umbral_covenant_formed
					country_event = { id = great_conqueror.7 }
				}
				#Lich Conqueror
				else = {
					country_event = { id = great_conqueror.1 }
				}
			}
			#No Variant Conqueror
			30 = {
			}
		}
	}
}

		has_not_banned_magic = yes

	}

		name = great_conqueror.1.a
		if = {
			limit = { ruler_is_powerful_mage = yes } #Already has flags
			define_type_to_war_wizard_effect = { type = ruler }
			become_a_witch_king_effect = yes
			become_a_lich_effect = yes
		}
		else = { define_powerful_mage = { type = ruler necromancy = 3 random = 5 lich = yes witch_king = yes war_wizard = yes } }
		
		capital_scope = {
			save_event_target_as = capital_province
		}
		every_country = {
			limit = {
				ai = no
				knows_country = ROOT
			}
			country_event = { id = great_conqueror.2 }
		}
	}
}

#Caehn (AI)
country_event = {
	id = great_conqueror.5
	title = great_conqueror.5.t
	desc = great_conqueror.5.d
	picture = CENTAUR_CAMPFIRE_eventPicture

	is_triggered_only = yes
	hidden = yes

	trigger = {
		ai = yes
		centaur_can_become_vc = yes
		has_country_flag = great_conqueror_flag
		total_development = 50
		army_size = 10
		if = {
			limit = { is_at_war = yes }
			war_score = -10									# isn't in the process of getting stomped
			all_war_enemy_countries = { ai = yes }			# isn't fighting a player
		}
	}

	option = {
		name = great_conqueror.5.a
		capital_scope = {
			save_event_target_as = capital_province
		}

		#Upgrade from Great Conqueror to Mythical Conqueror
		create_mythical_conqueror = yes
		
		export_to_variable = {
			which = ownDev
			value = total_development
			who = ROOT
		}
		log = "MYTHICAL CONQUEROR ---------------- [Root.GetName] |[Root.ownDev.GetValue]| became a Variant Mythical Conqueror in [GetYear] [GetMonth]. It's a [Root.GovernmentName] | Number of GC: [REB.nbGC.GetValue]"

		change_tag = I89
		custom_tooltip = centaur_unlock_cb_tooltip
		hidden_effect = { set_country_flag = centaur_special_cb }
		hidden_effect = { restore_country_name = yes }
		swap_non_generic_missions = yes
		remove_non_electors_emperors_from_empire_effect = yes
		change_government = tribal
		if = {
			limit = { NOT = { has_reform = I55_magocratic_caehnate } }
			custom_tooltip = great_centaur_horde_tt
			hidden_effect = { add_government_reform = great_centaur_horde }
		}
		grant_unlimited_state_action = { action = genocide }
		hidden_effect = {
			forbidden_lands_superregion = {
				limit = {
					OR = {
						region = west_forbidden_plains_region
						region = east_forbidden_plains_region
						region = ogre_valley_region
						region = yyl_moista_region
						region = serpent_gift_region
						region = nuzurbokh_region
					}
					NOT = { is_core = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
		increase_government_rank_scripted_effect = { rank = 3 }
		add_country_modifier = {
			name = centaur_unification_zeal
			duration = -1
			desc = centaur_unification_zeal_tooltip
		}
		add_prestige = 25
		add_country_modifier = {
			name = "centralization_modifier"
			duration = 7300
		}
		if = {
			limit = { ai = yes }
			add_country_modifier = {
				name = centaur_the_red_storm_ai_luck
				duration = 36500
			}
			if = { #If the Centaurs are struggling, stop spiraling
				limit = { inflation = 5 }
				add_inflation = -10
				add_treasury = 1000
			}
		}
		if = { limit = { has_country_modifier = centaur_horde_1 } remove_country_modifier = centaur_horde_1 }
		else_if = { limit = { has_country_modifier = centaur_horde_2 } remove_country_modifier = centaur_horde_2 }
		else_if = { limit = { has_country_modifier = centaur_horde_3 } remove_country_modifier = centaur_horde_3 }
		else_if = { limit = { has_country_modifier = centaur_horde_4 } remove_country_modifier = centaur_horde_4 }
		else_if = { limit = { has_country_modifier = centaur_horde_5 } remove_country_modifier = centaur_horde_5 }
		else_if = { limit = { has_country_modifier = centaur_horde_6 } remove_country_modifier = centaur_horde_6 }
		if = {
			limit = { has_country_modifier = subjugated_horde }
			remove_country_modifier = subjugated_horde
		}
		if = {
			limit = { has_country_modifier = settled_horde }
			remove_country_modifier = settled_horde
		}
		add_country_modifier = {
			name = centaur_zeal_timer
			duration = 365
			hidden = yes
		}
		if = {
			limit = { tag = I78 }
			set_variable = { centaur_zeal = 30 }
		}
		else = {
			set_variable = { centaur_zeal = 10 }
		}
		set_global_flag = crimson_caehn_formed

		country_event = { id = ideagroups.1 } #Swap Ideas

		every_country = {
			limit = {
				ai = no
				knows_country = ROOT
			}
			country_event = { id = great_conqueror.6 }
		}
	}
}

		change_tag = S90
		restore_country_name = yes
		increase_government_rank_scripted_effect = { rank = 3 }

                which = S90_exploit_count

            id = flavor_umbral_covenant.999

        	type = infantry

        	force_limit_percentage = 0.6

			limit = {
