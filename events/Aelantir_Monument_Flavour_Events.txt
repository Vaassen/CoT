country_event = { #Sacred grove all to arms count event
	id = sacred_grove_event.1
	title = sacred_grove_event.1.t
	desc = sacred_grove_event.1.d
	picture = HOUSE_OF_TRADE_eventPicture

	hidden = yes
	mean_time_to_happen = { months = 1 }

	trigger = {
		owns = 1016
		has_great_project = {
			typetype = sacred_grove
			tier = 0
		}
		NOT = { has_country_flag = blood_contract_1 }
		any_ally = {
			NOT = {
				OR = {
					is_in_war = {
						attacker_leader = ROOT
					}
					is_in_war = {
						defender_leader = ROOT
					}
				}
			}
			is_in_war = {
				participants = ROOT
			}
		}
		NOT = {
			has_country_flag = sacred_grove_war_flag
		}
	}
	option = {
		name = sacred_grove_event.1.a
		if = {
			limit = {
				NOT = {
					has_country_flag = sacred_grove_var_flag
				}
			}
			set_variable = {
				which = sacred_grove_call_to_arms
				value = 0
			}
		}
		set_country_flag = sacred_grove_var_flag
		change_variable = {
			which = sacred_grove_call_to_arms
			value = 1
		}
		set_country_flag = sacred_grove_war_flag
	}
}

country_event = { #Old Soara, clears modifiers given by monument if your subject breaks free
	id = old_soara_event.3
	title = old_soara_event.3.t
	desc = old_soara_event.3.d
	picture = HOUSE_OF_TRADE_eventPicture

	hidden = yes
	mean_time_to_happen = { months = 1 }

	trigger = {
		OR = {
			has_country_modifier = abandonned_city_overlord_1
			has_country_modifier = abandonned_city_overlord_2
			has_country_modifier = abandonned_city_overlord_3
		}
		NOT = {
			any_subject_country = {
				any_owned_province = {
					has_great_project = {
						type = old_soara
						tier = 0
					}
				}
			}
		}
	}
	option = {
		name = old_soara_event.3.a
		remove_country_modifier = abandonned_city_overlord_1
		remove_country_modifier = abandonned_city_overlord_2
		remove_country_modifier = abandonned_city_overlord_3
	}
