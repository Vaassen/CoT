country_event = { #Sacred grove all to arms count event
	id = aelantir_monument_flavour.3
	title = aelantir_monument_flavour.3.t
	desc = aelantir_monument_flavour.3.d
	picture = HOUSE_OF_TRADE_eventPicture

	hidden = yes
	mean_time_to_happen = { months = 1 }

	trigger = {
		owns = 1016
		1016 = {
			has_great_project = {
				type = sacred_grove
				tier = 0
			}
		}
		NOT = { has_country_flag = blood_contract_1 }
		any_ally = {
			NOT = {
				OR = {
					is_in_war = {
						attacker_leader = ROOT
					}
					is_in_war = {
						defender_leader = ROOT
					}
				}
			}
			is_in_war = {
				participants = ROOT
			}
		}
		NOT = {
			has_country_flag = sacred_grove_war_flag
		}
	}
	option = {
		name = aelantir_monument_flavour.3.a
		if = {
			limit = {
				NOT = {
					has_country_flag = sacred_grove_var_flag
				}
			}
		}
		set_country_flag = sacred_grove_var_flag
		change_variable = {
			which = sacred_grove_call_to_arms
			value = 1
		}
		set_country_flag = sacred_grove_war_flag
	}
}
