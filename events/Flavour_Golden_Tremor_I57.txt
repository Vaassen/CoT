province_event = { #Negeal Win Event
	id = golden_tremor_events.20
	title = golden_tremor_events.20.t
	desc = golden_tremor_events.20.desc
	picture = CONSPIRACY_eventPicture

	is_triggered_only = yes

	trigger = {
		owner = { NOT = { num_of_rebel_armies = 1 } }
	}

	option = {
		name = golden_tremor_events.20.a
		if = {
			limit = { owner = { has_country_flag = I57_enabled_stronger_negeal_flag } }
			area = {
				add_local_autonomy = -100
				cavalry = owner
				cavalry = owner
				add_province_modifier = {
					name = I57_defeated_in_negeal
					duration = -1
				}
			}
		}
		else = {
			area = {
				add_local_autonomy = -50
				add_unrest = -5
				add_nationalism = -15
				cavalry = owner
			}
		}
		owner = {
			clr_country_flag = I57_challenging_to_negeal_lvl_1_flag 
			clr_country_flag = I57_challenging_to_negeal_lvl_2_flag 
		}
		area = { set_province_flag = I57_won_negeal_flag }
	}
}
