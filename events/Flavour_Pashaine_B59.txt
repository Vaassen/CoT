			limit = { national_focus = MIL }

		}
	}

	option = { #their sons and succesors?
		name = flavour_pashaine.4.b
		set_country_flag = pashaine_vernmen_succesors
		custom_tooltip = pashaine_vernmen_prov_mod_tt
		tooltip = {
			add_province_modifier = {
				name = pashaine_successors_of_vernmen_mod
				duration = -1
			}
		}
		hidden_effect = {
			every_owned_province = {
				limit = { culture = vernmen }
				add_province_modifier = {
					name = pashaine_successors_of_vernmen_mod
					duration = -1
				}
			}
		}
		if = {
			limit = { national_focus = ADM }
			custom_tooltip = pashaine_pashainey_virtue_bonus_others_tt
			add_stability = 1
		}
	}

	option = { #or perhaps naught but neighbors
		name = flavour_pashaine.4.c
		remove_country_modifier = pashaine_careful_maneouvering
		if = {
			limit = { national_focus = DIP }
			custom_tooltip = pashaine_damerian_virtue_bonus_others_tt
			add_stability = 1
		}
	}
}

#Chancellor's Contest
country_event = {
	id = flavour_pashaine.5
	title = flavour_pashaine.5.t
	desc = flavour_pashaine.5.d
	picture = BATTLE_eventPicture
	
	is_triggered_only = yes

	major = yes
	major_trigger = { is_part_of_hre = yes }
	
	option = { #I guess we have to fold
		name = flavour_pashaine.5.a
		ai_chance = { factor = 100 }
		if = {
			limit = {
				Z41 = { exists = no } #Plumwall
			}
			hidden_effect = {
				281 = { add_core = Z41 } #Plumwall
			}
			release_with_religion_and_culture = {
				tag = Z41 #Plumwall
				religion = ROOT
				culture = east_damerian
			}
		}
		else_if = {
			limit = {
				Z41 = { is_subject_of = ROOT } #Plumwall
			}
			Z41 = { grant_independence = yes }
		}

		if = {
			limit = {
				Z40 = { exists = no } #Acromton
			}
			hidden_effect = {
				330 = { add_core = Z40 } #Acromton
			}
			release_with_religion_and_culture = {
				tag = Z40 #Acromton
				religion = ROOT
				culture = east_damerian
			}
		}
		else_if = {
			limit = {
				Z40 = { is_subject_of = ROOT } #Acromton
			}
			Z40 = { grant_independence = yes }
		}
	}

	option = { #We cannot spit on lothane legacy
		name = flavour_pashaine.5.b
		ai_chance = { factor = 0 }
		if = {
			limit = { is_emperor = yes }
			add_imperial_influence = -25
		}
		else = {
			add_stability = -1
		}
	}
}
