namespace =  cot_excavate_rewards

#Excavation rewards upon completing the construction of a building

# Give rewards
province_event = {
	id = cot_excavate_rewards.100
	title = cot_excavate_rewards.100.t
	desc = cot_excavate_rewards.100.d
	picture = CITY_DEVELOPMENT_eventPicture
	
	is_triggered_only = yes
	hidden = yes

	immediate = {
		remove_building = excavation_1
	}
	
	option = {
		name = cot_excavate_rewards.100.a
		ai_chance = {
			factor = 1
		}

		on_adm_development = yes

		random_list = {
			20 = {
				province_event = { id = cot_excavate_rewards.1 }
			}
			20 = {
				province_event = { id = cot_excavate_rewards.2 }
			}
			10 = {
				trigger = {
					has_terrain = desert
				}
				province_event = { id = cot_excavate_rewards.3 }
			}
			5 = {
				trigger = {
					trade_goods = gems
				}
				province_event = { id = cot_excavate_rewards.4 }
			}
		}
	}
}
