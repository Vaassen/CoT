namespace =  cot_excavate_rewards

#Excavation rewards upon completing the construction of a building

# Give rewards
province_event = {
	id = cot_excavate_rewards.100
	title = cot_excavate_rewards.100.t
	desc = cot_excavate_rewards.100.d
	picture = CITY_DEVELOPMENT_eventPicture
	
	is_triggered_only = yes
	hidden = yes

	immediate = {
		remove_building = excavation_1
	}
	
	option = {
		name = cot_excavate_rewards.100.a
		ai_chance = {
			factor = 1
		}

		on_adm_development = yes

		random_list = {
			20 = {
				province_event = { id = cot_excavate_rewards.1 } # A single gold coin
			}
			20 = {
				province_event = { id = cot_excavate_rewards.2 } # Old weapons
			}
			10 = {
				trigger = { 
					has_terrain = desert
					NOT = { development = 15 }
				}
				province_event = { id = cot_excavate_rewards.3 } # A desert water spring
			}
			5 = {
				trigger = { trade_goods = gems }
				province_event = { id = cot_excavate_rewards.4 } # A huge diamond
			}
			5 = {
				province_event = { id = cot_excavate_rewards.5 } # Goblin nest
			}
			10 = {
				trigger = { NOT = { has_province_modifier = cot_excavate_marble_deposit } }
				province_event = { id = cot_excavate_rewards.6 } # A marble deposit
			}
			20 = {
				province_event = { id = cot_excavate_rewards.7 } # Old scrolls
			}
			20 = {
				province_event = { id = cot_excavate_rewards.8 } # Old pottery
			}
			5 = {
				province_event = { id = cot_excavate_rewards.9 } # Ancient temple
			}
			5 = {
				province_event = { id = cot_excavate_rewards.10 } # Kobold hoard
			}
			3 = {
				trigger = {
					OR = {
						region = lencenor_region
						region = small_country_region
						region = west_dameshead_region
					}
					OR = {
						has_terrain = mountain
						has_terrain = hills
						has_terrain = cavern
						has_terrain = dwarven_hold
					}
					NOT = { trade_goods = rosemetal }
				}
				province_event = { id = cot_excavate_rewards.11 } # Rosemetal
			}
		}
	}
}

# Ancient temple
province_event = {
	id = cot_excavate_rewards.9
	title = cot_excavate_rewards.9.t
	desc = cot_excavate_rewards.9.d
	picture = ARTWORK_eventPicture
	
	is_triggered_only = yes
	
	option = {
		name = cot_excavate_rewards.9.a
		ai_chance = {
			factor = 1
		}
		if = {
			limit = { 
				OR = { 
					has_building = temple
					has_building = cathedral
				}
			}
			add_base_tax = 1
		}
		else = {
			add_building = temple
		}
	}
}
