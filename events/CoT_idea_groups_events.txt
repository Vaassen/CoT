#Artificers buff
country_event = {
	id = cot_idea_groups_events.2
	title = cot_idea_groups_events.2.t
	desc = cot_idea_groups_events.2.d
	picture = BIG_STAR_IN_SKY_eventPicture

	hidden = yes

	trigger = {
		OR = {
			AND = {
				has_idea = expanded_policies
				NOT = { has_country_modifier = cot_improved_artificers }
			}
			AND = {
				NOT = { has_idea = expanded_policies }
				has_country_modifier = cot_improved_artificers
			}
		}
	}

	option = {
		name = cot_i_events.119.a
		if = {
			limit = { has_idea = expanded_policies }
			add_country_modifier = { name = cot_improved_artificers duration = -1 }
		}
		if = {
			limit = { NOT = { has_idea = expanded_policies } }
			remove_country_modifier = cot_improved_artificers
		}
	}
}
