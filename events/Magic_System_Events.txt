country_event = { #Monthly magic upkeep
	id = magic_system.0
	title =  magic_system.0.t
	desc =  magic_system.0.d
	picture = MAGES_ESTATE_eventPicture
	
	is_triggered_only = yes
	hidden = yes

	trigger = {
		ai = no #AI has their own thing
		has_country_flag = magic_system_initialized
	}

	option = {
		name = magic_system.0.a
		ai_chance = { factor = 1 }

		update_mana_regen_vars = yes
		change_mana = { var_amount = manaRegenerationRate }

		export_to_variable = {
			which = last_months_xp_gain
			value = modifier:monthly_magic_experience_gain
		}

		set_variable = { which = max_experience_bank which = last_months_xp_gain }
		multiply_variable = { max_experience_bank = 24 } #Max XP bank is 2 years of passive gain

		set_variable = { banked_magic_xp_portion = 0 }
		
		if = {
			limit = { is_overall_advancing_magic = no }
			set_variable = { which = passive_experience_bank_change which = last_months_xp_gain }
		}
		else_if = {
			limit = {
				check_variable = { passive_experience_bank = 1 }
				is_overall_advancing_magic = yes
			}
			set_variable = { which = expFoo which = last_months_xp_gain }
			multiply_variable = { expFoo = 0.5 }

			if = { limit = { check_variable = { which = passive_experience_bank which = expFoo } }
				change_variable = { which = banked_magic_xp_portion which = expFoo }
				subtract_variable = { which = passive_experience_bank which = expFoo }
			}
			else = {
				change_variable = { which = banked_magic_xp_portion which = passive_experience_bank }
				subtract_variable = { which = passive_experience_bank which = banked_magic_xp_portion }
			}
			set_variable = { which = passive_experience_bank_change which = banked_magic_xp_portion }
			multiply_variable = { passive_experience_bank_change = -1 }
		}
		else = { set_variable = { passive_experience_bank_change = 0 } }

		if = { limit = { has_ruler_flag = magic_advancement }
			if = { limit = { is_long_lived_ruler = yes }
				divide_variable = { last_months_xp_gain = 2 }
				divide_variable = { banked_magic_xp_portion = 2 }
			}
		}
		else_if = { limit = { has_heir_flag = magic_advancement }
			if = { limit = { is_long_lived_heir = yes }
				divide_variable = { last_months_xp_gain = 2 }
				divide_variable = { banked_magic_xp_portion = 2 }
			}
		}
		else_if = { limit = { has_country_flag = magic_advancement }
			divide_variable = { which = last_months_xp_gain value = 3 }
			divide_variable = { which = banked_magic_xp_portion value = 3 }
		}

		increase_magic_experience = { var_amount = last_months_xp_gain unmodified_var_amount = banked_magic_xp_portion }

		if = { #For tooltip
			limit = { is_overall_advancing_magic = yes }
			set_variable = { which = last_months_xp_gain which = experienceGain }
		}
		else = { set_variable = { last_months_xp_gain = 0 } }

		update_magic_ui_experience_modifiers = yes
	}
}

country_event = { #Magic Advancement Events Handler (Scripted effects are borked so we use events)
	id = magic_system.100
	title = magic_system.100.t		
	desc = magic_system.100.d
	picture = BIG_BOOK_eventPicture
	
	hidden = yes
	is_triggered_only = yes

	option = { #Magic Advancement sorter
		name = finish_magic_advancement
		trigger = { magic_advancement_done = yes }
		ai_chance = { factor = 1 }

		if = { limit = { has_ruler_flag = magic_study }
			finish_magic_study = { type = ruler }
			country_event = { id = magic_study.2 }
		}
		else_if = { limit = { has_heir_flag = magic_study }
			finish_magic_study = { type = heir }
			country_event = { id = magic_study.2 } #TODO: give unique event
		}
		else_if = { limit = { has_country_flag = magic_study }
			finish_magic_study = { type = country }
			country_event = { id = magic_study.2 } #TODO: give unique event
			generic_calc_spell_levels = { type = country }
		}
		else_if = { limit = { has_country_flag = magic_project }
			country_event = { id = magic_project.0 }
		}
		else_if = { limit = { has_country_flag = magic_infrastructure }
			magic_infrastructure_level_up = yes
			country_event = { id = magic_study.2 } #TODO: give unique event
		}

		country_event = { id = magic_system.99 } #Abort happens after everything else
	}
	option = { #For events that happen while you're doing a project
		name = magic_advancement_update_event
		ai_chance = { factor = 1 }
		trigger = {
			is_advancing_project = yes
			magic_advancement_done = no
		}

		trigger_switch = {
			on_trigger = has_country_flag

			lichdom_project = { country_event = { id = magic_project_lichdom.0 } }
			homunculus_project = { country_event = { id = magic_project_homunculus.0 } }
		}
	}
	option = { #For events that happen while you're doing a study
		name = magic_study_radiant_event
		ai_chance = { factor = 1 }
		trigger = {
			is_overall_studying_magic = yes
			magic_advancement_done = no
		}
	}
	option = { #For events that happen while you're improving infrastructure..?
		name = magic_study_radiant_event
		ai_chance = { factor = 1 }
		trigger = {
			is_advancing_infrastructure = yes
			magic_advancement_done = no
		}
	}
	option = { #Catch-all, in case.
		name = magic_study_radiant_event
		ai_chance = { factor = 1 }
		trigger = {
			is_advancing_infrastructure = no
			is_advancing_project = no
			is_overall_studying_magic = no
			magic_advancement_done = no
		}
	}
}

country_event = { #AI can do advancements
	id = magic_system.350
	title = magic_system.350.t
	desc = magic_system.350.desc
	picture = MAGES_ESTATE_eventPicture

	hidden = yes
	is_triggered_only = yes

	immediate = { #give 60 months of xp
		export_to_variable = {
			which = ai_magic_foo
			value = modifier:monthly_magic_experience_gain
		}
		multiply_variable = { ai_magic_foo = 60 }
		change_variable = { which = ai_magic_experience which = ai_magic_foo }
	}

	option = { #not enough xp, keep waiting
		name = magic_system.350.a
		ai_chance = { factor = 1 }
		trigger = { NOT = { check_variable = { ai_magic_experience = 0 } } }
	}

	option = { #have enough xp, do the next advancement
		name = magic_system.350.b
		ai_chance = { factor = 1 }
		trigger = { check_variable = { ai_magic_experience = 0 } }

		generic_calc_spell_levels = { type = country }
		export_to_variable = { which = ai_magic_foo value = modifier:maximum_estate_spell_schools }

		if = { #upgrade infrastructure is top priority
			limit = {
				check_variable = { which = country_spell_schools which = ai_magic_foo }
				OR = {
					AND = {
						exactly_magical_infrastructure_level = { level = 1 }
						can_upgrade_magical_infrastructure_1 = yes
					}
					AND = {
						exactly_magical_infrastructure_level = { level = 2 }
						can_upgrade_magical_infrastructure_2 = yes
					}
					AND = {
						exactly_magical_infrastructure_level = { level = 3 }
						can_upgrade_magical_infrastructure_3 = yes
					}
					AND = {
						exactly_magical_infrastructure_level = { level = 4 }
						can_upgrade_magical_infrastructure_4 = yes

				}
			}
			ai_pay_for_xp = { infrastructure = yes }
