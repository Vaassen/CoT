					}
				}

			}
		}

		#Lose dev
		A38 = { country_event = { id = cot_frostheart_winter.3 } }
		A38 = { country_event = { id = cot_frostheart_winter.5 } }
	}

	option = {
		name = flavor_castanor_go_back
		ai_chance = { factor = 1 }
	}
}
