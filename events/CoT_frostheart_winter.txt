#Gerudia debuff
country_event = {
	id = cot_frostheart_winter.0
	title = cot_frostheart_winter.0.t
	desc = cot_frostheart_winter.0.d
	picture = BIG_BOOK_eventPicture

	fire_only_once = yes
	is_triggered_only = yes
	hidden = yes

	trigger = { 
		owns = 8
	}

	immediate = {
		set_global_flag = cot_eternal_winter_5
		every_province = {
			if = {
				limit = {
					OR = {
						superregion = western_cannor_superregion
						superregion = gerudia_superregion
						superregion = escann_superregion
						superregion = deepwoods_superregion
					}
				}
				add_province_modifier = { name = cot_eternal_winter_debuff_5 duration = -1 }
			}
		}
	}

	option = {
		name = flavor_castanor_go_back
		ai_chance = { factor = 1 }
	}
}
