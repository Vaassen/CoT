#Renovate Truces
country_event = {
	id = tsukqiao.4
	title = tsukqiao.4.t
	desc = tsukqiao.4.d
	picture = MILITARY_CAMP_eventPicture

	is_triggered_only = yes
	trigger = { }


	option = {
		name = tsukqiao.4.a

		custom_tooltip = tsukqiao_mass_truces.tt
		hidden_effect = {
			every_known_country = {
				limit = {
					capital_scope = {
						OR = {
							superregion = yanshen_superregion
							region = shuvuushudi_region
						}
						NOT = { tag = Y02 } #Jinqiu
					}
				}
				add_truce_with = Y02
				add_truce_with = ROOT
				Y02 = { add_truce_with = PREV }
				ROOT = { add_truce_with = PREV }
			}
			if = { #Event can only fire 4 times, calc if it still can fire
				limit = {
					NOT = { has_country_flag = tsukqiao_truce_use_3 }
				}
				country_event = { id = tsukqiao.4 days = 1825 }
				if = {
					limit = { has_country_flag = tsukqiao_truce_use_2 }
					set_country_flag = tsukqiao_truce_use_3
				}
				else_if = {
					limit = { has_country_flag = tsukqiao_truce_use_1 }
					set_country_flag = tsukqiao_truce_use_2
				}
				else = {
					set_country_flag = tsukqiao_truce_use_1
				}
			}
		}
	}
	option = {
		name = tsukqiao.4.b

		custom_tooltip = tsukqiao_no_more_truces.tt
	}
}

#Vision of a Loving Mother
country_event = {
	id = tsukqiao.5
	title = tsukqiao.5.t
	desc = tsukqiao.5.d
	picture = MILITARY_CAMP_eventPicture

	is_triggered_only = yes
	trigger = { }
	fire_only_once = yes


	option = {
		name = tsukqiao.5.a

		4842 = { cede_province = Z61 } #Jincheng
		4952 = { cede_province = Z61 } #Chatzin
		inherit = Y02

		custom_tooltip = tsukqiao_enable_excellent_might.tt
		hidden_effect = {
			set_country_flag = tsukqiao_excellent_might_active
			yanshen_superregion = {
				limit = { religion = devouring_path }
				set_province_flag = tsukqiao_excellent_might_target
			}
			set_country_flag = tsukqiao_independent_missions
		}
		swap_non_generic_missions = yes
		if = {
			limit = { has_ruler_modifier = exceptional_willpower_100 }
			change_dip = 1
			remove_ruler_personality = drunkard_personality
			define_heir = {
				female = yes
				dynasty = ROOT
				name = "Hejing"
				claim = 100
				age = 13
				adm = 5
				dip = 4
				mil = 4
			}
			anb_block_heir_death = { ruler = yes duration = -1 }
		}
		hidden_effect = { clr_country_flag = tsukqiao_jinqiu_war_flag }
	}
}
