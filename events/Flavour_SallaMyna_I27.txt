	}

	immediate = {
		hidden_effect = {
			every_known_country = {
				I26 = { set_variable = { which = salla_myna_var_ciranmyna_contribution value = 0 } }
				I28 = { set_variable = { which = salla_myna_var_ciranmyna_contribution value = 0 } }

				I33 = { set_variable = { which = salla_myna_var_ciranmyna_contribution value = 0 } }
				I34 = { set_variable = { which = salla_myna_var_ciranmyna_contribution value = 0 } }
			}

			every_known_country = { 
				limit = { 
					primary_culture = wood_elf
					capital_scope = { superregion = deepwoods_superregion } 
					OR = {
						is_subject_of = ROOT
						AND = {
							trust = { who = ROOT value = 75 }
							alliance_with = ROOT
						}
					}
				}

				if = { limit = { trust = { who = ROOT value = 100 } }
					add_years_of_income = -0.33
					export_to_variable = { which = salla_myna_var_ciranmyna_contribution value = trigger_value:monthly_income }
					multiply_variable = { which = salla_myna_var_ciranmyna_contribution value = 4 }
					round_variable = { which = salla_myna_var_ciranmyna_contribution value = 0 }
					ROOT = {
						export_to_variable = { which = salla_myna_var_ciranmyna_funds_increase value = trigger_value:monthly_income who = PREV }
						multiply_variable = { which = salla_myna_var_ciranmyna_funds_increase value = 4 }
						round_variable = { which = salla_myna_var_ciranmyna_funds_increase value = 0 }
						change_variable = { which = salla_myna_var_ciranmyna_funds which = salla_myna_var_ciranmyna_funds_increase }
					}
				}
				else = {
					add_years_of_income = -0.16
					export_to_variable = { which = salla_myna_var_ciranmyna_contribution value = trigger_value:monthly_income }
					multiply_variable = { which = salla_myna_var_ciranmyna_contribution value = 2 }
					round_variable = { which = salla_myna_var_ciranmyna_contribution value = 0 }
					ROOT = {
						export_to_variable = { which = salla_myna_var_ciranmyna_funds_increase value = trigger_value:monthly_income who = PREV }
						multiply_variable = { which = salla_myna_var_ciranmyna_funds_increase value = 2 }
						round_variable = { which = salla_myna_var_ciranmyna_funds_increase value = 0 }
						change_variable = { which = salla_myna_var_ciranmyna_funds which = salla_myna_var_ciranmyna_funds_increase }
					}
				}
			}

			set_variable = { which = salla_myna_var_treasury_required_1 value = 2500 }
