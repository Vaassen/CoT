		7008 = { add_permanent_province_modifier = { name = I27_overrun_with_animals_mod duration = -1 } add_devastation = 25 }

		if = { limit = { I38 = { ai = yes } }
			every_owned_province = {
				add_permanent_province_modifier = { name = I27_lack_of_animals_mod duration = -1 }
				if = { limit = { OR = { trade_goods = livestock trade_goods = fur } }
					add_permanent_province_modifier = { name = I27_little_to_hunt_mod duration = -1 }
				}
			}
		}

		ai_chance = {
			factor = 1
		}
	}
}
