#Birth of Ardimyan Culture
country_event = {
	id = ardimya.1
	title = ardimya.1.t
	desc = ardimya.1.d
	picture = COLONIZATION_eventPicture
	
	is_triggered_only = yes
	fire_only_once = yes
	
	major = yes
	major_trigger = {
		capital_scope = { continent = europe }
		num_of_colonists = 1
		any_owned_province = {
			region = ardimya_region
		}
	}
	
	trigger = { 
		capital_scope = { continent = europe }
		has_country_flag = cannorian_ardimya_colonist
		OR = {
			culture_is_human = yes
			culture_is_half_elven = yes
		}
		OR = {
			AND = {
				current_age = age_of_reformation
				num_of_owned_provinces_with = {
					value = 30
					region = ardimya_region
				}
			}
			AND = {
				current_age = age_of_absolutism
				num_of_owned_provinces_with = {
					value = 15
					region = ardimya_region
				}
			}
		}
	}
	
	immediate = {
		random_owned_province = {
			limit = {
				region = ardimya_region
				is_colony = no
				culture_is_human = yes
			}
			save_event_target_as = ardimyan_rise_province
		}
		set_global_flag = ardimya_event_chain_started
	}
	
	option = {
		name = ardimya.1.a
		event_target:ardimyan_rise_province = {
			add_province_modifier = { name = ardimya_rise_of_new_people duration = 18250 }
			change_culture = ardimyan
		}
		hidden_effect = {
			#this one is less important though 
			country_event = {
				id = ardimya.2
				days = 730
			}
			#at least 1 country should fire this event to continue event chain.
			every_country = {
				limit = {
					has_country_flag = cannorian_ardimya_colonist
				}
				country_event = {
					id = ardimya.3
					days = 7300
				}
			}

		}
	}
}

#Choose a tag to create ardimya
country_event = {
	id = ardimya.3
	title = ardimya.3.t
	desc = ardimya.3.d
	picture = COLONIZATION_eventPicture
	hidden = yes
	
	is_triggered_only = yes
	fire_only_once = yes
	trigger = { 
		always = yes
	}
	
	
	option = {
		name = ardimya.3.a
		

		
		every_country = {
			limit = {
				capital_scope = { continent = europe }
				has_country_flag = cannorian_ardimya_colonist
			}
			if = {
				limit = { NOT = { has_saved_event_target = ardimya_overlord } }
				save_event_target_as = ardimya_overlord
			}

			every_owned_province = {
				limit = {
					region = ardimya_region
				}
				PREV = {
					change_variable = {
						which = ardimya_provinces
						value = 1
					}				
				}
			}
			if = {
				limit = {
					check_variable = {
						which = ardimya_provinces
						which = event_target:ardimya_overlord
					}
				}
				save_event_target_as = ardimya_overlord
			}			
		}
		
		event_target:ardimya_overlord = {
			country_event = {
				id = ardimya.5
			}
		}
	}
}

	desc = ardimya.6.d
	picture = REVOLT_HUNGARIAN_eventPicture

	is_triggered_only = yes
	trigger = { 
		capital_scope = { continent = europe }
		S91 = {
			OR = {
				AND = {
					exists = yes
					is_subject_of = ROOT
				}
				exists = no
			}
		}
	}
	option = {
		name = ardimya.6.a
		set_country_flag = ardimya_war_of_independence_colonial_power

			limit = {
				owned_by = ROOT
			}
			add_local_autonomy = -60
			remove_province_modifier = restless_ardimyans
			cede_province = S91
			add_core = S91
		}
		if = {
			limit = {
				NOT = { S91 = { is_subject_of = ROOT } }
			}

				limit = { has_dlc = "Leviathan" }
				create_subject = {
					subject_type = crown_colony

				}

				create_subject = {
					subject_type = colony

country_event = {
	id = ardimya.7
	title = ardimya.7.t
	desc = ardimya.7.d
	picture = REVOLT_HUNGARIAN_eventPicture
	is_triggered_only = yes

	option = {
		name = ardimya.7.a		
		set_country_flag = ardimya_war_of_independence
		
		add_country_modifier = {
			name = ardimya_independence_war
			duration = -1
		}
		add_country_modifier = {
			name = ardimya_independence_war_income
			duration = 18250
		}
		random_owned_province = {
			limit  = {
				has_port = yes
			}
			build_to_forcelimit = {
				infantry = 0.65
				cavalry = 0.15
				artillery = 0.2
				heavy_ship = 0.4
				light_ship = 0.4
			}
		}
		while = {
			limit = {
				NOT = { manpower_percentage = 0.99 }
			}
			add_manpower = 5
		}
		declare_war_with_cb = {
			casus_belli = cb_independence_war
			who = overlord
		}
		custom_tooltip = ardimya_independece_timelimit_tt
		hidden_effect = {
			country_event = {
				id = ardimya.11
				days = 3650
			}
		}

	}

}
