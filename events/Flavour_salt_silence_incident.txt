country_event = { # Echo of Authority
	id = flavor_salt_silence_incident.4
	title = flavor_salt_silence_incident.4.t
	desc = flavor_salt_silence_incident.4.d
	picture = BYZ_FORT_eventPicture

	is_triggered_only = yes
    major = yes
    major_trigger = { 
        is_part_of_hre = yes
    }
	
	option = {		
		name = flavor_salt_silence_incident.4.a
		ai_chance = { factor = 75 }

        if = {
            limit = { is_part_of_hre = yes }
            set_in_empire = no
            set_government_rank_scripted_effect = { rank = 2 }
            the_borders_region = {
                add_permanent_claim = ROOT
            }
            add_country_modifier = {
                name = iiss_the_bold
				duration = 9125
            }
        } 
        else = {
            every_subject_country = {
                limit = { 
                    is_subject_of_type = personal_union 
                    is_part_of_hre = yes
                }
                ROOT = { inherit = PREV }
            }
            add_country_modifier = {
                name = iiss_the_brash
				duration = 7300
            }
            add_country_modifier = {
                name = iiss_the_arrogant
				duration = 14600
            }
        }
        emperor = {
            add_casus_belli = {
                target = event_target:salt_and_silence_giant
                type = cb_insult
                months = 120
            }
        }
        set_global_flag = war_of_salt_and_silence
        every_known_country = {
            limit = {
                is_part_of_hre = yes
                capital_scope = { region = the_borders_region }
                NOT = { has_opinion = { who = emperor value = 150 } }
                is_emperor = no
            }
            set_country_flag = rebelling_salt_prince
            set_in_empire = no
        }
	}

    option = {		
		name = flavor_salt_silence_incident.4.b
		ai_chance = { factor = 25 }

        if = {
            limit = { is_part_of_hre = yes }
            add_legitimacy = -25
            add_prestige = -25
        }
        else = {
            every_subject_country = {
                limit = { 
                    is_subject_of_type = personal_union 
                    is_part_of_hre = yes
                }
                grant_independence = yes
            }
            add_country_modifier = {
                name = iiss_the_peaceful
				duration = 7300
            }
            add_country_modifier = {
                name = iiss_the_regal
				duration = 14600
            }
            
        }
        clear_global_event_target = salt_and_silence_giant	
	}
}
