					}

						value = 8

			}
			add_country_modifier = {
				name = rezankand_dispelled_darkness_modifier
				duration = -1
			}
			every_country = {
				limit = {
					capital_scope = {
						superregion = taychend_superregion
					}
					NOT = { religion = oren_nayiru }
				}
				add_country_modifier = {
					name = rezankand_dispel_the_dark_modifier

				}
			}

				limit = {
					previous_owner = ROOT
				}
				random_list = {
					1 = { change_religion = oren_nayiru }
					1 = {}
				}
			}
			hidden_effect = {
				change_variable = { #Bonus 20 dev or 4 more colonizable provinces
					which = rezankand_slave_provinces
					value = 2
				}
			}
		}

		if = {
			limit = {

			}

				while = {
					limit = {
						check_variable = {
							which = rezankand_slave_provinces
							value = 1
						}
					}
					for = {
						amount = 10
						effect = " 
						random_owned_province = {
							limit = {
								region = dry_coast_region
								owned_by = ROOT
							}
							random_list = {
								1 = { add_base_tax = 1 }
								1 = { add_base_production = 1 }
								1 = { add_base_manpower = 1 }
							}
						}"
					}
					change_variable = {
						which = rezankand_slave_provinces

					}
				}
			}
		}
		else = {
			while = {
				limit = {
					check_variable = {
						which = rezankand_slave_provinces
						value = 1
					}
				}

					dry_coast_region = {
