		goto = 6068
		name = flavour_AYN.401.a
		custom_tooltip = S88_island_qeyikul_move_and_fight_tt
		hidden_effect = {
			6068 = { cede_province = ROOT move_capital_effect = yes }
			add_core = 6068
			event_target:S85_old_capital = {
				while = {
					limit = {
						num_of_units_in_province = {
								amount = 1
								type = infantry
								who = ROOT
							}
						}
					kill_units = { who = ROOT amount = 1 type = infantry }
					6068 = { spawn_units_for = { amount = 1 who = ROOT type = infantry } }
				}
				while = {
					limit = {
						num_of_units_in_province = {
								amount = 1
								type = cavalry
								who = ROOT
							}
						}
					kill_units = { who = ROOT amount = 1 type = cavalry }

				while = {

							}
