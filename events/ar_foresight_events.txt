namespace = ar_foresight

country_event = {
    id = ar_foresight.10
    title = ar_foresight.fort.t
    desc = ar_foresight.fort.d
    picture = CONQUEST_eventPicture
    is_triggered_only = yes
    trigger = { has_country_flag = ar_fs_fort_menu_open }
    option = {
        name = ar_foresight.fort.confirm
        trigger = {
            ar_fs_can_spend = yes
            event_target:ar_fs_target_fort = { ar_fs_fort_candidate = yes }
        }
        ai_chance = { factor = 100 }
        custom_tooltip = ar_fs_fort_reward_tt
        hidden_effect = { ar_fs_buy_selected_fort = yes }
    }
    option = {
        name = ar_foresight.fort.next
        trigger = {
            ar_fs_can_spend = yes
            any_province = {
                ar_fs_fort_candidate = yes
                NOT = { province_id = event_target:ar_fs_target_fort }
            }
        }
        ai_chance = { factor = 0 }
        hidden_effect = {
            ar_fs_select_next_fort = yes
            country_event = { id = ar_foresight.11 }
        }
    }
    option = {
        name = ar_foresight.fort.cancel
        ai_chance = { factor = 1 }
        hidden_effect = { ar_fs_close_fort_menu = yes }
    }
}
