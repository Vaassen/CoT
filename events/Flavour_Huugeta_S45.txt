# Mercenary Enterprise System: End of Cycle Event
country_event = {
	id = huugeta.1 
	title = huugeta.1.t
	desc = huugeta.1.d
	picture = VIKING_KING_eventPicture
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			clr_country_flag = S45_cycle_mil_advisor_hired

			set_variable = {
				which = mercenaryIncome
				which = mercenaryMomentum
			}
			multiply_variable = {
				which = mercenaryIncome
				value = 10
			}

			set_variable = {
				which = mercenaryTradition
				which = mercenaryMomentum
			}
			multiply_variable = {
				which = mercenaryTradition
				value = 0.25
			}

			set_variable = {
				which = mercenaryProfessionalism
				which = mercenaryMomentum
			}
			multiply_variable = {
				which = mercenaryProfessionalism
				value = 0.001
			}
		}
	}

	option = {
		name = huugeta.1.a
		S45_upgrade_mercenary_excellence = yes

		hidden_effect = {
			var_effect = { effect = add_treasury var = mercenaryIncome sign = + }
		}

		custom_tooltip = S45_mercenary_enterprise_reward_crowns_tt
	}

	option = {
		name = huugeta.1.b
		S45_upgrade_mercenary_excellence = yes

		hidden_effect = {
			var_effect_granular = { effect = add_army_tradition var = mercenaryTradition sign = + }
		}

		custom_tooltip = S45_mercenary_enterprise_reward_army_tradition_tt
	}

	option = {
		name = huugeta.1.c
		S45_upgrade_mercenary_excellence = yes
		custom_tooltip = S45_mercenary_enterprise_reward_army_professionalism_tt

		hidden_effect = {
			var_effect_granular = { effect = add_army_professionalism var = mercenaryProfessionalism sign = + }
		}

		custom_tooltip = S45_mercenary_enterprise_reward_army_professionalism_tt
	}

	after = {
		hidden_effect = {

			# Reset per-action counters
			set_variable = {
				which = mercActionValue_building
				value = 0
			}
			set_variable = {
				which = mercActionValue_manufactory
				value = 0
			}
			set_variable = {
				which = mercActionValue_dev
				value = 0
			}
			set_variable = {
				which = mercActionValue_condottieri
				value = 0
			}
			set_variable = {
				which = mercActionValue_mercenaries
				value = 0
			}
			set_variable = {
				which = mercActionValue_favors
				value = 0
			}
			set_variable = {
				which = mercActionValue_advisor
				value = 0
			}
			
			# Set momentum to base value
			set_variable = {
				which = mercenaryMomentum
				which = baseMercenaryMomentum
			}

			country_event = {
				id = huugeta.1
				days = 3650
			}
		}
	}

}
