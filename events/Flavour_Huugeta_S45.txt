# Poison Their Hearts, fired from declarewar diplomatic action
country_event = {
	id = huugeta.2 
	title = huugeta.2.t
	desc = huugeta.2.d
	picture = VIKING_KING_eventPicture
	is_triggered_only = yes

	option = {
		name = huugeta.2.a
		trigger = {
			event_target:S45_poison_their_hearts_target = { 
				OR = {
					NOT = { ruler_has_max_personalities = yes }
					ruler_has_personality = inspiring_leader_personality
				}
			}
		}

		event_target:S45_poison_their_hearts_target = {
			if = {
				limit = {
					ruler_has_personality = inspiring_leader_personality
				}
				remove_ruler_personality = inspiring_leader_personality
			}

			add_ruler_personality = craven_personality
		}


		add_spy_network_in = {
			who = event_target:S45_poison_their_hearts_target
			value = -30
		}
	}

	option = {
		name = huugeta.2.b

		trigger = {
			event_target:S45_poison_their_hearts_target = {
				has_leaders = {
					value = 1
					include_monarch = no
					include_heir = no
				}
			}

		}

		event_target:S45_poison_their_hearts_target = {
			kill_leader = {
				type = random
			}
		}
		add_spy_network_in = {
			who = event_target:S45_poison_their_hearts_target
			value = -30
		}
	}

	option = {
		name = huugeta.2.c
		
		event_target:S45_poison_their_hearts_target = {
			add_country_modifier = {
				name = S45_poisoned_hearts
				duration = 1825
			}
		}
		add_spy_network_in = {
			who = event_target:S45_poison_their_hearts_target
			value = -30
		}
	}
}

# The New Empire - Capstone Event
country_event = {
	id = huugeta.4 
	title = huugeta.4.t
	desc = huugeta.4.d
	picture = VIKING_KING_eventPicture
	is_triggered_only = yes

	option = {
		name = huugeta.4.a
		
		if = {
			limit = {
				check_variable = {
					which = officerPower
					value = 1
				}
			}
			custom_tooltip = S45_free_company_rosters_upgrade_tt
			set_country_flag = remove_absolutism_from_free_company_rosters
		}

		if = {
			limit = {
				check_variable = {
					which = officerPower
					value = 2
				}
			}
			custom_tooltip = S45_unlock_plutocratic_ideas_tt
			set_country_flag = unlocked_plutocracy_gov_ideas
		}

		if = {
			limit = {
				check_variable = {
					which = officerPower
					value = 3
				}
			}
			add_country_modifier = {
				name = S45_meritocratic_advancement
				duration = -1
			}
		}

		if = {
			limit = {
				check_variable = {
					which = officerPower
					value = 4
				}
			}
			change_variable = {
				which = baseMercenaryMomentum
				value = 25
			}
		}

		if = {
			limit = {
				check_variable = {
					which = officerPower
					value = 5
				}
			}
			add_country_modifier = {
				name = S45_entrenched_officer_class
				duration = -1
			}
		}
	}
}
