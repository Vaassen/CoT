namespace = cot_golden_era

country_event = {
	id = cot_golden_era.0
	title = cot_golden_era.0.t
	desc = cot_golden_era.0.d
	picture = BIG_BOOK_eventPicture

	hidden = yes

	trigger = {
		OR = {
			AND = {
				current_age = age_of_reformation
				in_golden_age = no
				NOT = { has_country_flag = cot_had_golden_age_reset_2 }
			}
			AND = {
				current_age = age_of_absolutism
				in_golden_age = no
				NOT = { has_country_flag = cot_had_golden_age_reset_3 }
			}
			AND = {
				current_age = age_of_revolutions
				in_golden_age = no
				NOT = { has_country_flag = cot_had_golden_age_reset_4 }
			}
		}
	}

	immediate = {
		reset_golden_age = yes
		if = { limit = { current_age = age_of_reformation } set_country_flag = cot_had_golden_age_reset_2 }
		if = { limit = { current_age = age_of_absolutism } set_country_flag = cot_had_golden_age_reset_3 }
		if = { limit = { current_age = age_of_revolutions } set_country_flag = cot_had_golden_age_reset_4 }
	}

	option = {
		name = cot_golden_era.0.a
	}
}

#Choose golden decate type
country_event = {
	id = cot_golden_era.1
	title = cot_golden_era.1.t
	desc = cot_golden_era.1.d
	picture = BIG_BOOK_eventPicture

	trigger = {
		OR = {
			AND = {
				current_age = age_of_discovery
				in_golden_age = yes
				NOT = { has_country_flag = cot_had_golden_age_1 }
			}
			AND = {
				current_age = age_of_reformation
				in_golden_age = yes
				NOT = { has_country_flag = cot_had_golden_age_2 }
			}
			AND = {
				current_age = age_of_absolutism
				in_golden_age = yes
				NOT = { has_country_flag = cot_had_golden_age_3 }
			}
			AND = {
				current_age = age_of_revolutions
				in_golden_age = yes
				NOT = { has_country_flag = cot_had_golden_age_4 }
			}
		}
	}

	immediate = {
		if = {
			limit = {
				has_country_flag = cot_had_golden_age_adm
				has_country_flag = cot_had_golden_age_dip
				has_country_flag = cot_had_golden_age_mil
			}
			clr_country_flag = cot_had_golden_age_adm
			clr_country_flag = cot_had_golden_age_dip
			clr_country_flag = cot_had_golden_age_mil
		}
		if = { limit = { current_age = age_of_discovery } set_country_flag = cot_had_golden_age_1 }
		if = { limit = { current_age = age_of_reformation } set_country_flag = cot_had_golden_age_2 }
		if = { limit = { current_age = age_of_absolutism } set_country_flag = cot_had_golden_age_3 }
		if = { limit = { current_age = age_of_revolutions } set_country_flag = cot_had_golden_age_4 }
	}

	option = { # Adm
		name = cot_golden_era.1.a
		trigger = { NOT = { has_country_flag = cot_had_golden_age_adm } }

		add_country_modifier = {
			name = golden_era_adm
			duration = 3650
		}

		random_list = {
			1 = { country_event = { id = cot_golden_era.3 days = 10 } }
			1 = { trigger = { government = monarchy } country_event = { id = cot_golden_era.4 days = 10 } }
			1 = { trigger = { any_owned_province = { has_port = yes } } country_event = { id = cot_golden_era.5 days = 10 } }
			1 = { country_event = { id = cot_golden_era.6 days = 10 } }
			1 = { country_event = { id = cot_golden_era.7 days = 10 } }
		}

		set_country_flag = cot_had_golden_age_adm
	}

	option = { # Dip
		name = cot_golden_era.1.b
		trigger = { NOT = { has_country_flag = cot_had_golden_age_dip } }

		add_country_modifier = {
			name = golden_era_dip
			duration = 3650
		}

		random_list = {
			1 = { country_event = { id = cot_golden_era.8 days = 10 } }
			1 = { trigger = { government = monarchy } country_event = { id = cot_golden_era.9 days = 10 } }
			1 = { trigger = { any_owned_province = { has_port = yes } } country_event = { id = cot_golden_era.10 days = 10 } }
			1 = { country_event = { id = cot_golden_era.11 days = 10 } }
			1 = { country_event = { id = cot_golden_era.12 days = 10 } }
		}

		set_country_flag = cot_had_golden_age_dip
	}

	option = { # Mil
		name = cot_golden_era.1.c
		trigger = { NOT = { has_country_flag = cot_had_golden_age_mil } }

		add_country_modifier = {
			name = golden_era_mil
			duration = 3650
		}

		random_list = {
			1 = { country_event = { id = cot_golden_era.13 days = 10 } }
			1 = { trigger = { government = monarchy } country_event = { id = cot_golden_era.14 days = 10 } }
			1 = { country_event = { id = cot_golden_era.15 days = 10 } }
			1 = { country_event = { id = cot_golden_era.16 days = 10 } }
			1 = { country_event = { id = cot_golden_era.17 days = 10 } }
		}
		set_country_flag = cot_had_golden_age_mil
	}
}
