		tag = G87
		has_country_flag = chaqwai_marriages_unlocked

		any_owned_province = {
			has_province_modifier = chaqway_markets
			NOT = { has_province_modifier = chaqway_marriage_fair }
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {

					NOT = { has_province_modifier = chaqway_marriage_fair }
				}
				save_event_target_as = marriage_province
			}
		}
	}

	option = {
		name = chaqway.6.a
		goto = marriage_province
		hidden_effect = {
			every_owned_province = { remove_province_modifier = chaqway_marriage_fair }
		}
		event_target:marriage_province = {
			add_province_modifier = {
				name = chaqway_marriage_fair
				duration = -1
			}
		}	
	}
}

country_event = {
	id = chaqway.12
	title = chaqway.12.t
	desc = chaqway.12.d
	picture = northamericagfx_SHIP_SAILING_eventPicture

	is_triggered_only = yes

	immediate = {
		random_known_country = {
			limit = {
				has_80_opinion_improved_relation = { target = ROOT }
				culture_is_ruinborn = no
				OR = {
					num_of_owned_provinces_with = {	value = 8 }
					capital_scope = { continent = europe }
				}
			}
			save_event_target_as = colonialist_scum
		}
	}

	option = {
		name = chaqway.12.a
		remove_country_modifier = chaqway_migrating
		add_country_modifier = {
			name = chaqway_fury
			duration = 18250
		}
		event_target:colonialist_scum = {
			if = {
				limit = { alliance_with = ROOT }
				break_alliance = ROOT
			}
			if = {
				limit = { marriage_with = ROOT }
				break_marriage = ROOT
			}
			add_historical_rival_mutual = { tag = ROOT }
		}
		every_known_country = {
			limit = {
				culture_is_ruinborn = no
				any_owned_province = {
					continent = north_america
					continent = south_america
				}
			}
			reverse_add_opinion = {
				who = ROOT
				modifier = chaqway_fury_opinion
			}
		}
	}
}
