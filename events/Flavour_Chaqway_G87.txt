country_event = {
	id = chaqway.6
	title = chaqway.6.t
	desc = chaqway.6.d
	picture = northamericagfx_SHIP_SAILING_eventPicture

	is_triggered_only = yes

	trigger = {
		tag = G87
		has_country_flag = chaqwai_marriages_unlocked
		NOT = { has_country_modifier = chaqwai_force_capital }
		any_owned_province = {
			has_province_modifier = chaqway_markets
			is_capital = no
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					has_province_modifier = chaqway_markets
					is_capital = no
				}
				save_event_target_as = new_capital
			}
		}
	}

	option = {
		name = chaqway.6.a
		goto = new_capital
		hidden_effect = {
			capital_scope = { remove_province_modifier = chaqway_marriage_fair }
		}
		set_capital = event_target:new_capital
		capital_scope = {
			add_province_modifier = {
				name = chaqway_marriage_fair
				duration = -1
			}
		}	
	}
}
