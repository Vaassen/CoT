# 3. Overlord accept (Fires back for W46 - The Subject. ROOT = W46, FROM = Overlord)
country_event = {
	id = flavour_vrak_dromaz.5
	title = flavour_vrak_dromaz.5.t
	desc = flavour_vrak_dromaz.5.d
	picture = DIPLOMACY_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	
	option = {
		name = flavour_vrak_dromaz.5.a
		
		add_treasury = -100
		
		FROM = {
			add_treasury = 100
		}
		
		573 = {
			if = {
				limit = {
					owned_by = FROM      # Owned by the Overlord
					controlled_by = FROM # Controlled by the Overlord
				}
				cede_province = ROOT    # Given to W46
				add_core = ROOT         # W46 gets a core
				remove_core = FROM      # Overlord loses their core
			}
		}
	}
}
