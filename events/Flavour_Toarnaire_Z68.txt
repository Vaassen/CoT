# A Taste of Blood
country_event = {
	id = flavor_toarnaire.3
	title = flavor_toarnaire.3.t
	desc = flavor_toarnaire.3.d
	picture = DIPLOMACY_eventPicture
	
	is_triggered_only = yes
	
	trigger = {
	}
	option = {
		name = flavor_toarnaire.3.c
		if = {
			limit = {
				diplomatic_reputation = 5
				navy_size = A16
				reverse_has_opinion = {
					who = A16
					value = 200
				}
			}
		
			A16 = { free_vassal = Z68 }
			vassalize = A16
			add_government_reform = presidential_despot_reform
			33 = {
				add_permanent_province_modifier = { 
					name = toarnaire_many_thorns
					duration = -1
					desc = toarnaire_sprawl_tooltip
				}
			}
			101 = { anti_tax_rebels = 1 }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						diplomatic_reputation = 5
						navy_size = A16
						NOT = {
							reverse_has_opinion = {
								who = A16
								value = 200
							}
						}
					}
					AND = {
						NOT = { diplomatic_reputation = 5 }
						navy_size = A16
					}
				}
			}
			random_list = {
				90 = {
					33 = {
						add_permanent_province_modifier = { 
							name = toarnaire_many_thorns
							duration = -1  
							desc = toarnaire_sprawl_tooltip
						}
					}
					add_government_reform = presidential_despot_reform
					A16 = { free_vassal = Z68 }
					declare_war_with_cb = {
						who = A16
						casus_belli = cb_vassalize_mission
					}
				}
				10 = {
					33 = {
						add_permanent_province_modifier = { 
							name = toarnaire_many_thorns
							duration = -1  
							desc = toarnaire_sprawl_tooltip
						}
					}
					add_government_reform = presidential_despot_reform
					A16 = { free_vassal = Z68 }
					vassalize = A16
					101 = { anti_tax_rebels = 1 }
				}
			}
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						diplomatic_reputation = 5
						NOT = { navy_size = A16 }
					}
					NOT = { diplomatic_reputation = 5 }
				}
			}
			random_list = {
				90 = {
					33 = {
						add_permanent_province_modifier = { 
							name = toarnaire_many_thorns
							duration = -1  
							desc = toarnaire_sprawl_tooltip
						}
					}
					add_government_reform = presidential_despot_reform
					A16 = { free_vassal = Z68 }
					add_casus_belli = {
						target = A16
						type = cb_vassalize_mission
						months = 120
					}
				}
				10 = {
					33 = {
						add_permanent_province_modifier = { 
							name = toarnaire_many_thorns
							duration = -1  
							desc = toarnaire_sprawl_tooltip
						}
					}
					add_government_reform = presidential_despot_reform
					A16 = { free_vassal = Z68 }
					vassalize = A16
					101 = { anti_tax_rebels = 1 }
				}
			}
		}
	}
}
