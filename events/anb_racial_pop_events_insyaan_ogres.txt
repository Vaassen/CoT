# This is an event file for ogre-related content.
namespace = insyaan_ogre_flavour_events
country_event = {
	id = insyaan_ogre_flavour_events.1
	title = insyaan_ogre_flavour_events.1.t
	desc = insyaan_ogre_flavour_events.1.d
	picture = FUNGRUM_eventPicture
	is_triggered_only = yes
	
	trigger = {
		any_owned_province = {
			region = coltkae_region
		}
		culture_is_human = yes
	}

	# Option A: Investigative Approach (Tolerant)
	option = {
		name = insyaan_ogre_flavour_events.1.a
		ai_chance = { factor = 10 }
		
		add_treasury = -50
		add_estate_influence_modifier = {
			desc = "Investigating the Plateau"
			estate = estate_adventurers 
			influence = 5
			duration = 1825 # 5 years
		}
		
		country_event = { # Triggers "A Meticulous Garden"
			id = insyaan_ogre_flavour_events.2
			days = 90
		}
	}
	
	# Option B: What use are a bunch of plants? (Indifferent Approach)
	option = {
		name = insyaan_ogre_flavour_events.1.b
		
		small_decrease_of_ogre_tolerance_effect = yes
		any_country = {
			limit = {
				region = great_plateau_region
			}
			any_owned_province = {
				add_province_modifier = {
					name = insyaan_autonomous_ogre_farms #leaves the ogres alone, so random province in the great plateau region gets positive modifier
					duration = 1825 # 5 years
				}	
			}
		}
	}		
	
	# Option C: Greenthumb (Extra Tolerant Approach)
	trigger = {
		OR = {
			ruler_has_personality = architectural_visionary_personality 
			full_idea_group = humanist_ideas
			}
		}
		option = {
			name = insyaan_ogre_flavour_events.1.c
			ai_chance = { factor = 10 }
			highlight = yes
				add_treasury = -100
				add_estate_influence_modifier = {
					desc = "Investigating the Plateau"
					estate = estate_adventurers 
					influence = 5
					duration = 1825 # 5 years
				}
				random_country = {
					limit = {
						region = great_plateau_region
					}
					any_owned_province = {
						add_province_modifier = {
							name = harvest_of_the_hunger_surpressing_root #trading more money for more tolerance
							duration = 5475 # 15 years
						}
					}
				}
				medium_increase_of_ogre_tolerance_effect = yes 	
				country_event = { # Triggers "A Meticulous Garden"
					id = insyaan_ogre_flavour_events.2
					days = 90
				}
			}
	
	# Option D: Ogre Farms Used as Mulch (The "Malevolent/Cruel" route)
	trigger = {
			OR = {
				ruler_has_personality = malevolent_personality
				ruler_has_personality = cruel_personality
			}
		option = {
			name = insyaan_ogre_flavour_events.1.e
			ai_chance = { factor = 5 }
			highlight = yes
			every_owned_province = {
				limit = {
					OR = {
					trade_goods = grain
					trade_goods = wine
					}
				}
				add_province_modifier = {
					name = insyaan_ogre_farms_used_as_mulch
					duration = 3650 # 10 years
					}
				}
			}
		}

	# Option E: Sending the Tax Man (Greedy Approach)
	trigger = {
			OR = {
				ruler_has_personality = embezzler_personality
				ruler_has_personality = greedy_personality
					}
		option = {
			name = insyaan_ogre_flavour_events.1.f
			ai_chance = { factor = 25}
			highlight = yes
			random_owned_province = {
				limit = {
					region = coltkae_region
					}
					add_province_modifier = {
						name = ogre_tax_farms
						duration = 3650 # 10 years
					}
			}	
		small_decrease_of_ogre_tolerance_effect
		add_treasury = 50
			}
		}
	}

country_event = {
	id = insyaan_ogre_flavour_events.2
	title = insyaan_ogre_flavour_events.2.t
	desc = insyaan_ogre_flavour_events.2.d
	picture = FEAST_eventPicture
	is_triggered_only = yes

	#Option A: Distribute the crops
	option = {
		name = insyaan_ogre_flavour_events.2.a
		ai_chance = { factor = 10 }
		any_owned_province = {
			add_province_modifier = {
				name = skyfall_cornucopia
				days = 3650 # 10 years
			}
		}
		any_owned_province = {
			limit = {
				NOT = {
					has_province_modifier = skyfall_cornucopia
				}
			}
			add_province_modifier = {
				name = skyfall_cornucopia
				days = 3650 # 10 years
			}
		}		
		any_owned_province = {
			limit = {
				NOT = {
					has_province_modifier = skyfall_cornucopia
				}
			}
			add_province_modifier = {
				name = skyfall_cornucopia
				days = 3650 # 10 years
			}
		}
		small_increase_of_ogre_tolerance_effect = yes
	}

	#Option B: The Royal Banquet
	option = {
		name = insyaan_ogre_flavour_events.2.b
		ai_chance = { factor = 15 }
		add_country_modifier = {
			name = the_great_feast
			duration = 1825 # 5 years
		}
		add_treasury = -100
			add_estate_loyalty_modifier = {
				desc = "The Royal Banquet"
				estate = estate_adventurers 
				loyalty = 5
				duration = 1825 # 5 years
			}
	}

	#Option C: Feed The Guards
	option = {
		name = insyaan_ogre_flavour_events.2.c
		ai_chance = { factor = 20 }

		if = {
			limit = {
				OR = {
					ruler_has_personality = strict_personality
					ruler_has_personality = inspiring_leader_personality
				}
			}
		}
		random_owned_province = {
			limit = {
				any_owned_province = {
				NOT = {
					has_province_modifier = well_fed_guards
					}
				}
				has_fort = yes
			}
			add_province_modifier = {
				name = well_fed_guards
			}
			add_ogre_minority_size_effect = yes
		}
		any_owned_province = {
			limit = {
				NOT = {
					has_province_modifier = well_fed_guards
				}
				has_fort = yes
			}
			add_province_modifier = {
				name = well_fed_guards
			}
			add_ogre_minority_size_effect = yes
		}
		any_owned_province = {
			limit = {
				NOT = {
					has_province_modifier = well_fed_guards
				}
				has_fort = yes
			}
			add_province_modifier = {
				name = well_fed_guards
			}
			add_ogre_minority_size_effect = yes
		}
		small_increase_of_ogre_tolerance_effect
	}
	#Option D: Why Shouldn't I Keep it?
	option = {
			name = insyaan_ogre_flavour_events.2.e
			ai_chance = { factor = 20 }

			if = {
				limit = {
					OR = {
						ruler_has_personality = greedy_personality
						ruler_has_personality = embezzler_personality
					}
				}
			}
			random_owned_province = {
				limit = {
					trade_goods = grain
					trade_goods = wine
				}
				add_years_of_owned_provinces_production_income = {
					days = 1095 # 3 years
				}
			}
			add_country_modifier = {
				name = confiscated_crops
				days = 1825 # 5 years
			}
			add_estate_loyalty_modifier = {
				desc = "Investigating the Plateau"
				estate = estate_adventurers 
				loyalty = 5
				duration = 1825 # 5 years
			}
	}
	#Option E: Misunderstood the Ogres 
	option = {
			name = insyaan_ogre_flavour_events.2.f
			ai_chance = { factor = 25 }

			if = {
				limit = {
					OR = {
						ruler_has_personality = kind_hearted_personality
						ruler_has_personality = naive_enthusiast_personality
						ruler_has_personality = well_connected_personality
						full_idea_group = humanist_ideas
					}
				}
			}
			add_country_modifier = {
				name = encouraged_ogre_emigration
				days = 3650 # 10 years
			}
			any_owned_province = {
				limit = {
					fast_has_pop_of_race = {
						race = ogre
					}
					region = coltkae_region
				}
			add_ogre_minority_size_effect = yes				
		}
		major_increase_of_ogre_tolerance_effect
	}
}
