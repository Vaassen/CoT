#Re'uyel response Irrliam vassal
country_event = {
	id = irrliam_flavour.4
	title = irrliam_flavour.4.t
	desc = irrliam_flavour.4.d
	picture = STREET_SPEECH_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	
	trigger = {
		
	}
	
	option = {
		name = "irrliam_flavour.4.a"
		if = {
			limit = {
				has_country_flag = f37_cot_vassalize_flag
			}
			create_subject = {
				subject_type = cot_phoenix_governorship
				subject = F26
			}
			U18 = {
				add_liberty_desire = -20
			}
			every_known_country = {
				reverse_add_aggressive_expansion = {
					who = F37
					value = 10
					apply_calc = yes
				}
			}
			clr_country_flag = f37_cot_vassalize_flag
		}
		else = {
			add_dip_power = 100
			add_prestige = -10
		}
	}
}

#Fate of the Gelkari

	}
	
	option = {
		name = "irrliam_flavour.5.a" #They are but one among many
		add_legitimacy = 10
		add_prestige = 10
		F26 = { add_liberty_desire = 50 }
	}
	
	option = {
		name = "irrliam_flavour.5.b" #The proud Gelkari shall govern the harpy hills
		add_war_exhaustion = -2
		F26 = {
			set_country_flag = cot_phoenix_governorship_flag
			add_liberty_desire = -20
		}
		create_subject = {
			subject_type = cot_phoenix_governorship
			subject = F26
		}
		hidden_effect = {
			harpy_hills_region = {
				limit = { country_or_non_sovereign_subject_holds = ROOT }
				cede_province = F26
				add_permanent_claim = F26
				remove_core = ROOT
			}
		}
	}
}
