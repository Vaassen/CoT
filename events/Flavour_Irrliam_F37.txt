#Kuzaram response Irrliam vassal
country_event = {
	id = irrliam_flavour.4
	title = irrliam_flavour.4.t
	desc = irrliam_flavour.4.d
	picture = STREET_SPEECH_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	
	trigger = {
		
	}
	
	option = {
		name = "irrliam_flavour.4.a"
		if = {
			limit = {
				has_country_flag = f37_cot_vassalize_flag
			}
			vassalize = U18
			every_known_country = {
				reverse_add_aggressive_expansion = {
					who = F37
					value = 5
					apply_calc = yes
				}
			}
		}
		else = {
			add_dip_power = 100
			kuz_mountains_area = {
				limit = {
					NOT = { country_or_non_sovereign_subject_holds = ROOT }
					NOT = { is_core = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
	}
}

	}
	
	option = {
		name = "irrliam_flavour.5.a"
		add_legitimacy = 10
		add_prestige = 10
		F26 = { add_liberty_desire = 10 }
	}
	
	option = {
		name = "irrliam_flavour.5.b"
		add_war_exhaustion = -2
		F26 = {
			set_country_flag = cot_phoenix_governorship_flag
			remove_government_reform = autocracy_reform
			add_government_reform = phoenix_servant_reform
			add_liberty_desire = -20
		}
		hidden_effect = {
			harpy_hills_region = {
				limit = { country_or_non_sovereign_subject_holds = ROOT }
				cede_province = F26
				add_permanent_claim = F26
				remove_core = ROOT
			}
		}
	}
}
