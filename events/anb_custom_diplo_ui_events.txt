country_event = { #Y20_feiten_buy_skyport_rights
	id = anb_custom_diplomacy.001
	title = feiten_buy_skyport_rights_title
	desc = feiten_buy_skyport_rights_desc
	picture = NOT_DISCIPLINED_SOLDIERS_eventPicture

	is_triggered_only = yes

	immediate = {
		# hidden_effect = {
		# 	feiten_skyport_ai_acceptance_effect = yes
		# 	if = {
		# 		limit = { check_variable = { ai_accept_var = 1 } }
		# 		set_country_flag = accept_feiten_buy_skyport_rights
		# 	}
		# 	else = {
		# 		set_country_flag = decline_feiten_buy_skyport_rights
		# 	}
		# }
		hidden_effect = {
			if = {
				limit = { feiten_skyport_ai_acceptance_trigger = yes }
				set_country_flag = accept_feiten_buy_skyport_rights
			}
			else = {
				set_country_flag = decline_feiten_buy_skyport_rights
			}
		}
	}
	after = {
		hidden_effect = {
			clr_country_flag = decline_feiten_buy_skyport_rights
			clr_country_flag = accept_feiten_buy_skyport_rights
		}
	}
	option = {
		name = anb_custom_diplomacy.decline
		trigger = {
			OR = {
				has_country_flag = decline_feiten_buy_skyport_rights
				ai = no
			}
		}
	}
	option = {
		name = anb_custom_diplomacy.accept
		trigger = {
			OR = {
				has_country_flag = accept_feiten_buy_skyport_rights
				ai = no
			}
		}
		event_target:feiten_or_was_feiten_target = {
			custom_tooltip = feitenConstructSkypostInfoShortTT
			add_treasury = -1000
			add_dip_power = -50
		}
		ROOT = {
			home_trade_node_effect_scope = {
				random_trade_node_member_province = {
					limit = {
						owned_by = ROOT
						province_has_center_of_trade_of_level = 1
					}
					add_province_triggered_modifier = feiten_skyport_handler_mod
				}
			}
		}
	}
}
