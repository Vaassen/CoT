# The desert wanderers.
country_event = {
	id = sarhaly_ogre_events.1
	title = sarhaly_ogre_events.1.t
	desc = sarhaly_ogre_events.1.d
	picture = AFRICAN_CARAVAN_eventPicture 
	is_triggered_only = yes
	immediate = {
		random_owned_province = {
			limit = {
				has_ogre_minority_trigger = yes
				OR = { 
					province_has_center_of_trade_of_level = 1
					OR = { 
						has_building = marketplace
						has_building = trade_depot
						has_building = stock_exchange
					}
				}
				NOT = { 
					has_province_modifier = ogre_trade_route_protect
					has_province_modifier = ogres_are_not_wanted
					has_province_modifier = gov_sp_ogres_protection 
				} 
			}
			save_event_target_as = sarhaly_ogre_event_prov
		}
	}
	option = {
		name = sarhaly_ogre_events.1.a
		add_country_modifier = {
			name = lethal_to_boredom
			duration = 3650
		}
		event_target:sarhaly_ogre_event_prov = {
			add_province_modifier = { 
				name = ogre_trade_route_protect
				duration = 3650
			}
		}
		if = { 
			limit = { 
				event_target:sarhaly_ogre_event_prov = { province_has_center_of_trade_of_level = 2 }
			} 
			add_treasury = 100 
		}
		else = { 
			add_treasury = -100 
			event_target:sarhaly_ogre_event_prov = { add_center_of_trade_level = 1 }
		}
		add_dip_power = -20
		small_increase_of_ogre_tolerance_effect = yes
	}
	option = {
		name = sarhaly_ogre_events.1.b
		add_country_modifier = {
			name = adventurers_over_ogres
			duration = 3650
		}
		event_target:sarhaly_ogre_event_prov = { 
			add_province_modifier = { 
				name = ogres_are_not_wanted
				duration = 3650
			}
		}
		small_decrease_of_ogre_tolerance_effect = yes
	}
	option = {
		name = sarhaly_ogre_events.1.c
		trigger = { 
			OR = { 
				ruler_has_personality = entrepreneur_personality
				ruler_has_personality = well_connected_personality 
			}
		}
		add_country_modifier = {
			name = adventurers_merchant_dilemma
			duration = 5475
		}
		event_target:sarhaly_ogre_event_prov = { 
			add_province_modifier = { 
				name =  gov_sp_ogres_protection
				duration = 5475
			}
		}
		# doesn't add level to center
		if = { 
			limit = { 
				event_target:sarhaly_ogre_event_prov = { province_has_center_of_trade_of_level = 3 }
			} 
			add_treasury = 400 
		}
		else_if = { 
			limit = {
				event_target:sarhaly_ogre_event_prov = { province_has_center_of_trade_of_level = 2 }
			} 
			add_treasury = -400 
			event_target:sarhaly_ogre_event_prov = { add_center_of_trade_level = 1 }
		}
		else = { 
			add_treasury = -100 
			event_target:sarhaly_ogre_event_prov = { add_center_of_trade_level = 1 }
		}
		add_dip_power = -40
		small_increase_of_ogre_tolerance_effect = yes
	}
}
