

namespace = flavor_khozrugan

#Chief Artug Ostokhan
country_event = {
	id = flavor_khozrugan.1
	title = flavor_khozrugan.1.t
	desc = flavor_khozrugan.1.d
	picture = FORT_eventPicture

	trigger = {
		tag = Z51
		has_regency = no
		has_consort_regency = no
		is_lesser_in_union = no
		NOT = { has_country_flag = ostokhan_started }
	}

	immediate = { set_country_flag = ostokhan_started }	

	#Give up the throne
	option = {
		name = flavor_khozrugan.1.a
		ai_chance = {
			factor = 80
		}
		add_prestige = -50
		reduce_stability_or_adm_power = yes
		add_dip_power = 100
		define_ruler = {
			name = "Artug"
			dynasty = "Ostokhan"
			ADM = 5
			DIP = 5
			MIL = 6
			age = 21
		}
		if = {
			limit = {
				has_dlc = "Rights of Man" 
			}
			hidden_effect = { add_ruler_personality = bold_fighter_personality }
		}
		define_heir = {
			name = "Ornog"
			dynasty = "Ostokhan"
			culture = green_orc
			change_adm = 4
			change_dip = 4
			change_mil = 5
			age = 0
		}
		add_estate_monstrous_tribes_loyalty_effect = yes
		set_country_flag = ostokhan_event
	}
	
	#Fight! (lose)
	option = {
		name = flavor_khozrugan.1.b
		trigger = {
			NOT = { mil = 6 }
		}
		ai_chance = {
			factor = 20 
		}
		custom_tooltip = lets_fight_tooltip
		hidden_effect = {
			country_event = { id = flavor_khozrugan.2 }
		}
	}
	
	#Fight! (win)
	option = {
		name = flavor_khozrugan.1.c
		trigger = {
			mil = 6
		}
		ai_chance = {
			factor = 20 
		}
		custom_tooltip = lets_fight_tooltip
		hidden_effect = {
			country_event = { id = flavor_khozrugan.3 }
		}
	}
}

country_event = {
	id = flavor_khozrugan.5
	title = flavor_khozrugan.5.t
	desc = flavor_khozrugan.5.d
	picture = FORT_eventPicture
	
	is_triggered_only = yes
	
	trigger = {
		always = yes
	}
	
	#Ascension
	option = {
		name = flavor_khozrugan.5.a
		if = {
			limit = {
				AND = {
					431 = { has_province_modifier = cot_shrine_to_dookan_adm }
					221 = { has_province_modifier = cot_shrine_to_dookan_dip }
					OR = {
						308 = { has_province_modifier = cot_shrine_to_dookan }
						832 = { has_province_modifier = cot_shrine_to_dookan }
					}
				}
			}
			add_country_modifier = {
				name = Z51_cot_show_em
				duration = -1
			}
			every_owned_province = {
				limit = {
					has_province_modifier = cot_searched_for_dookan
				}
				province_event = { id = flavor_khozrugan.6 }
			}
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						431 = { has_province_modifier = cot_shrine_to_dookan_adm }
						221 = { has_province_modifier = cot_shrine_to_dookan_dip }
					}
					AND = {
						431 = { has_province_modifier = cot_shrine_to_dookan_adm }
						OR = {
							308 = { has_province_modifier = cot_shrine_to_dookan }
							832 = { has_province_modifier = cot_shrine_to_dookan }
						}
					}
					AND = {
						221 = { has_province_modifier = cot_shrine_to_dookan_dip }
						OR = {
							308 = { has_province_modifier = cot_shrine_to_dookan }
							832 = { has_province_modifier = cot_shrine_to_dookan }
						}
					}
				}
			}
			add_country_modifier = {
				name = Z51_cot_show_some
				duration = -1
			}
			every_owned_province = {
				limit = {
					has_province_modifier = cot_searched_for_dookan
				}
				province_event = { id = flavor_khozrugan.6 }
			}
		}
		else_if = {
			limit = {
				OR = {
					431 = { has_province_modifier = cot_shrine_to_dookan_adm }
					221 = { has_province_modifier = cot_shrine_to_dookan_dip }
					OR = {
						308 = { has_province_modifier = cot_shrine_to_dookan }
						832 = { has_province_modifier = cot_shrine_to_dookan }
					}
				}
			}
			add_country_modifier = {
				name = Z51_cot_show_a_little
				duration = -1
			}
			every_owned_province = {
				limit = {
					has_province_modifier = cot_searched_for_dookan
				}
				province_event = { id = flavor_khozrugan.6 }
			}
		}
		else = { #failsafe
			add_mil_power = 50
		}
	}
}
