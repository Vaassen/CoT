namespace = utubodi_unification

country_event = { #We all want Kakunor!
	id = utubodi_unification.0
	title = utubodi_unification.0.t
	desc = utubodi_unification.0.d
	picture = COMET_SIGHTED_eventPicture

	is_triggered_only = yes

	trigger = { capital_scope = { region = great_plateau_region } }

	immediate = {
		hidden_effect = {
			set_variable = {
				which = utubodiKakunorGrainVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorLivestockVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorExWoodVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorWoolVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorIronVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorIncenseVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorFurVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorCopperVar
				value = 0
			}
			set_variable = {
				which = utubodiKakunorDamestearVar
				value = 0
			}
			3675 = { save_event_target_as = event_province }
		}
	}

	goto = event_province

	option = { #We want Kakunor NOW!
		name = utubodi_unification.0.a
		trigger = { NOT = { owns = 3675 } }
	}

	option = { #Wait, shit, that's us!
		name = utubodi_unification.0.b
		trigger = { owns = 3675 }
	}
}

country_event = { #We got Kakunor
	id = utubodi_unification.1
	title = utubodi_unification.1.t
	desc = utubodi_unification.1.d
	picture = COMET_SIGHTED_eventPicture

	is_triggered_only = yes

	option = {
		name = utubodi_unification.1.a
		hidden_effect = {
			if = {
				limit = { has_country_flag = kakunorTier1flag }
				set_great_project_tier_1 = {
					type = kakunor_cloud_city
				}
			}
			if = {
				limit = { has_country_flag = kakunorTier2flag }
				set_great_project_tier_2 = {
					type = kakunor_cloud_city
				}
			}
			if = {
				limit = { has_country_flag = kakunorTier3flag }
				set_great_project_tier_3 = {
					type = kakunor_cloud_city
				}
			}
		}
	}
}

country_event = { #Collapsed Scaffold
	id = utubodi_unification.2
	title = utubodi_unification.2.t
	desc = utubodi_unification.2.d
	picture = COMET_SIGHTED_eventPicture

	is_triggered_only = yes

	trigger = {
		has_construction = kakunor_cloud_city
	}

	option = { #Double the shifts
		name = utubodi_unification.2.a
		reduce_stability_or_adm_power = yes
		capital_scope = {
			add_province_modifier = {
				name = utubodi_unification_unrest
				duration = -1
			}
		}
		add_government_power = {
			mechanic_type = utubodi_unification_mechanic
			power_type = utubodi_unification_power
			value = 15
		}
	}

	option = { #Train the workers
		name = utubodi_unification.2.b
		add_stability_or_adm_power = yes
		add_government_power = {
			mechanic_type = utubodi_unification_mechanic
			power_type = utubodi_unification_power
			value = -15
		}
	}
}
