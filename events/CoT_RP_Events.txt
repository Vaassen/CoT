#Cyranvar event I
country_event = {
	id = cot_misc_events.1
	title = cot_misc_events.1.t
	desc = cot_misc_events.1.d
	picture = BIG_STAR_IN_SKY_eventPicture

	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = I45
		OR = {
			ayeth_area = {
				type = all
				country_or_non_sovereign_subject_owns = ROOT
			}
			lonfarch_area = {
				type = all
				country_or_non_sovereign_subject_owns = ROOT
			}
		}
		has_country_modifier = cot_i45_mercs_I
	}
	
	option = {
		name = cot_misc_events.1.a
		if = {
			limit = {
				ayeth_area = {
					type = all
					country_or_non_sovereign_subject_owns = ROOT
				}
			}
			ayeth_area = {
				cede_province = B13
				add_core = B13
			}
		}
		else_if = {
			limit = {
				lonfarch_area = {
					type = all
					country_or_non_sovereign_subject_owns = ROOT
				}
			}
			lonfarch_area = {
				cede_province = B13
				add_core = B13
			}
		}
		B13 = {
			add_historic_friend = ROOT
		}
		add_historic_friend = B13
		create_march = B13
		remove_country_modifier cot_i45_mercs_I
		add_country_modifier = cot_i45_mercs_II
	}
}
