			duration = 73000 #20 years
		}
	}
}

#Cyranvar event I
country_event = {
	id = cot_misc_events.1
	title = cot_misc_events.1.t
	desc = cot_misc_events.1.d
	picture = BIG_STAR_IN_SKY_eventPicture

	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = I45
		OR = {
			ayeth_area = {
				type = all
				country_or_non_sovereign_subject_owns = ROOT
			}
			lonfarch_area = {
				type = all
				country_or_non_sovereign_subject_owns = ROOT
			}
		}
		has_country_modifier = cot_i45_mercs_I
	}
	
	option = {
		name = cot_misc_events.1.a
		if = {
			limit = {
				ayeth_area = {
					type = all
					country_or_non_sovereign_subject_owns = ROOT
				}
			}
			ayeth_area = {
				cede_province = B13
				add_core = B13
			}
		}
		else_if = {
			limit = {
				lonfarch_area = {
					type = all
					country_or_non_sovereign_subject_owns = ROOT
				}
			}
			lonfarch_area = {
				cede_province = B13
				add_core = B13
			}
		}
		B13 = {
			add_historic_friend = ROOT
		}
		add_historic_friend = B13
		create_march = B13
		remove_country_modifier cot_i45_mercs_I
		add_country_modifier = cot_i45_mercs_II
	}
}

#Ilyeth Event
country_event = {
	id = cot_misc_events.3
	title = cot_misc_events.1.t
	desc = cot_misc_events.1.d
	picture = BIG_STAR_IN_SKY_eventPicture

	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = X10
	}
	
	option = {
		X10 = {
			every_owned_province = {
				limit = { religion = elven_forebears }
				add_core = ROOT
			}
			change_dip = 2
			change_mil = 2
			add_ruler_modifier = {
				name = cot_x10_lenros_II
				duration = -1
			}
			every_owned_province = {
				limit = {
					has_terrains = woods
				}
				remove_province_modifier = cot_newtowns
				add_province_modifier = {
					name = cot_newtowns
					duration = -1
				}
			}
		}
	}
}
