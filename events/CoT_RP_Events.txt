country_event = {
	id = cot_rp_events.3
	title = cot_rp_events.1.t
	desc = cot_rp_events.1.d
	picture = BIG_STAR_IN_SKY_eventPicture

	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = I45
		ayeth_area = {
			type = all
			country_or_non_sovereign_subject_owns = ROOT
		}
		lonfarch_area = {
			type = all
			country_or_non_sovereign_subject_owns = ROOT
		}
		has_country_modifier = cot_i45_mercs_II
	}
	
	option = {
		name = cot_rp_events.1.a
		if = {
			limit = {
				ayeth_area = {
					type = all
					country_or_non_sovereign_subject_owns = B13
				}
			}
			lonfarch_area = {
				cede_province = B08
				add_core = B08
			}
		}
		else_if = {
			limit = {
				lonfarch_area = {
					type = all
					country_or_non_sovereign_subject_owns = B13
				}
			}
			ayeth_area = {
				cede_province = B08
				add_core = B08
			}
		}
		B08 = {
			add_historic_friend = ROOT
			add_historic_friend = B13
		}
		B13 = {
			add_historic_friend = B08
		}
		add_historic_friend = B08
		create_march = B08
		remove_country_modifier = cot_i45_mercs_II
		add_country_modifier = {
			name = cot_i45_mercs_III
			duration = -1
		}
	}
}

country_event = {
	id = cot_rp_events.7
	title = cot_rp_events.7.t
	desc = cot_rp_events.7.d
	picture = BIG_STAR_IN_SKY_eventPicture
		fire_only_once = yes
		trigger = {
		OR = {
			check_variable = {
				which = marik_progress_cot
				value = 4
			}
			any_province = {
				NOT = {
					has_province_modifier = cot_hunt_for_marik
				}
			}
		}
	}
		option = {
		name = cot_rp_events.7.a
		if = {
   			limit = {
       			check_variable = {
					which = marik_progress_cot
					value = 4
					}
   			}
			add_stability = 3
			set_country_flag = killed_marik_cot
		}
		else_if = {
   			limit = {
      				AND = {
						check_variable = {
						which = marik_progress_cot
						value = 2
					}
					NOT = { 
						check_variable = {
							which = marik_progress_cot
							value = 4
						}
					}
				}
   			}
			set_country_flag = deal_with_marik_cot
			define_general = {
  				shock = 3
    			fire = 5
   				manuever = 1
  				siege = 4
				name = "Marik Vaulrest"
			}
		}
		else = {
			define_ruler = {
   				name = "Marik Vaulrest"
   				adm = 6
  				dip = 1
   				mil = 4
   				age = 37
   				claim = 100
			}
			1093 = {
				remove_province_modifier = cot_bloom_unchecked
				add_permanent_province_modifier = {
					name = cot_heart_of_bloom
					duration = -1
				}
			}
			add_ruler_personality = sinner_personality
			add_ruler_personality = secretive_personality
		}
	}
}
