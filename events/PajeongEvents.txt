
namespace = pajeong

country_event = { #Disaster Start
	id = pajeong.1
	title = pajeong.1.t
	desc = pajeong.1.desc
	picture = east_asian_COLONIZATION_eventPicture
	
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			set_country_flag = pajeong_disaster_active_flag
			export_to_variable = {
				which = pajeong_crownland
				value = trigger_value:crown_land_share
			}
			divide_variable = {
				which = pajeong_crownland
				value = 4
			}
			set_estate_privilege = estate_burghers_disobedience_of_the_merchants
			set_estate_privilege = estate_nobles_disobedience_of_the_hwalbaek
			set_estate_privilege = estate_church_disobedience_of_the_neoksamaji
		}
	}
	
	option = {
		name = pajeong.1.a
		hidden_effect = {
			for_variable_amount = {
				variable = pajeong_crownland
				effect = "
					change_estate_land_share = {
						estate = estate_burghers
						share = 1
					}
				"
			}
			for_variable_amount = {
				variable = pajeong_crownland
				effect = "
					change_estate_land_share = {
						estate = estate_nobles
						share = 1
					}
				"
			}
			for_variable_amount = {
				variable = pajeong_crownland
				effect = "
					change_estate_land_share = {
						estate = estate_church
						share = 1
					}
				"
			}
		}
		custom_tooltip = pajeong_crownland_tt
		custom_tooltip = pajeong_privileges_tt
		custom_tooltip = pajeong_explaination_tt
	}
}

country_event = { #Disaster End
	id = pajeong.2
	title = pajeong.2.t
	desc = pajeong.2.desc
	picture = east_asian_COLONIZATION_eventPicture
	
	is_triggered_only = yes
	
	option = {
		name = pajeong.2.a
		if = {
			limit = { has_estate_privilege = estate_burghers_disobedience_of_the_merchants }
			release = V27 #Nuwenara
			hidden_effect = {
				V27 = {
					country_event = { id = anbennar_setup.3 }
					add_treasury = 100
					add_adm_power = 100
					add_dip_power = 100
					add_mil_power = 100
					add_manpower = 10
					capital_scope = { build_to_forcelimit = { infantry = 0.7 cavalry = 0.2 } }
				}
			}
		}
		if = {
			limit = { has_estate_privilege = estate_nobles_disobedience_of_the_hwalbaek }
			release = Y94 #Dalmangaji
			hidden_effect = {
				Y94 = {
					country_event = { id = anbennar_setup.3 }
					add_treasury = 100
					add_adm_power = 100
					add_dip_power = 100
					add_mil_power = 100
					add_manpower = 10
					capital_scope = { build_to_forcelimit = { infantry = 0.7 cavalry = 0.2 } }
				}
			}
		}
		if = {
			limit = { has_estate_privilege = estate_church_disobedience_of_the_neoksamaji }
			release = Y95 #Baegye
			hidden_effect = {
				Y95 = {
					country_event = { id = anbennar_setup.3 }
					add_treasury = 100
					add_adm_power = 100
					add_dip_power = 100
					add_mil_power = 100
					add_manpower = 10
					capital_scope = { build_to_forcelimit = { infantry = 0.7 cavalry = 0.2 } }
				}
			}
		}
		if = {
			limit = {
				calc_true_if = {
					amount = 2
					has_estate_privilege = estate_burghers_disobedience_of_the_merchants
					has_estate_privilege = estate_nobles_disobedience_of_the_hwalbaek
					has_estate_privilege = estate_church_disobedience_of_the_neoksamaji
				}
				NOT = {
					AND = {
						has_estate_privilege = estate_burghers_disobedience_of_the_merchants
						has_estate_privilege = estate_nobles_disobedience_of_the_hwalbaek
						has_estate_privilege = estate_church_disobedience_of_the_neoksamaji
					}
				}
			}
			change_estate_land_share = {
				estate = all
				share = -5
			}
			add_prestige = 10
		}
		else_if = {
			limit = {
				OR = {
					has_estate_privilege = estate_burghers_disobedience_of_the_merchants
					has_estate_privilege = estate_nobles_disobedience_of_the_hwalbaek
					has_estate_privilege = estate_church_disobedience_of_the_neoksamaji
				}
			}
			change_estate_land_share = {
				estate = all
				share = -10
			}
			add_prestige = 10
			add_stability = 1
		}
		else = {
			change_estate_land_share = {
				estate = all
				share = -15
			}
			add_prestige = 10
			add_stability = 1
			add_ruler_modifier = {
				name = pajeong_united_court
				duration = -1
			}
		}
	}
}
