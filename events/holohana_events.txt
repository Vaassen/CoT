#Maliu Fanua Controller
country_event = {
	id = holohana_events.99
	title = holohana_events.0.t
	desc = holohana_events.0.d
	picture = BAD_WITH_MONARCH_eventPicture
	
	hidden = yes
	is_triggered_only = yes

	immediate = {
		holohana_migrate_timer_effect = { holohana = maliu_fanua}
	}

	option = {
		name = holohana_events.99.a
		ai_chance = { factor = 100 }
		trigger  = {
			event_target:holohana_target_maliu_fanua = {
				holohana_migration_start = { holohana = maliu_fanua}
			}
		}
		event_target:holohana_target_maliu_fanua = {
			province_event = {
				id = holohana_events.2
			}
			province_event = {
				id = holohana_events.999
				days = 20
			}
		}
	}

	option = {
		name = holohana_events.99.b
		ai_chance = { factor = 100 }
		trigger  = {
			NOT = {
				event_target:holohana_target_maliu_fanua = {
					holohana_migration_start = { holohana = maliu_fanua}
				}
			}
		}
	}

}
