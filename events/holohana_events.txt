namespace = holohana_events

	id = holohana_events.0
	title = holohana_events.0.t
	desc = holohana_events.0.d
	picture = BAD_WITH_MONARCH_eventPicture

	is_triggered_only = yes

	trigger = {
		owns = 6234
	}

	immediate = {
		holohana_province_save_effect = { holohana = the_watcher }
	}

	option = {
		name = holohana_events.0.a
		ai_chance = { factor = 100 }
		holohana_spawn_effect = { holohana = the_watcher leader_the_watcher = yes}
	}

country_event = {
	id = holohana_events.1
	title = holohana_events.1.t
	desc = holohana_events.1.d
	picture = BAD_WITH_MONARCH_eventPicture

	is_triggered_only = yes

	trigger = {
		owns = 6234
	}

	immediate = {
		random_province = {
			limit = {
				has_province_modifier = holohana_modifier_the_watcher
			}
			save_event_target_as = holohana_target_the_watcher
			remove_province_modifier = holohana_modifier_the_watcher
			clear_rebels = yes
			if = {
				limit = {
					NOT = { is_empty = yes }
				}
				owner = {
					save_event_target_as = holohana_province_owner_the_watcher
				}
			}
		}
	}

	option = {
		name = holohana_events.1.a
		ai_chance = { factor = 3 }
		event_target:holohana_target_the_watcher = {
			if = {
				limit = {
					any_empty_neighbor_province = {
						is_valid_holohana_move_target = yes
					}
				}
				random_empty_neighbor_province = {
					add_permanent_province_modifier = {
						name = holohana_modifier_the_watcher
						duration = -1
					}
				}
			}
			else = {
				add_permanent_province_modifier = {
					name = holohana_modifier_the_watcher
					duration = -1
				}
			}
		}
	}

	option = {
		name = holohana_events.1.b
		trigger = {
			event_target:holohana_target_the_watcher = {
				any_neighbor_province = { 
					is_city = yes
				}
			}
		}
		ai_chance = { factor = 1 }
		event_target:holohana_target_the_watcher = {
			if = {
				limit = {
					any_neighbor_province = {
						is_valid_holohana_move_target = yes
					}
				}
				random_neighbor_province = {
					add_permanent_province_modifier = {
						name = holohana_modifier_the_watcher
						duration = -1
					}
				}
			}
			else = {
				add_permanent_province_modifier = {
					name = holohana_modifier_the_watcher
					duration = -1
				}
			}
		}
	}

	after = {
		oceania = {
			limit = {
				has_province_modifier = holohana_rumbling
				OR = {
					NOT = {
						any_neighbor_province = {
							has_holohana_province_modifier = yes
						}
						any_empty_neighbor_province = {
							has_holohana_province_modifier = yes
						}
					}
					has_holohana_province_modifier = yes
				}
			}
			remove_province_modifier = holohana_rumbling
		}
		country_event = { 
    		id = holohana_events.0
    		days = 1
		}
	}
}
