vacyntassi_incident_slot_1 = {
	slot = 1
	generic = no
	ai = no
	potential = {
		tag = F31 #Eduz-Vacyn
		NOT = { is_incident_happened = incident_thirteenth_night }
	}
	has_country_shield = yes

	#Summon the Zanites
	vacyntassi_summon_zanites = {
		icon = mission_confrontation
		required_missions = { vacyntassi_ashen_phoenix }
		position = 2
		
		provinces_to_highlight = {
			culture = zanite
			owner = {
				primary_culture = zanite
			}
			is_capital = yes
		}
		
		trigger = {
			OR = {
				advisor = diplomat
				advisor = statesman
			}
			calc_true_if = {
				all_known_country = {
					has_opinion = {
						who = ROOT
						value = 100
					}
					primary_culture = zanite
				}
				amount = 2
			}
		}
		
		effect = {
			if = { limit = { NOT = { mission_completed = vacyntassi_enbolden_eledas } }
				custom_tooltip = vacyntassi_completing_enbolden_eledas_last_effect_tt
				custom_tooltip = nsc_chosen_role_down_tt
				add_incident_variable_value = { incident = incident_thirteenth_night value = 1 }
				add_manpower = 5
			}
			add_country_modifier = {
				name = vacyntassi_summon_zanites_modifier
				duration = 10950
			}
		}
	}
}
