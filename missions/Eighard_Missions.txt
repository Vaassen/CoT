H14_missions_refusal_1 = { #Adds generic Tuathak missions if you refuse the cult of rot
	slot = 1
	generic = no
	ai = yes
	has_country_shield = yes

	potential = {
		tag = H14
		has_country_flag = H14_refused_rot
	}

	tuathak_prepare_praisegranting_festival = {
		icon = mission_rb_colonise_spice_islands
		required_missions = { }
		position = 5
		
		provinces_to_highlight = {
			OR = {
				AND = {
					area_for_scope_province = { is_capital_of = ROOT }
					NOT = { owned_by = ROOT }
				}
				AND = {
					is_capital_of = ROOT
					NOT = { has_tax_building_trigger = yes }
				}
			}
		}
		
		trigger = {
			capital_scope = {
				area_for_scope_province = {
					type = all
					owned_by = ROOT 
				}
				has_tax_building_trigger = yes
			}
			any_known_country = {
				OR = {
					capital_scope = {
						continent = europe
					}
					colonial_parent = {
						capital_scope = {
							continent = europe
						}
					}
				}
			}
			treasury = 60
		}
		
		effect = {
			add_treasury = -60
			add_dip_power = 25
			add_country_modifier = {
				name = "tuathak_festival_preparations"
				duration = 9125
			}
		}
	}
	
	tuathak_host_the_festival = {
		icon = mission_discussion_arch
		required_missions = { tuathak_prepare_praisegranting_festival }
		position = 6
		
		trigger = {
			num_of_allies = 2
			OR = {
				num_accepted_cultures = 2
				government_rank = 3
			}
			dip_power = 100
		}
		
		effect = {
			add_dip_power = -100
			country_event = { id = eordand.17 }
			remove_country_modifier = tuathak_festival_preparations
		}
	}
	

}
