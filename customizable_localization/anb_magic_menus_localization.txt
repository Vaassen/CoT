# you must define custom loc for 'getAelantirSlot3Name' and 'getAelantirSlot3Desc'
# if a different spell list reuses that spell, it needs an entry with the flag for that spell list

        trigger = { has_custom_spell_list_in_group = { group = template subgroup = tag_specific_example } } #XXDAXAAA
