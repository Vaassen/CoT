country_decisions = {
	fight_qeyikul_warband = {
		potential = {
			OR = {
				culture = guryadagga
				any_owned_province = { has_qeyikul_warband = yes }
			}
		}
		major = yes
		allow = {
			any_owned_province = { has_qeyikul_warband = yes }
			is_at_war = no
			NOT = { has_country_flag = fighting_qeyikul_warband }
			army_size = 1
		}
		provinces_to_highlight = {
			has_qeyikul_warband = yes
		}
		effect = {
			if = {
				limit = {
					calc_true_if = { ## Settled nation with warbands
						all_owned_province = { has_qeyikul_warband = yes }
						amount = 2
					}
				}
				country_event = { id = flavour_AYN.1000 }
			}
			else = {
				every_owned_province = {
					limit = { has_qeyikul_warband = yes }
					spawn_qeyikul_warband_rebels = yes
				}
			}
		}
	}
}
