country_decisions = {
	fight_qeyikul_warband = {
		potential = {
			hidden_trigger = { ai = no }
			OR = {
				culture = guryadagga
				any_owned_province = { has_qeyikul_warband = yes }
			}
		}
		major = yes
		allow = {
			hidden_trigger = { ai = no } #Colonisers get it removed when the colonise the province
			any_owned_province = { has_qeyikul_warband = yes }
			is_at_war = no
			NOT = { has_country_flag = fighting_qeyikul_warband }
			army_size = 1
		}
		provinces_to_highlight = {
			has_qeyikul_warband = yes
		}
		effect = {
			if = {
				limit = {
					num_of_owned_provinces_with = { value = 2 has_qeyikul_warband = yes }
				}
				country_event = { id = flavour_AYN.1000 }
			}
			else = {
				random_owned_province = {
					limit = { has_qeyikul_warband = yes }
					spawn_qeyikul_warband_rebels = yes
				}
			}
		}
	}

	S88_migrate_to_island = {

					has_country_modifier = S88_ai_slow_sea_migrate
				}
			}
			capital_scope = { has_port = yes }

			}
			hidden_effect = {

						name = S88_ai_slow_sea_migrate
						duration = 1825
					}
				}
			}

				id = flavour_AYN.400
			}
		}
