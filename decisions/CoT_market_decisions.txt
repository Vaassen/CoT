country_decisions = {
	open_market_menu = {
		color = { 48 166 166 }
		potential = {
			ai = no
			NOT = { has_ruler_flag = market_menu_opened }
		}
		
		allow = {
			ai = no
		}
		effect = {
			custom_tooltip = open_market_menu_tt
			hidden_effect = { country_event = { id = cot_market_events.2 } }
			open_single_menu = yes	#prevents people from opening the menu multiple times
		}
	}

	visit_a_market = {
		color = { 48 166 166 }
		potential = {
			ai = no
		}

		allow = {
			custom_trigger_tooltip = {
				tooltip = visited_market_recently
				OR = {
					had_country_flag = {
						flag = visited_market
						days = 3652
					}
					NOT = {
						has_country_flag = visited_market
					}
				}
			}
			if = {
				limit = {
					has_country_modifier = monstrous_nation
				}
				check_variable = {
					which = CoT_monster_power
					value = 0.1
				}
			}
			else = {
				check_variable = {
					which = CoT_market_power
					value = 0.1
				}
			}
			OR = { #MAKE IT SO ONLY VISIT MARKET IF U HAVE ONE  OF THESE CONDITIONS
				any_owned_province = {
					OR = {
						has_province_modifier = CoT_grain_market
						has_province_modifier = CoT_glass_market
						has_province_modifier = CoT_gold_market
						has_province_modifier = CoT_gems_market
						has_province_modifier = CoT_fish_market
						has_province_modifier = CoT_cloth_market
						has_province_modifier = CoT_copper_market
						has_province_modifier = CoT_cattle_market
						has_province_modifier = CoT_wool_market
						has_province_modifier = CoT_spice_market
						has_province_modifier = CoT_salt_market
						has_province_modifier = CoT_dyes_market
						has_province_modifier = CoT_paper_market
						has_province_modifier = CoT_incense_market
						has_province_modifier = CoT_slave_market
					}
				}
				AND = { 
					adm_power = 200
					mil_power = 200
					dip_power = 200
				}
			}
			adm_power = 200
			mil_power = 200
			dip_power = 200
			
		}
		effect = {
			custom_tooltip = visit_a_market_tt
			hidden_effect = { country_event = { id = cot_market_events.5 } }
			clr_country_flag = visited_market 
			set_country_flag = visited_market 
			if = {
				limit = {
					NOT = {
						has_country_modifier = monstrous_nation
					}
				}
				add_adm_power = -25
				add_dip_power = -25
			}
		}
	}

	trade_inert_staff = {
		potential = {
			OR = {
				has_country_modifier = inert_staff_one
				has_country_modifier = inert_staff_two
				has_country_modifier = inert_staff_three
			}
			OR = {
				has_estate = estate_mages
				ruler_has_personality = mage_personality
			}
		}
		allow = {
			ai = no
		}
		effect = {
			update_magic_study_experience_modifier = { value = 5 }
			remove_country_modifier = inert_staff_one
			remove_country_modifier = inert_staff_two
			remove_country_modifier = inert_staff_three
		}
	}

	sell_trade_goods = {
		potential = {
			has_country_flag = has_trade_good_market_item
		}
		allow = {
			ai = no
		}
		effect = {
			custom_tooltip = sell_trade_goods_tt
			hidden_effect = { country_event = { id = cot_market_events.24 } }
			open_single_menu = yes
		}
	}

	show_market_locations = {
		potential = {
			ai = no

		provinces_to_highlight = {
			OR = {
				province_id = 603

				province_id = 7964

				province_id = 584
				province_id = 522
				province_id = 512
				province_id = 6760
				province_id = 7811
				province_id = 625
				province_id = 578
				province_id = 7803
			}
		}
		allow = {
			ai = yes 
		}
	}
